# [3] settings panel

Below is a complete updated single-file `app.py` that adds:

- a settings panel with sliders for:
  - `temperature`
  - `top_p`
  - `top_k`
  - `min_p`
  - `repeat_penalty`
  - `max_tokens`
- a system prompt textbox
- all settings passed through to the llama.cpp request
- non-standard OpenAI schema parameters sent through `extra_body`

The persistent JSON chat sessions, session dropdown, **New Chat**, and **Delete Chat** remain included.

## 2. Save this as `app.py`

```python
import json
import os
import re
import tempfile
import threading
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional, Tuple

import gradio as gr
from openai import OpenAI


# -----------------------------------------------------------------------------
# llama.cpp OpenAI-compatible server settings
# -----------------------------------------------------------------------------

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("LLAMA_CPP_API_KEY", "EMPTY")

# Optional. If unset, the app tries to read the model name from /v1/models.
MODEL = os.getenv("LLAMA_CPP_MODEL", "")

# Default generation settings. These are also exposed in the UI and can be
# changed per request.
DEFAULT_SYSTEM_PROMPT = os.getenv(
    "SYSTEM_PROMPT",
    "You are a helpful assistant. Use the prior conversation as context.",
)

TEMPERATURE = float(os.getenv("TEMPERATURE", "0.7"))
TOP_P = float(os.getenv("TOP_P", "1.0"))
TOP_K = int(os.getenv("TOP_K", "40"))
MIN_P = float(os.getenv("MIN_P", "0.0"))
REPEAT_PENALTY = float(os.getenv("REPEAT_PENALTY", "1.05"))
MAX_TOKENS = int(os.getenv("MAX_TOKENS", "1024"))
TIMEOUT = float(os.getenv("LLAMA_CPP_TIMEOUT", "600"))

# Local persistent storage for chat sessions.
CHATS_DIR = Path(os.getenv("CHATS_DIR", "./chats"))
CHATS_DIR.mkdir(parents=True, exist_ok=True)


client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=TIMEOUT,
)


_cached_model: Optional[str] = None

_lock = threading.Lock()
_stream_generation = 0


# -----------------------------------------------------------------------------
# Model discovery
# -----------------------------------------------------------------------------


def resolve_model() -> str:
    """
    Prefer LLAMA_CPP_MODEL if set.
    Otherwise try to discover the first model from llama.cpp's /v1/models endpoint.
    Fallback to "local-model", which many llama.cpp OpenAI servers accept.
    """
    global _cached_model

    if MODEL:
        return MODEL

    if _cached_model:
        return _cached_model

    try:
        response = client.models.list()
        models = getattr(response, "data", None)

        if models:
            _cached_model = models[0].id
            return _cached_model
    except Exception:
        pass

    return "local-model"


# -----------------------------------------------------------------------------
# Stream cancellation helpers
# -----------------------------------------------------------------------------


def next_stream_generation() -> int:
    """
    Increment the global stream generation.

    Long-running streamed responses check this value and stop updating
    if the user starts a new stream, switches sessions, deletes a session,
    or starts a new chat.
    """
    global _stream_generation

    with _lock:
        _stream_generation += 1
        return _stream_generation


def current_stream_generation() -> int:
    with _lock:
        return _stream_generation


# -----------------------------------------------------------------------------
# Chat persistence helpers
# -----------------------------------------------------------------------------


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def safe_session_id(session_id: Any) -> str:
    """
    Make a filesystem-safe session id.
    UUIDs are already safe; this also protects against odd inputs.
    """
    sid = re.sub(r"[^A-Za-z0-9_-]", "", str(session_id))

    if not sid:
        sid = str(uuid.uuid4())

    return sid[:128]


def session_path(session_id: str) -> Path:
    return CHATS_DIR / f"{safe_session_id(session_id)}.json"


def make_title(text: str) -> str:
    text = " ".join((text or "").strip().split())

    if not text:
        return "New chat"

    if len(text) > 48:
        return text[:47].rstrip() + "…"

    return text


def normalize_messages(messages: Any) -> List[Dict[str, str]]:
    """
    Return a clean list of chatbot messages:

        {
            "role": "user" | "assistant",
            "content": str
        }

    Gradio may sometimes produce richer message objects. This keeps the
    stored JSON and API payload simple and predictable.
    """
    out: List[Dict[str, str]] = []

    if not isinstance(messages, list):
        return out

    for msg in messages:
        if not isinstance(msg, dict):
            continue

        role = msg.get("role")
        content = msg.get("content")

        if role not in {"user", "assistant"}:
            continue

        if content is None:
            content = ""

        if not isinstance(content, str):
            # If Gradio gives a list of content blocks, flatten visible text.
            if isinstance(content, list):
                parts: List[str] = []

                for part in content:
                    if isinstance(part, dict):
                        if isinstance(part.get("text"), str):
                            parts.append(part["text"])
                        elif isinstance(part.get("path"), str):
                            parts.append(f"[file: {part['path']}]")
                    elif isinstance(part, str):
                        parts.append(part)

                content = "\n".join(parts)
            else:
                content = str(content)

        out.append(
            {
                "role": role,
                "content": content,
            }
        )

    return out


def copy_messages(messages: List[Dict[str, str]]) -> List[Dict[str, str]]:
    """
    Return a shallow copy of messages.
    This prevents Gradio from accidentally mutating shared state.
    """
    return [dict(msg) for msg in messages]


def new_session() -> Dict[str, Any]:
    """
    Create an empty session.

    It is not saved until the first user message is sent.
    """
    return {
        "id": str(uuid.uuid4()),
        "title": "New chat",
        "created_at": now_iso(),
        "updated_at": now_iso(),
        "messages": [],
    }


def normalize_session(session: Any) -> Dict[str, Any]:
    if not isinstance(session, dict) or not session.get("id"):
        return new_session()

    out = dict(session)

    out["id"] = safe_session_id(out.get("id"))
    out["title"] = str(out.get("title") or "New chat")
    out["created_at"] = str(out.get("created_at") or now_iso())
    out["updated_at"] = str(out.get("updated_at") or now_iso())
    out["messages"] = normalize_messages(out.get("messages"))

    return out


def save_session(session: Dict[str, Any]) -> Dict[str, Any]:
    """
    Atomically save one session JSON file.

    Stored structure:
        {
            "id": "...",
            "title": "...",
            "created_at": "ISO timestamp",
            "updated_at": "ISO timestamp",
            "messages": [
                {"role": "user", "content": "..."},
                {"role": "assistant", "content": "..."}
            ]
        }
    """
    session = normalize_session(session)
    session["updated_at"] = now_iso()

    path = session_path(session["id"])
    tmp_path: Optional[Path] = None

    try:
        with tempfile.NamedTemporaryFile(
            mode="w",
            dir=CHATS_DIR,
            prefix=f".{session['id']}.",
            suffix=".json.tmp",
            delete=False,
            encoding="utf-8",
        ) as tmp_file:
            tmp_path = Path(tmp_file.name)
            json.dump(session, tmp_file, ensure_ascii=False, indent=2)

        os.replace(tmp_path, path)

    except Exception as exc:
        print(f"Failed to save session {session.get('id')}: {exc}")

    finally:
        if tmp_path is not None:
            try:
                tmp_path.unlink(missing_ok=True)
            except Exception:
                pass

    return session


def load_session(session_id: str) -> Dict[str, Any]:
    if not session_id:
        return new_session()

    sid = safe_session_id(session_id)
    path = session_path(sid)

    try:
        if path.exists():
            data = json.loads(path.read_text(encoding="utf-8"))

            if not isinstance(data, dict):
                data = {}

            data.setdefault("id", sid)

            session = normalize_session(data)
            session["id"] = sid
            return session

    except Exception as exc:
        print(f"Failed to load session {session_id}: {exc}")

    return new_session()


def file_updated_at(path: Path, data: Dict[str, Any]) -> str:
    value = data.get("updated_at")

    if isinstance(value, str) and value:
        return value

    try:
        return datetime.fromtimestamp(path.stat().st_mtime, tz=timezone.utc).isoformat()
    except Exception:
        return now_iso()


def session_entries() -> List[Tuple[str, str, str]]:
    """
    Return list of:

        (updated_at, dropdown_label, session_id)

    sorted newest first.
    """
    entries: List[Tuple[str, str, str]] = []

    for path in CHATS_DIR.glob("*.json"):
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        messages = normalize_messages(data.get("messages"))

        if not messages:
            continue

        sid = safe_session_id(data.get("id") or path.stem)
        title = str(data.get("title") or "").strip()

        if not title or title.lower() == "new chat":
            title = ""

            for msg in messages:
                if msg["role"] == "user" and msg["content"].strip():
                    title = make_title(msg["content"])
                    break

            if not title:
                title = "Untitled chat"

        updated = file_updated_at(path, data)
        display_updated = str(updated)[:19].replace("T", " ")
        display = f"{title} — {display_updated}"

        entries.append((updated, display, sid))

    entries.sort(reverse=True)
    return entries


def session_choices() -> List[Tuple[str, str]]:
    return [(display, session_id) for _, display, session_id in session_entries()]


# -----------------------------------------------------------------------------
# OpenAI message conversion
# -----------------------------------------------------------------------------


def prepare_messages(
    history: List[Dict[str, Any]],
    system_prompt: str,
) -> List[Dict[str, str]]:
    """
    Build the message payload for the OpenAI-compatible API.

    The Chatbot UI contains the visible conversation.
    We send:
      - system prompt, if present
      - all previous user/assistant messages
    """
    messages: List[Dict[str, str]] = []

    if system_prompt and system_prompt.strip():
        messages.append(
            {
                "role": "system",
                "content": system_prompt,
            }
        )

    for msg in history:
        role = msg.get("role")
        content = msg.get("content")

        if (
            role in {"user", "assistant"}
            and isinstance(content, str)
            and content.strip()
        ):
            messages.append(
                {
                    "role": role,
                    "content": content,
                }
            )

    return messages


# -----------------------------------------------------------------------------
# Request parameter sanitization
# -----------------------------------------------------------------------------


def sanitize_float(value: Any, default: float, minimum: float, maximum: float) -> float:
    try:
        out = float(value)
    except Exception:
        out = float(default)

    return min(max(out, minimum), maximum)


def sanitize_int(value: Any, default: int, minimum: int, maximum: int) -> int:
    try:
        out = int(value)
    except Exception:
        out = int(default)

    return min(max(out, minimum), maximum)


# -----------------------------------------------------------------------------
# Initial state: load most recent session if available
# -----------------------------------------------------------------------------


initial_entries = session_entries()
initial_choices = [(display, sid) for _, display, sid in initial_entries]


def load_initial_session() -> Dict[str, Any]:
    if initial_entries:
        _, _, sid = initial_entries[0]
        session = load_session(sid)
        session["id"] = sid
        session["messages"] = normalize_messages(session.get("messages"))
        return session

    return new_session()


initial_session = load_initial_session()
initial_chat_messages = normalize_messages(initial_session.get("messages"))
initial_dropdown_value = initial_session["id"] if initial_chat_messages else None


# -----------------------------------------------------------------------------
# Response handler
# -----------------------------------------------------------------------------


def respond(
    history: List[Dict[str, Any]],
    user_text: str,
    session: Dict[str, Any],
    temperature: float,
    top_p: float,
    top_k: int,
    min_p: float,
    repeat_penalty: float,
    max_tokens: int,
    system_prompt: str,
):
    """
    Stream a chatbot response.

    Multi-turn history is kept in the Gradio Chatbot component.
    The full conversation is sent on every request.

    Standard OpenAI-compatible parameters:
      - temperature
      - top_p
      - max_tokens

    llama.cpp-specific parameters sent through extra_body:
      - top_k
      - min_p
      - repeat_penalty
    """
    my_gen = next_stream_generation()

    history = normalize_messages(history)
    session = normalize_session(session)
    user_text = (user_text or "").strip()
    system_prompt = str(system_prompt or "")

    if not user_text:
        yield history, "", gr.update(), session
        return

    # Sanitize settings.
    temperature = sanitize_float(temperature, TEMPERATURE, 0.0, 2.0)
    top_p = sanitize_float(top_p, TOP_P, 0.0, 1.0)
    top_k = sanitize_int(top_k, TOP_K, 0, 100)
    min_p = sanitize_float(min_p, MIN_P, 0.0, 1.0)
    repeat_penalty = sanitize_float(repeat_penalty, REPEAT_PENALTY, 1.0, 2.0)
    max_tokens = sanitize_int(max_tokens, MAX_TOKENS, 1, 8192)

    # Append the user message.
    history.append(
        {
            "role": "user",
            "content": user_text,
        }
    )

    # If this is the first user message, create a title from it.
    if not any(item["role"] == "user" for item in history[:-1]):
        session["title"] = make_title(user_text)

    # Save the session as soon as the user message is accepted.
    session["messages"] = history
    session = save_session(session)

    yield copy_messages(history), "", gr.update(), session

    # Append an empty assistant message that will be filled token by token.
    history.append(
        {
            "role": "assistant",
            "content": "",
        }
    )

    yield copy_messages(history), "", gr.update(), session

    if current_stream_generation() != my_gen:
        return

    # Do not send the empty assistant placeholder to the model.
    api_messages = prepare_messages(history[:-1], system_prompt)

    # Standard OpenAI-compatible parameters.
    request_kwargs: Dict[str, Any] = {
        "model": resolve_model(),
        "messages": api_messages,
        "stream": True,
        "temperature": temperature,
        "top_p": top_p,
        "max_tokens": max_tokens,
    }

    # llama.cpp-specific parameters that are not part of the standard OpenAI
    # schema. These are injected into the request body via extra_body.
    extra_body: Dict[str, Any] = {
        "top_k": int(top_k),
        "min_p": float(min_p),
        "repeat_penalty": float(repeat_penalty),
    }

    stream = None

    try:
        stream = client.chat.completions.create(
            **request_kwargs,
            extra_body=extra_body,
        )

        for chunk in stream:
            # If the user switched sessions, deleted, or started another
            # generation, stop updating this UI stream.
            if current_stream_generation() != my_gen:
                break

            if not getattr(chunk, "choices", None):
                continue

            delta = chunk.choices[0].delta
            piece = getattr(delta, "content", None)

            if not piece:
                continue

            history[-1]["content"] += piece

            yield copy_messages(history), "", gr.update(), session

    except Exception as exc:
        if current_stream_generation() == my_gen:
            history[-1]["content"] = (
                f"Could not reach llama.cpp at {BASE_URL}.\n\n"
                f"Error: {exc}"
            )

    finally:
        try:
            if stream is not None and hasattr(stream, "close"):
                stream.close()
        except Exception:
            pass

    # If the user moved on while this response was streaming, do not
    # overwrite the newer session/state.
    if current_stream_generation() != my_gen:
        return

    # If the model produced no content, remove the empty assistant bubble.
    if history and history[-1]["role"] == "assistant" and not history[-1].get("content"):
        history.pop()

    # Save the completed assistant message.
    session["messages"] = history
    session = save_session(session)

    yield (
        copy_messages(history),
        "",
        gr.update(
            choices=session_choices(),
            value=session["id"] if session["messages"] else None,
        ),
        session,
    )


# -----------------------------------------------------------------------------
# Gradio UI
# -----------------------------------------------------------------------------


with gr.Blocks(title="llama.cpp Gradio Chat") as demo:
    gr.Markdown("## Gradio ⇄ llama.cpp chat")
    gr.Markdown(f"OpenAI-compatible endpoint: `{BASE_URL}`")
    gr.Markdown(f"Saved chats folder: `{CHATS_DIR}`")

    with gr.Accordion("Generation settings", open=True):
        with gr.Row():
            temperature = gr.Slider(
                label="Temperature",
                minimum=0.0,
                maximum=2.0,
                step=0.05,
                value=TEMPERATURE,
            )
            top_p = gr.Slider(
                label="Top-p",
                minimum=0.0,
                maximum=1.0,
                step=0.01,
                value=TOP_P,
            )

        with gr.Row():
            top_k = gr.Slider(
                label="Top-k — llama.cpp extra_body",
                minimum=0,
                maximum=100,
                step=1,
                value=TOP_K,
            )
            min_p = gr.Slider(
                label="Min-p — llama.cpp extra_body",
                minimum=0.0,
                maximum=1.0,
                step=0.01,
                value=MIN_P,
            )

        with gr.Row():
            repeat_penalty = gr.Slider(
                label="Repeat penalty — llama.cpp extra_body",
                minimum=1.0,
                maximum=2.0,
                step=0.05,
                value=REPEAT_PENALTY,
            )
            max_tokens = gr.Slider(
                label="Max tokens",
                minimum=1,
                maximum=8192,
                step=1,
                value=MAX_TOKENS,
            )

        system_prompt = gr.Textbox(
            label="System prompt",
            value=DEFAULT_SYSTEM_PROMPT,
            lines=4,
            interactive=True,
        )

    with gr.Row():
        with gr.Column(scale=1, min_width=280):
            gr.Markdown("### Sessions")

            session_dropdown = gr.Dropdown(
                label="Conversation",
                choices=initial_choices,
                value=initial_dropdown_value,
                interactive=True,
            )

            new_chat_btn = gr.Button("New Chat", variant="primary")
            delete_chat_btn = gr.Button("Delete Chat", variant="stop")

            gr.Markdown(
                "Each conversation is saved as a JSON file.\n\n"
                "**New Chat** starts a new session.\n\n"
                "**Delete Chat** deletes the current session file."
            )

        with gr.Column(scale=4):
            chatbot = gr.Chatbot(
                label="Chat",
                value=initial_chat_messages,
                height=560,
                type="messages",
            )

            with gr.Row():
                msg = gr.Textbox(
                    label="Message",
                    placeholder="Ask the model…",
                    scale=8,
                    lines=2,
                )
                send_btn = gr.Button("Send", variant="primary", scale=2)

    current_session = gr.State(initial_session)

    # -------------------------------------------------------------------------
    # Event handlers
    # -------------------------------------------------------------------------

    def on_select(session_id: str):
        """
        Load a session when the user selects one from the dropdown.
        """
        next_stream_generation()

        if not session_id:
            return [], gr.update(choices=session_choices(), value=None), new_session()

        session = load_session(session_id)
        choices = session_choices()
        value = session_id if session["messages"] else None

        return session["messages"], gr.update(choices=choices, value=value), session

    def on_new_chat():
        """
        Start a new unsaved session.
        """
        next_stream_generation()
        return [], gr.update(choices=session_choices(), value=None), "", new_session()

    def on_delete(session: Dict[str, Any]):
        """
        Delete the current session file and start a new session.
        """
        next_stream_generation()

        session = normalize_session(session) if isinstance(session, dict) else {}
        sid = session.get("id")

        if sid:
            try:
                session_path(sid).unlink(missing_ok=True)
            except Exception as exc:
                print(f"Failed to delete session {sid}: {exc}")

        choices = session_choices()

        return [], gr.update(choices=choices, value=None), "", new_session()

    # -------------------------------------------------------------------------
    # Bindings
    # -------------------------------------------------------------------------

    session_dropdown.change(
        fn=on_select,
        inputs=[session_dropdown],
        outputs=[chatbot, session_dropdown, current_session],
        queue=False,
    )

    new_chat_btn.click(
        fn=on_new_chat,
        inputs=None,
        outputs=[chatbot, session_dropdown, msg, current_session],
        queue=False,
    )

    delete_chat_btn.click(
        fn=on_delete,
        inputs=[current_session],
        outputs=[chatbot, session_dropdown, msg, current_session],
        queue=False,
    )

    msg.submit(
        fn=respond,
        inputs=[
            chatbot,
            msg,
            current_session,
            temperature,
            top_p,
            top_k,
            min_p,
            repeat_penalty,
            max_tokens,
            system_prompt,
        ],
        outputs=[chatbot, msg, session_dropdown, current_session],
    )

    send_btn.click(
        fn=respond,
        inputs=[
            chatbot,
            msg,
            current_session,
            temperature,
            top_p,
            top_k,
            min_p,
            repeat_penalty,
            max_tokens,
            system_prompt,
        ],
        outputs=[chatbot, msg, session_dropdown, current_session],
    )


# Required for reliable streaming in Gradio.
demo.queue()


if __name__ == "__main__":
    demo.launch(
        server_name="0.0.0.0",
        server_port=int(os.getenv("PORT", "7860")),
        share=False,
    )
```

## 3. Start llama.cpp’s OpenAI-compatible server

Example using `llama-server`:

```bash
llama-server \
  -m /path/to/your/model.gguf \
  --host 0.0.0.0 \
  --port 8001 \
  --api-key EMPTY
```

The important part is that it serves:

```text
http://localhost:8001/v1
```

## What was added

The settings panel exposes:

| Setting | Sent as | Notes |
|---|---:|---|
| `temperature` | standard body param | OpenAI-compatible |
| `top_p` | standard body param | OpenAI-compatible |
| `max_tokens` | standard body param | OpenAI-compatible |
| `top_k` | `extra_body` | llama.cpp-specific |
| `min_p` | `extra_body` | llama.cpp-specific |
| `repeat_penalty` | `extra_body` | llama.cpp-specific |

The system prompt textbox is sent as a `system` message at the start of each request.
