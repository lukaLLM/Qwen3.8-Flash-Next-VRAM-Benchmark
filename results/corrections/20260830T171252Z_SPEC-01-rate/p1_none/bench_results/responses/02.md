# [2] chat history

Here's a complete, dependency-free build: **Archivist** — a chat client whose sessions are *files*. One JSON document per conversation on disk, an archive rail you can live inside, and a raw-record drawer so you always see exactly what just got written.

## `server.js` — zero-dependency Node, JSON-per-session storage

```javascript
import http from 'node:http';
import fsp from 'node:fs/promises';
import path from 'node:path';
import crypto from 'node:crypto';
import { fileURLToPath } from 'node:url';

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const PUBLIC_DIR = path.join(ROOT, 'public');
const DATA_DIR = process.env.DATA_DIR ? path.resolve(process.env.DATA_DIR) : path.join(ROOT, 'data', 'sessions');
const PORT = Number(process.env.PORT || 4173);
const HOST = process.env.HOST || '127.0.0.1';
const MAX_BODY = 2 * 1024 * 1024;
const ID_RE = /^[A-Za-z0-9_-]{4,64}$/;

const MIME = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.ico': 'image/x-icon',
  '.woff2': 'font/woff2',
};

/* ────────────────────────── storage primitives ────────────────────────── */

const nowISO = () => new Date().toISOString();
const makeId = () => Date.now().toString(36) + '-' + crypto.randomBytes(4).toString('hex');
const validDate = (v) => { const d = new Date(v); return Number.isNaN(+d) ? null : d.toISOString(); };
const fileFor = (id) => {
  if (!ID_RE.test(id)) throw Object.assign(new Error('bad session id'), { status: 400 });
  return path.join(DATA_DIR, `${id}.json`);
};
const cleanTitle = (v, fallback = 'Untitled session') => {
  const s = String(v ?? '').replace(/\s+/g, ' ').trim();
  return (s || fallback).slice(0, 120);
};
const normalizeMessage = (m) => ({
  id: typeof m?.id === 'string' && m.id ? m.id.slice(0, 64) : makeId(),
  role: ['user', 'assistant', 'system'].includes(m?.role) ? m.role : 'user',
  content: String(m?.content ?? '').slice(0, 20000),
  ts: Number.isFinite(+m?.ts) ? +m.ts : Date.now(),
});

function normalizeSession(raw, id) {
  const createdAt = validDate(raw?.createdAt) || nowISO();
  return {
    id,
    title: cleanTitle(raw?.title),
    createdAt,
    updatedAt: validDate(raw?.updatedAt) || createdAt,
    pinned: Boolean(raw?.pinned),
    messages: Array.isArray(raw?.messages) ? raw.messages.slice(0, 2000).map(normalizeMessage) : [],
  };
}

function summarize(s, raw) {
  const firstUser = s.messages.find((m) => m.role === 'user') || s.messages[0];
  return {
    id: s.id,
    title: s.title,
    createdAt: s.createdAt,
    updatedAt: s.updatedAt,
    pinned: s.pinned,
    bytes: Buffer.byteLength(raw ?? '', 'utf8'),
    messageCount: s.messages.length,
    preview: (firstUser?.content || 'No messages yet').replace(/\s+/g, ' ').slice(0, 110),
    excerpt: s.messages.map((m) => m.content).join('\n').slice(0, 3000).toLowerCase(),
  };
}

async function writeSession(session) {
  const file = fileFor(session.id);
  const tmp = `${file}.${process.pid}.tmp`;
  const json = JSON.stringify(session, null, 2) + '\n';
  await fsp.mkdir(DATA_DIR, { recursive: true });
  await fsp.writeFile(tmp, json, 'utf8');       // atomic: temp file, then rename
  await fsp.rename(tmp, file);
  return json;
}

async function readSession(id) {
  const raw = await fsp.readFile(fileFor(id), 'utf8');
  return normalizeSession(JSON.parse(raw), id);
}

async function listSessions() {
  const names = await fsp.readdir(DATA_DIR).catch(() => []);
  const out = [];
  for (const name of names) {
    if (!name.endsWith('.json')) continue;
    const id = name.slice(0, -5);
    if (!ID_RE.test(id)) continue;
    try {
      const raw = await fsp.readFile(path.join(DATA_DIR, name), 'utf8');
      out.push(summarize(normalizeSession(JSON.parse(raw), id), raw));
    } catch (err) {
      console.warn(`skipping unreadable session ${name}:`, err.message);
    }
  }
  out.sort((a, b) => (b.pinned - a.pinned) || (b.updatedAt > a.updatedAt ? 1 : -1));
  return out;
}

/* ────────────────────────── tiny http plumbing ────────────────────────── */

const json = (res, status, data) => {
  const body = JSON.stringify(data);
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Content-Length': Buffer.byteLength(body) });
  res.end(body);
};

function readBody(req) {
  return new Promise((resolve, reject) => {
    let size = 0;
    const chunks = [];
    req.on('data', (c) => {
      size += c.length;
      if (size > MAX_BODY) return reject(Object.assign(new Error('payload too large'), { status: 413 }));
      chunks.push(c);
    });
    req.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')));
    req.on('error', reject);
  });
}

async function parseJSON(req) {
  const raw = await readBody(req);
  if (!raw) return {};
  try { return JSON.parse(raw); }
  catch { throw Object.assign(new Error('invalid JSON body'), { status: 400 }); }
}

async function api(req, res, pathname) {
  if (pathname === '/api/health') {
    return json(res, 200, { ok: true, dataDir: DATA_DIR, time: nowISO() });
  }

  const seg = pathname.split('/').filter(Boolean);   // ['api','sessions',id?]
  if (seg[1] !== 'sessions') return json(res, 404, { error: 'not found' });
  const id = seg[2] ? decodeURIComponent(seg[2]) : null;

  // GET /api/sessions  ·  POST /api/sessions
  if (!id) {
    if (req.method === 'GET') {
      const sessions = await listSessions();
      const bytes = sessions.reduce((n, s) => n + s.bytes, 0);
      return json(res, 200, { sessions, count: sessions.length, bytes, dataDir: DATA_DIR });
    }
    if (req.method === 'POST') {
      const body = await parseJSON(req);
      const newId = body.id && ID_RE.test(body.id) ? body.id : makeId();
      const session = normalizeSession({ ...body, createdAt: body.createdAt ?? nowISO() }, newId);
      const raw = await writeSession(session);
      return json(res, 201, { session, summary: summarize(session, raw) });
    }
    return json(res, 405, { error: 'method not allowed' });
  }

  // /api/sessions/:id
  if (req.method === 'GET') {
    const session = await readSession(id).catch((e) => {
      if (e.code === 'ENOENT') throw Object.assign(new Error('session not found'), { status: 404 });
      throw e;
    });
    return json(res, 200, { session, summary: summarize(session, JSON.stringify(session)) });
  }

  if (req.method === 'PUT' || req.method === 'PATCH') {
    const body = await parseJSON(req);
    const existing = await readSession(id).catch((e) =>
      e.code === 'ENOENT' ? normalizeSession({}, id) : Promise.reject(e));

    const next = { ...existing };
    if ('title' in body) next.title = cleanTitle(body.title, existing.title);
    if ('pinned' in body) next.pinned = Boolean(body.pinned);
    if (Array.isArray(body.messages)) next.messages = body.messages.slice(0, 2000).map(normalizeMessage);
    next.createdAt = existing.createdAt;
    next.updatedAt = nowISO();                       // the server owns the clock

    const raw = await writeSession(next);
    return json(res, 200, { session: next, summary: summarize(next, raw) });
  }

  if (req.method === 'DELETE') {
    await fsp.rm(fileFor(id), { force: true });
    return json(res, 204, null);
  }

  return json(res, 405, { error: 'method not allowed' });
}

async function serveStatic(req, res, pathname) {
  if (req.method !== 'GET' && req.method !== 'HEAD') return json(res, 405, { error: 'method not allowed' });
  const rel = pathname === '/' ? '/index.html' : path.posix.normalize(pathname);
  const file = path.join(PUBLIC_DIR, rel);
  if (!file.startsWith(PUBLIC_DIR)) return json(res, 403, { error: 'forbidden' });

  try {
    const data = await fsp.readFile(file);
    res.writeHead(200, {
      'Content-Type': MIME[path.extname(file)] || 'application/octet-stream',
      'Cache-Control': 'no-store',
    });
    res.end(req.method === 'HEAD' ? undefined : data);
  } catch {
    if (path.extname(file)) return json(res, 404, { error: 'not found' });
    const html = await fsp.readFile(path.join(PUBLIC_DIR, 'index.html'));
    res.writeHead(200, { 'Content-Type': MIME['.html'], 'Cache-Control': 'no-store' });
    res.end(html);
  }
}

/* ────────────────────────── boot ────────────────────────── */

const SEED = {
  title: 'Welcome to the archive',
  messages: [
    { role: 'user', content: 'Where does any of this actually live?' },
    { role: 'assistant', content: 'On disk. Every session is one JSON file inside ./data/sessions — id, title, timestamps, and the full message array. Nothing is held only in memory; refresh the page and the transcript walks back in.' },
    { role: 'user', content: 'And if I delete something?' },
    { role: 'assistant', content: 'The file is removed. The toast that follows gives you a few seconds of undo, which simply writes the record back under the same name. After that it is gone for real.' },
  ],
};

async function seedIfEmpty() {
  await fsp.mkdir(DATA_DIR, { recursive: true });
  const names = await fsp.readdir(DATA_DIR);
  if (names.some((n) => n.endsWith('.json'))) return;
  const base = Date.now();
  await writeSession(normalizeSession({
    ...SEED,
    createdAt: new Date(base - 6e4).toISOString(),
    updatedAt: new Date(base - 6e4).toISOString(),
    messages: SEED.messages.map((m, i) => ({ ...m, ts: base - (SEED.messages.length - i) * 40000 })),
  }, makeId()));
  console.log('· seeded a welcome session');
}

const server = http.createServer(async (req, res) => {
  const { pathname } = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  try {
    if (pathname.startsWith('/api/')) return await api(req, res, pathname);
    return await serveStatic(req, res, pathname);
  } catch (err) {
    const status = err.status || 500;
    if (status >= 500) console.error(err);
    json(res, status, { error: err.message || 'server error' });
  }
});

await seedIfEmpty();
server.listen(PORT, HOST, () => {
  console.log(`\n  Archivist → http://${HOST}:${PORT}`);
  console.log(`  sessions  → ${DATA_DIR}\n`);
});
```

async function deleteSession(id) {
  const summary = state.sessions.find((s) => s.id === id);
  let record = null;
  try { record = (await api('GET', `/api/sessions/${id}`)).session; } catch {}
  await api('DELETE', `/api/sessions/${id}`);
  lastDeleted = record;
  if (state.current?.id === id) {
    state.current = null;
    const rest = state.sessions.filter((s) => s.id !== id);
    if (rest.length) { await loadSessions(); await openSession(rest[0].id); }
    else { await loadSessions(); renderAll(); }
  } else await loadSessions();

  toast('Session deleted', 'bad', record ? {
    label: 'Undo',
    fn: async () => {
      await api('POST', '/api/sessions', { id: record.id, title: record.title, pinned: record.pinned, createdAt: record.createdAt, messages: record.messages });
      await loadSessions(); openSession(record.id);
      toast('Restored from the recycle window');
    },
  } : null);
  if (summary && !record) toast('Session deleted', 'bad');
}

async function commitSave(keepalive = false) {
  if (!state.current) return;
  clearTimeout(state.saveTimer);
  setSave('saving', 'saving…');
  const payload = {
    title: state.current.title,
    pinned: state.current.pinned,
    messages: state.current.messages.map(({ streaming, ...m }) => m),
  };
  try {
    const { session, summary } = await api('PUT', `/api/sessions/${state.current.id}`, payload, { keepalive });
    state.current = session;
    const i = state.sessions.findIndex((s) => s.id === summary.id);
    if (i > -1) state.sessions[i] = summary; else state.sessions.unshift(summary);
    state.sessions.sort((a, b) => (b.pinned - a.pinned) || (b.updatedAt > a.updatedAt ? 1 : -1));
    setSave('saved', `saved · ${clock(session.updatedAt)}`);
    renderList(); renderHead(); renderRaw();
  } catch (e) {
    setSave('error', 'save failed');
    if (!keepalive) toast('Could not reach the server — nothing written', 'bad');
  }
}

function highlightJSON(j) {
  return esc(j).replace(
    /("(?:\\u[\da-fA-F]{4}|\\[^u]|[^\\"])*"(\s*:)?|\b(?:true|false)\b|\bnull\b|-?\d+(?:\.\d+)?(?:[eE][+-]?\d+)?)/g,
    (m) => {
      let cls = 'num';
      if (/^"/.test(m)) cls = /:\s*$/.test(m) ? 'key' : 'str';
      else if (/true|false/.test(m)) cls = 'bool';
      else if (/null/.test(m)) cls = 'null';

    });
}

async function send() {

  finishStream();
  if (!state.current) await newSession();

  els.input.value = ''; autosize(); els.counter.textContent = '0 chars';
  els.thread.querySelector('.empty')?.remove();
  els.thread.append(messageNode(msg));
  scrollDown(true);
  autoTitle();
  scheduleSave();

  const pending = { id: uid(), role: 'assistant', content: '', ts: Date.now(), streaming: true };
  state.current.messages.push(pending);

  setBubble(node, reply);
  stream = { msg: pending, node, text: reply, timer: null };

  scrollDown();
  scheduleSave(200);
}

## How it behaves

**Run it:** `node server.js` → `http://127.0.0.1:4173`. Zero installs. It seeds one welcome session on first boot, then never again.

**Storage shape** — `data/sessions/<id>.json`, one file per conversation:
```json
{ "id": "…", "title": "…", "createdAt": "ISO", "updatedAt": "ISO", "pinned": false,
  "messages": [{ "id": "…", "role": "user", "content": "…", "ts": 1710000000000 }] }
```

**Persistence decisions worth knowing about:** writes go to a temp file then `rename()`, so a crash mid-save can't leave you with half a transcript; the server owns `updatedAt` and normalizes/discards junk fields, so a hand-edited file can't poison the client; `GET /api/sessions` returns summaries with a preview and byte size, which is why the rail can show size and preview without loading transcripts.

**The lifecycle in the UI:** New Chat writes an empty file immediately (one row, one file — honest about it), the first user message becomes the title, autosave debounces 700 ms into a `PUT`, and the save pill tells you which state you're in. Delete asks inline, then offers Undo for a few seconds by re-writing the record under its original name. The right drawer renders the exact JSON you just persisted, highlighted, with its path and byte count.

**To wire a real model:** replace `echoReply(text, session)` in `app.js` — return a string (or a promise of one) and everything downstream, including the write-on-send ordering, keeps working.
