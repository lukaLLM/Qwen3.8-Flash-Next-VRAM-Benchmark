# [3] settings panel

Good call keeping the wire format explicit. The rule of thumb I implemented:

| Param | In OpenAI schema? | Where it goes | On the wire |
|---|---|---|---|
| `temperature`, `top_p`, `max_tokens` | ✅ named kwargs | top-level | top-level |
| `top_k`, `min_p`, `repeat_penalty` | ❌ llama.cpp-only | **`extra_body`** | **top-level** (llama-server reads these at root) |

`extra_body` is an OpenAI *SDK* mechanism — the client flattens it into the JSON root before it leaves. Since llama-server parses its sampler keys at the root too, a direct request must send them flat. So the code keeps them in a logically-separate `extra_body` group (so they're obvious and one-liner-addable to `mirostat`, `tfs_z`, etc.) but **flattens on send by default**, with a one-click **`nest`** toggle if you're fronting it with a proxy that re-splats `extra_body`. The settings drawer has a **live request preview** that shows the exact body — the three sampler keys are tinted lime and tagged `// extra_body` — so you watch the grouping appear/disappear as you flip wire mode. That's the honest version of "put them in `extra_body`" without shipping something llama.cpp silently ignores.

Settings persist to `data/settings.json`, and each assistant message records a `params` provenance snapshot — so the archive shows *which sampling produced which reply*. If llama.cpp isn't up, the send path falls back to the local echo (badged), so it stays demo-able.

import fsp from 'node:fs/promises';
import path from 'node:path';
import crypto from 'node:crypto';
import { Readable } from 'node:stream';
import { fileURLToPath } from 'node:url';

const ROOT = path.dirname(fileURLToPath(import.meta.url));
const PUBLIC_DIR = path.join(ROOT, 'public');
const DATA_DIR = process.env.DATA_DIR ? path.resolve(process.env.DATA_DIR) : path.join(ROOT, 'data', 'sessions');
const SETTINGS_FILE = path.join(path.dirname(DATA_DIR), 'settings.json');
const PORT = Number(process.env.PORT || 4173);
const HOST = process.env.HOST || '127.0.0.1';
const MAX_BODY = 2 * 1024 * 1024;
const ID_RE = /^[A-Za-z0-9_-]{4,64}$/;

// ── llama.cpp backend (OpenAI-compatible llama-server) ───────────────────
const LLAMA_URL = (process.env.LLAMA_URL || 'http://127.0.0.1:8080').replace(/\/$/, '');
const LLAMA_KEY = process.env.LLAMA_API_KEY || '';
const LLAMA_MODEL = process.env.LLAMA_MODEL || '';

const MIME = {
  '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8', '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml', '.png': 'image/png', '.ico': 'image/x-icon', '.woff2': 'font/woff2',
};

/* ─────────────────────── settings model (shared ranges) ─────────────────────── */
const SETTINGS_DEFAULT = {
  model: LLAMA_MODEL || 'local-model',
  system: 'You are Archivist, a concise assistant whose replies are saved verbatim to a JSON archive.',
  temperature: 0.7, top_p: 0.95,
  top_k: 40, min_p: 0.05, repeat_penalty: 1.1,   // ── llama.cpp extras ──
  max_tokens: 512,
  wireMode: 'flatten',                              // 'flatten' (direct llama) | 'nest' (proxy)
};
const RANGES = {
  temperature: [0, 2, 0.01], top_p: [0, 1, 0.01], min_p: [0, 1, 0.01],
  top_k: [0, 200, 1], repeat_penalty: [1, 2, 0.01], max_tokens: [1, 8192, 1],
};
const clamp = (v, [min, max]) => Math.min(max, Math.max(min, Number.isFinite(+v) ? +v : min));

function normalizeSettings(raw) {
  const s = SETTINGS_DEFAULT;
  const num = (k) => clamp(raw?.[k], RANGES[k]);
  return {
    model: String(raw?.model ?? s.model).trim().slice(0, 120) || s.model,
    system: String(raw?.system ?? s.system).replace(/\r/g, '').slice(0, 8000),
    temperature: num('temperature'), top_p: num('top_p'), min_p: num('min_p'),
    top_k: Math.round(num('top_k')), repeat_penalty: num('repeat_penalty'),
    max_tokens: Math.round(num('max_tokens')),
    wireMode: raw?.wireMode === 'nest' ? 'nest' : 'flatten',
  };
}
const readSettings = () => fsp.readFile(SETTINGS_FILE, 'utf8').then((r) => normalizeSettings(JSON.parse(r))).catch(() => ({ ...SETTINGS_DEFAULT }));
async function writeSettings(s) {
  await fsp.mkdir(path.dirname(SETTINGS_FILE), { recursive: true });
  await fsp.writeFile(SETTINGS_FILE, JSON.stringify(s, null, 2) + '\n', 'utf8');
  return s;
}

/* ────────────────────────── session primitives ────────────────────────── */
const nowISO = () => new Date().toISOString();
const makeId = () => Date.now().toString(36) + '-' + crypto.randomBytes(4).toString('hex');
const validDate = (v) => { const d = new Date(v); return Number.isNaN(+d) ? null : d.toISOString(); };
const fileFor = (id) => {
  if (!ID_RE.test(id)) throw Object.assign(new Error('bad session id'), { status: 400 });
  return path.join(DATA_DIR, `${id}.json`);
};
const cleanTitle = (v, f = 'Untitled session') => (String(v ?? '').replace(/\s+/g, ' ').trim() || f).slice(0, 120);
const normalizeMessage = (m) => ({
  id: typeof m?.id === 'string' && m.id ? m.id.slice(0, 64) : makeId(),
  role: ['user', 'assistant', 'system'].includes(m?.role) ? m.role : 'user',
  content: String(m?.content ?? '').slice(0, 20000),
  ts: Number.isFinite(+m?.ts) ? +m.ts : Date.now(),
  // provenance snapshot — recorded on the record, shown in the archive
  ...(m?.params ? { params: {
    model: String(m.params.model ?? '').slice(0, 80),
    temperature: +m.params.temperature || 0, top_p: +m.params.top_p || 0,
    top_k: Math.round(+m.params.top_k || 0), min_p: +m.params.min_p || 0,
    repeat_penalty: +m.params.repeat_penalty || 0, max_tokens: Math.round(+m.params.max_tokens || 0),
    via: m.params.via === 'llama.cpp' ? 'llama.cpp' : 'echo',
  } } : {}),
});

function normalizeSession(raw, id) {
  const createdAt = validDate(raw?.createdAt) || nowISO();
  return {
    id, title: cleanTitle(raw?.title), createdAt,
    updatedAt: validDate(raw?.updatedAt) || createdAt,
    pinned: Boolean(raw?.pinned),
    messages: Array.isArray(raw?.messages) ? raw.messages.slice(0, 2000).map(normalizeMessage) : [],
  };
}
function summarize(s, raw) {
  const fu = s.messages.find((m) => m.role === 'user') || s.messages[0];
  return {
    id: s.id, title: s.title, createdAt: s.createdAt, updatedAt: s.updatedAt, pinned: s.pinned,
    bytes: Buffer.byteLength(raw ?? '', 'utf8'), messageCount: s.messages.length,
    preview: (fu?.content || 'No messages yet').replace(/\s+/g, ' ').slice(0, 110),
    excerpt: s.messages.map((m) => m.content).join('\n').slice(0, 3000).toLowerCase(),
  };
}
async function writeSession(session) {
  const file = fileFor(session.id), tmp = `${file}.${process.pid}.tmp`;
  const json = JSON.stringify(session, null, 2) + '\n';
  await fsp.mkdir(DATA_DIR, { recursive: true });
  await fsp.writeFile(tmp, json, 'utf8');
  await fsp.rename(tmp, file);
  return json;
}
const readSession = async (id) => normalizeSession(JSON.parse(await fsp.readFile(fileFor(id), 'utf8')), id);
async function listSessions() {
  const names = await fsp.readdir(DATA_DIR).catch(() => []);
  const out = [];
  for (const name of names) {
    if (!name.endsWith('.json')) continue;
    const id = name.slice(0, -5);
    if (!ID_RE.test(id)) continue;
    try { const raw = await fsp.readFile(path.join(DATA_DIR, name), 'utf8'); out.push(summarize(normalizeSession(JSON.parse(raw), id), raw)); }
    catch (e) { console.warn(`skipping ${name}:`, e.message); }
  }
  out.sort((a, b) => (b.pinned - a.pinned) || (b.updatedAt > a.updatedAt ? 1 : -1));
  return out;
}

/* ────────────────────────── http plumbing ────────────────────────── */
const json = (res, status, data) => {
  const body = JSON.stringify(data);
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Content-Length': Buffer.byteLength(body) });
  res.end(body);
};
function readBody(req) {
  return new Promise((resolve, reject) => {
    let size = 0; const chunks = [];
    req.on('data', (c) => { if ((size += c.length) > MAX_BODY) return reject(Object.assign(new Error('too large'), { status: 413 })); chunks.push(c); });
    req.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')));
    req.on('error', reject);
  });
}
async function parseJSON(req) {
  const raw = await readBody(req);
  if (!raw) return {};
  try { return JSON.parse(raw); } catch { throw Object.assign(new Error('invalid JSON'), { status: 400 }); }
}

/* llama.cpp helpers — never trust a client-supplied URL (SSRF). Base is fixed. */
function llamaHeaders(extra = {}) {
  const h = { 'Content-Type': 'application/json', ...extra };
  if (LLAMA_KEY) h.Authorization = `Bearer ${LLAMA_KEY}`;
  return h;
}
async function llamaFetch(pathname, body, signal) {
  return fetch(LLAMA_URL + pathname, {
    method: body ? 'POST' : 'GET',
    headers: llamaHeaders(),
    body: body ? JSON.stringify(body) : undefined,
    signal,
  });
}

async function api(req, res, pathname) {
  /* ---- infra ---- */
  if (pathname === '/api/health') return json(res, 200, { ok: true, dataDir: DATA_DIR, llamaUrl: LLAMA_URL, time: nowISO() });

  if (pathname === '/api/llama/health') {
    const ac = new AbortController(); const t = setTimeout(() => ac.abort(), 2000);
    try {
      const r = await llamaFetch('/v1/models', null, ac.signal);
      const models = r.ok ? ((await r.json()).data || []).map((m) => m.id) : [];
      json(res, 200, { ok: r.ok, url: LLAMA_URL, models, defaultModel: LLAMA_MODEL || models[0] || SETTINGS_DEFAULT.model });
    } catch { json(res, 200, { ok: false, url: LLAMA_URL, models: [], defaultModel: LLAMA_MODEL || SETTINGS_DEFAULT.model }); }
    clearTimeout(t);
    return;
  }

  /* ---- settings ---- */
  if (pathname === '/api/settings') {
    if (req.method === 'GET') return json(res, 200, await readSettings());
    if (req.method === 'PUT') return json(res, 200, await writeSettings(normalizeSettings(await parseJSON(req))));
    return json(res, 405, { error: 'method not allowed' });
  }

  /* ---- chat proxy: browser sends { body } (the fully-formed llama body) ---- */
  if (pathname === '/api/chat' && req.method === 'POST') {
    const { body } = await parseJSON(req);
    if (!body || !Array.isArray(body.messages) || !body.messages.length)
      return json(res, 400, { error: 'body.messages required' });
    const forward = { ...body, model: body.model || LLAMA_MODEL || SETTINGS_DEFAULT.model, stream: Boolean(body.stream) };
    const ac = new AbortController(); const timer = setTimeout(() => ac.abort(), 300000);
    try {
      const up = await llamaFetch('/v1/chat/completions', forward, ac.signal);
      if (!up.ok) {
        const text = await up.text();
        return json(res, up.status, { error: 'llama.cpp upstream error', status: up.status, detail: text.slice(0, 800) });
      }
      // stream pass-through: hand the SSE straight to the browser
      res.writeHead(200, {
        'Content-Type': 'text/event-stream; charset=utf-8',
        'Cache-Control': 'no-cache, no-transform', 'Connection': 'keep-alive', 'X-Accel-Buffering': 'no',
      });
      if (forward.stream) {
        Readable.fromWeb(up.body).on('error', () => res.end()).pipe(res);
      } else {
        res.end(await up.text()); // non-stream: passthrough raw JSON
      }
      res.on('close', () => { clearTimeout(timer); ac.abort(); });
    } catch (e) {
      if (!res.headersSent) return json(res, 502, { error: 'llama.cpp unreachable', detail: String(e.message || e) });
      res.end();
    }
    return;
  }

  /* ---- sessions (unchanged) ---- */
  const seg = pathname.split('/').filter(Boolean);
  if (seg[1] !== 'sessions') return json(res, 404, { error: 'not found' });
  const id = seg[2] ? decodeURIComponent(seg[2]) : null;

  if (!id) {
    if (req.method === 'GET') {
      const sessions = await listSessions();
      return json(res, 200, { sessions, count: sessions.length, bytes: sessions.reduce((n, s) => n + s.bytes, 0), dataDir: DATA_DIR });
    }
    if (req.method === 'POST') {
      const b = await parseJSON(req);
      const nid = b.id && ID_RE.test(b.id) ? b.id : makeId();
      const session = normalizeSession({ ...b, createdAt: b.createdAt ?? nowISO() }, nid);
      const raw = await writeSession(session);
      return json(res, 201, { session, summary: summarize(session, raw) });
    }
    return json(res, 405, { error: 'method not allowed' });
  }

  if (req.method === 'GET') {
    try { const s = await readSession(id); return json(res, 200, { session: s, summary: summarize(s, JSON.stringify(s)) }); }
    catch (e) { if (e.code === 'ENOENT') throw Object.assign(new Error('not found'), { status: 404 }); throw e; }
  }
  if (req.method === 'PUT' || req.method === 'PATCH') {
    const b = await parseJSON(req);
    const existing = await readSession(id).catch((e) => (e.code === 'ENOENT' ? normalizeSession({}, id) : Promise.reject(e)));
    const next = { ...existing };
    if ('title' in b) next.title = cleanTitle(b.title, existing.title);
    if ('pinned' in b) next.pinned = Boolean(b.pinned);
    if (Array.isArray(b.messages)) next.messages = b.messages.slice(0, 2000).map(normalizeMessage);
    next.createdAt = existing.createdAt; next.updatedAt = nowISO();
    const raw = await writeSession(next);
    return json(res, 200, { session: next, summary: summarize(next, raw) });
  }
  if (req.method === 'DELETE') { await fsp.rm(fileFor(id), { force: true }); return json(res, 204, null); }
  return json(res, 405, { error: 'method not allowed' });
}

async function serveStatic(req, res, pathname) {
  if (req.method !== 'GET' && req.method !== 'HEAD') return json(res, 405, { error: 'method not allowed' });
  const rel = pathname === '/' ? '/index.html' : path.posix.normalize(pathname);
  const file = path.join(PUBLIC_DIR, rel);
  if (!file.startsWith(PUBLIC_DIR)) return json(res, 403, { error: 'forbidden' });
  try {
    const data = await fsp.readFile(file);
    res.writeHead(200, { 'Content-Type': MIME[path.extname(file)] || 'application/octet-stream', 'Cache-Control': 'no-store' });
    res.end(req.method === 'HEAD' ? undefined : data);
  } catch {
    if (path.extname(file)) return json(res, 404, { error: 'not found' });
    const html = await fsp.readFile(path.join(PUBLIC_DIR, 'index.html'));
    res.writeHead(200, { 'Content-Type': MIME['.html'], 'Cache-Control': 'no-store' }); res.end(html);
  }
}

const SEED = {
  title: 'Welcome to the archive',
  messages: [
    { role: 'user', content: 'Where does any of this actually live?' },
    { role: 'assistant', content: 'On disk. Every session is one JSON file inside ./data/sessions — id, title, timestamps, and the full message array. Sampling params are archived alongside each reply, too.' },
    { role: 'user', content: 'And if I delete something?' },
    { role: 'assistant', content: 'The file is removed. The toast gives you a few seconds of undo, which writes the record back under the same name.' },
  ],
};
async function seedIfEmpty() {
  await fsp.mkdir(DATA_DIR, { recursive: true });
  if ((await fsp.readdir(DATA_DIR)).some((n) => n.endsWith('.json'))) return;
  const base = Date.now();
  await writeSession(normalizeSession({ ...SEED, createdAt: new Date(base - 6e4).toISOString(), updatedAt: new Date(base - 6e4).toISOString(),
    messages: SEED.messages.map((m, i) => ({ ...m, ts: base - (SEED.messages.length - i) * 40000 })) }, makeId()));
  console.log('· seeded welcome session');
}

const server = http.createServer(async (req, res) => {
  const { pathname } = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  try {
    if (pathname.startsWith('/api/')) return await api(req, res, pathname);
    return await serveStatic(req, res, pathname);
  } catch (err) {
    const s = err.status || 500; if (s >= 500) console.error(err);
    json(res, s, { error: err.message || 'server error' });
  }
});
server.requestTimeout = 0; server.headersTimeout = 0; // allow long generations

await writeSettings(await readSettings()).catch(() => {}); // ensure settings.json exists
await seedIfEmpty();
server.listen(PORT, HOST, () => {
  console.log(`\n  Archivist → http://${HOST}:${PORT}`);
  console.log(`  sessions  → ${DATA_DIR}`);
  console.log(`  llama.cpp → ${LLAMA_URL || '(unset)'}   (start llama-server, or it falls back to echo)\n`);
});

function buildRequest(conv, stream = true) {
  const s = state.settings;
  const messages = (s.system && s.system.trim())
    ? [{ role:'system', content:s.system }, ...conv]
    : [...conv];
  // named kwargs that ARE in the OpenAI schema → top level
  const standard = {
    model: s.model || 'local-model',
    messages, stream,
    temperature: rnd(s.temperature, 2),
    top_p: rnd(s.top_p, 2),
    max_tokens: Math.round(s.max_tokens),
  };
  // llama.cpp samplers that are NOT in the OpenAI schema → extra_body
  const extra_body = {
    top_k: Math.round(s.top_k),
    min_p: rnd(s.min_p, 3),
    repeat_penalty: rnd(s.repeat_penalty, 3),
  };
  // SDK semantics flatten extra_body into the JSON root, which is exactly what
  // llama-server parses. Flip to 'nest' only behind a proxy that re-splats it.
  return s.wireMode === 'nest' ? { ...standard, extra_body } : { ...standard, ...extra_body };
}
