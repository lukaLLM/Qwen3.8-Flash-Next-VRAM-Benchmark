Below is the updated `app.py` with a **Generation settings** panel.

It adds sliders and a textbox for:

- `temperature` → standard OpenAI request field
- `top_p` → standard OpenAI request field
- `max_tokens` → standard OpenAI request field
- `system prompt` → sent as the first `system` message
- `top_k` → llama.cpp-specific, sent via `extra_body`
- `min_p` → llama.cpp-specific, sent via `extra_body`
- `repeat_penalty` → llama.cpp-specific, sent via `extra_body`

The `openai` client supports `extra_body`, which merges extra keys directly into the JSON request body sent to llama.cpp.

```python
#!/usr/bin/env python3
"""
Persistent Gradio chat for an OpenAI-compatible llama.cpp server.

Runs against:
    http://localhost:8001/v1

Chat sessions are saved as JSON files under SESSION_DIR, default: ./sessions

llama.cpp sampling parameters that are not part of the standard OpenAI schema
are sent via the openai client's `extra_body` argument.
"""

import json
import os
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional, Tuple

import gradio as gr
from openai import OpenAI


# -----------------------------------------------------------------------------
# llama.cpp / OpenAI client settings
# -----------------------------------------------------------------------------

def _env_number(name: str, cast):
    raw = os.getenv(name)
    if raw is None or raw.strip() == "":
        return None
    try:
        return cast(raw)
    except (TypeError, ValueError):
        return None


BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("LLAMA_CPP_API_KEY", "EMPTY")
MODEL = os.getenv("LLAMA_CPP_MODEL", "")

# Default system prompt. In the UI, this can be changed and is sent as the
# first system message for each request.
SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "")

client = OpenAI(base_url=BASE_URL, api_key=API_KEY)
_model_name = MODEL


# -----------------------------------------------------------------------------
# Session persistence
# -----------------------------------------------------------------------------

SESSION_DIR = Path(os.getenv("SESSION_DIR", "sessions"))
SESSION_DIR.mkdir(parents=True, exist_ok=True)


def _now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def _history_snapshot(
    history: List[Dict[str, str]],
) -> List[Dict[str, str]]:
    """
    Return a shallow deep-ish copy of message dicts.

    This prevents Gradio from seeing shared mutable message dictionaries while
    streaming.
    """
    return [msg.copy() for msg in history]


def get_model_name() -> str:
    """
    Try to detect the served model name from llama.cpp's /v1/models endpoint.

    If detection fails, fall back to "local-model".
    """
    global _model_name

    if _model_name:
        return _model_name

    try:
        models = client.models.list()
        if models.data:
            _model_name = models.data[0].id
            return _model_name
    except Exception:
        pass

    _model_name = "local-model"
    return _model_name


def normalize_history(history: Any) -> List[Dict[str, str]]:
    """
    Convert Gradio chat history into a simple OpenAI-style message list.

    Supports both:
      - newer messages format: {"role": "...", "content": "..."}
      - older tuple format:    ("user", "hello")
    """
    out: List[Dict[str, str]] = []

    if not isinstance(history, (list, tuple)):
        return out

    for item in history:
        if isinstance(item, dict):
            role = str(item.get("role") or "user")
            content = item.get("content", "")
        elif isinstance(item, (tuple, list)) and len(item) >= 2:
            role = str(item[0] or "user")
            content = item[1]
        else:
            role = "user"
            content = str(item)

        if role not in {"user", "assistant", "system"}:
            role = "user"

        if isinstance(content, list):
            text_parts: List[str] = []
            for part in content:
                if isinstance(part, dict) and part.get("type") == "text":
                    text_parts.append(str(part.get("text", "")))
                elif isinstance(part, str):
                    text_parts.append(part)
            content = " ".join(text_parts)

        out.append({"role": role, "content": str(content)})

    return out


def _safe_path(session_id: Optional[str]) -> Optional[Path]:
    """
    Convert a session id into a safe path inside SESSION_DIR.

    Session files are always SESSION_DIR/<sanitized-id>.json.
    """
    if session_id is None:
        return None

    s = str(session_id).strip()
    if not s:
        return None

    # Prevent path traversal and odd characters.
    s = Path(s).stem
    s = "".join(ch for ch in s if ch.isalnum() or ch in "-_")

    if not s:
        return None

    return SESSION_DIR / f"{s}.json"


def _new_session_data(session_id: Optional[str] = None) -> dict:
    path = _safe_path(session_id)
    sid = path.stem if path is not None else uuid.uuid4().hex

    now = _now()
    return {
        "id": sid,
        "title": "New Chat",
        "created_at": now,
        "updated_at": now,
        "messages": [],
    }


def _read_json(path: Path) -> Optional[dict]:
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
        if isinstance(data, dict):
            return data
        return None
    except Exception:
        return None


def _write_json(path: Path, data: dict) -> None:
    tmp = path.with_name(path.name + ".tmp")
    with tmp.open("w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
    os.replace(tmp, path)


def _prepare_session(session_id: str, path: Path, data: Optional[dict]) -> dict:
    if not isinstance(data, dict):
        data = _new_session_data(session_id)

    data["id"] = path.stem

    if not isinstance(data.get("title"), str) or not data["title"].strip():
        data["title"] = "New Chat"

    if not isinstance(data.get("created_at"), str) or not data["created_at"].strip():
        data["created_at"] = _now()

    if not isinstance(data.get("updated_at"), str) or not data["updated_at"].strip():
        data["updated_at"] = _now()

    if not isinstance(data.get("messages"), list):
        data["messages"] = []

    return data


def _ensure_current_session(current_id: Optional[str]) -> Tuple[str, Path, dict]:
    """
    Return (session_id, path, data), creating a new session if needed.
    """
    path = _safe_path(current_id)

    if path is None:
        data = _new_session_data(None)
        path = _safe_path(data["id"])
        if path is None:
            raise RuntimeError("Could not create a safe session path.")
    else:
        data = _read_json(path)
        if data is None:
            # If the file exists but is invalid JSON, start a fresh session
            # with the same id so the dropdown selection remains stable.
            data = _new_session_data(path.stem)

    data = _prepare_session(path.stem, path, data)
    return path.stem, path, data


def _list_sessions():
    """
    Read all session JSON files and return dropdown choices sorted by recency.
    """
    rows = []

    for path in SESSION_DIR.glob("*.json"):
        if path.name.endswith(".tmp"):
            continue

        session = _read_json(path) or {}
        title = str(session.get("title") or path.stem).strip() or path.stem

        updated = str(session.get("updated_at") or session.get("created_at") or "")
        if not updated:
            try:
                updated = datetime.fromtimestamp(
                    path.stat().st_mtime,
                    tz=timezone.utc,
                ).isoformat(timespec="seconds")
            except OSError:
                updated = ""

        label = f"{title} | {updated[:16].replace('T', ' ')}"
        rows.append((updated, label, path.stem))

    rows.sort(key=lambda r: (r[0], r[1]), reverse=True)
    choices = [(label, sid) for _, label, sid in rows]
    values = {sid for _, _, sid in rows}

    return choices, values


def _dropdown_choices(value: Optional[str] = None):
    choices, values = _list_sessions()

    if value is not None:
        value = str(value)
        if value not in values:
            value = None
    else:
        value = None

    return gr.update(choices=choices, value=value)


def _dropdown_value(value: Optional[str]):
    return gr.update(value=str(value) if value is not None else None)


def _derive_title(history: List[Dict[str, str]]) -> str:
    """
    Use the first user message as the title.
    """
    for msg in history:
        if msg.get("role") == "user" and msg.get("content", "").strip():
            raw = msg["content"].strip().splitlines()[0].strip()
            if raw:
                if len(raw) > 50:
                    return raw[:50] + "..."
                return raw

    return "New Chat"


# -----------------------------------------------------------------------------
# Settings helpers
# -----------------------------------------------------------------------------

def _clean_float(value: Any) -> Optional[float]:
    if value is None:
        return None
    try:
        return float(value)
    except (TypeError, ValueError):
        return None


def _clean_int(value: Any) -> Optional[int]:
    if value is None:
        return None
    try:
        return int(round(float(value)))
    except (TypeError, ValueError):
        return None


def build_sampling_kwargs(
    temperature: Optional[float],
    top_p: Optional[float],
    max_tokens: Optional[int],
    top_k: Optional[float],
    min_p: Optional[float],
    repeat_penalty: Optional[float],
) -> Dict[str, Any]:
    """
    Build kwargs for OpenAI client chat.completions.create().

    Standard OpenAI schema fields:
      - temperature
      - top_p
      - max_tokens

    llama.cpp-specific fields are passed through `extra_body`, which merges
    them into the JSON request body sent to llama.cpp:
      - top_k
      - min_p
      - repeat_penalty
    """
    kwargs: Dict[str, Any] = {}

    t = _clean_float(temperature)
    if t is not None:
        kwargs["temperature"] = t

    p = _clean_float(top_p)
    if p is not None:
        kwargs["top_p"] = p

    mt = _clean_int(max_tokens)
    if mt is not None and mt > 0:
        kwargs["max_tokens"] = mt

    extra_body: Dict[str, Any] = {}

    tk = _clean_int(top_k)
    if tk is not None and tk > 0:
        extra_body["top_k"] = tk

    mp = _clean_float(min_p)
    if mp is not None:
        extra_body["min_p"] = mp

    rp = _clean_float(repeat_penalty)
    if rp is not None and rp > 0:
        extra_body["repeat_penalty"] = rp

    if extra_body:
        kwargs["extra_body"] = extra_body

    return kwargs


# -----------------------------------------------------------------------------
# Handlers
# -----------------------------------------------------------------------------

def user_submit(
    message: str,
    history: Any,
    current_session_id: Optional[str],
):
    """
    User submit handler.

    Adds the user message to the history, saves it, and signals that the
    assistant should stream a response.
    """
    history = normalize_history(history)

    if not message or not message.strip():
        return message, _history_snapshot(history), current_session_id or "", False

    current_session_id, path, data = _ensure_current_session(current_session_id)

    history.append({"role": "user", "content": message.strip()})

    if not data.get("title") or data["title"] == "New Chat":
        data["title"] = _derive_title(history)

    data["messages"] = history
    data["updated_at"] = _now()
    _write_json(path, data)

    return "", _history_snapshot(history), current_session_id, True


def stream_reply(
    history: Any,
    current_session_id: Optional[str],
    pending_response: bool,
    temperature: Optional[float],
    top_p: Optional[float],
    top_k: Optional[float],
    min_p: Optional[float],
    repeat_penalty: Optional[float],
    max_tokens: Optional[int],
    system_prompt: Optional[str],
):
    """
    Assistant handler.

    Sends the full conversation history to llama.cpp and streams the assistant
    response into the chatbot.
    """
    history = normalize_history(history)

    if not pending_response:
        yield _history_snapshot(history), False
        return

    if not history or history[-1]["role"] != "user":
        yield _history_snapshot(history), False
        return

    current_session_id, path, data = _ensure_current_session(current_session_id)

    # Add an assistant placeholder that will be streamed into.
    if not (
        history
        and history[-1]["role"] == "assistant"
        and history[-1]["content"] == ""
    ):
        history.append({"role": "assistant", "content": ""})

    yield _history_snapshot(history), True

    messages_to_send: List[Dict[str, str]] = []

    sys_prompt = (system_prompt or "").strip()
    if sys_prompt:
        messages_to_send.append({"role": "system", "content": sys_prompt})

    # Send the full stored conversation history except the current assistant
    # placeholder being streamed into.
    for msg in history[:-1]:
        if msg["role"] in {"user", "assistant", "system"} and msg["content"]:
            messages_to_send.append(
                {"role": msg["role"], "content": msg["content"]}
            )

    if not messages_to_send:
        history[-1]["content"] = "(no text to send)"

        if not data.get("title") or data["title"] == "New Chat":
            data["title"] = _derive_title(history)

        data["messages"] = history
        data["updated_at"] = _now()
        _write_json(path, data)

        yield _history_snapshot(history), False
        return

    request_kwargs: Dict[str, Any] = {
        "model": get_model_name(),
        "messages": messages_to_send,
        "stream": True,
    }

    sampling_kwargs = build_sampling_kwargs(
        temperature=temperature,
        top_p=top_p,
        max_tokens=max_tokens,
        top_k=top_k,
        min_p=min_p,
        repeat_penalty=repeat_penalty,
    )

    request_kwargs.update(sampling_kwargs)

    try:
        stream = client.chat.completions.create(**request_kwargs)

        for chunk in stream:
            if not chunk.choices:
                continue

            choice = chunk.choices[0]
            delta = None

            if getattr(choice, "delta", None) is not None:
                delta = getattr(choice.delta, "content", None)

            # Some servers may provide the final text via choice.message.
            if (
                not delta
                and history[-1]["content"] == ""
                and getattr(choice, "message", None) is not None
            ):
                delta = getattr(choice.message, "content", None)

            if delta:
                history[-1]["content"] += str(delta)
                yield _history_snapshot(history), True

        if not history[-1]["content"]:
            history[-1]["content"] = "(no text response)"
            yield _history_snapshot(history), True

        if not data.get("title") or data["title"] == "New Chat":
            data["title"] = _derive_title(history)

        data["messages"] = history
        data["updated_at"] = _now()
        _write_json(path, data)

        yield _history_snapshot(history), False

    except Exception as exc:
        error_text = f"llama.cpp request failed: {exc}"

        if history[-1]["content"]:
            history[-1]["content"] += f"\n\n{error_text}"
        else:
            history[-1]["content"] = error_text

        if not data.get("title") or data["title"] == "New Chat":
            data["title"] = _derive_title(history)

        data["messages"] = history
        data["updated_at"] = _now()
        _write_json(path, data)

        yield _history_snapshot(history), False


def select_session(
    selected_id: Optional[str],
    current_session_id: Optional[str],
    pending_response: bool,
    history: Any,
):
    """
    Load a session when the user selects it from the dropdown.
    """
    selected = str(selected_id) if selected_id is not None else None
    path = _safe_path(selected)

    if path is None or not path.exists():
        return (
            _history_snapshot(normalize_history(history)),
            current_session_id or "",
            False,
            _dropdown_value(selected),
            "No saved session found for that selection.",
        )

    data = _read_json(path) or {}
    loaded_history = normalize_history(data.get("messages", []))

    title = str(data.get("title") or path.stem)

    return (
        _history_snapshot(loaded_history),
        selected,
        False,
        _dropdown_value(selected),
        f'Loaded "{title}".',
    )


def new_chat():
    """
    Start a new unsaved chat.

    The session is not written to disk until the first user message is sent.
    """
    return (
        [],
        "",
        False,
        _dropdown_choices(None),
        "Started a new chat. It will be saved after your first message.",
    )


def delete_session(
    selected_id: Optional[str],
    current_session_id: Optional[str],
    pending_response: bool,
    history: Any,
):
    """
    Delete the selected session from disk.

    If the deleted session is the currently loaded one, start a new chat.
    """
    target = selected_id or current_session_id
    path = _safe_path(target)

    if path is None or not path.exists():
        return (
            _history_snapshot(normalize_history(history)),
            current_session_id or "",
            pending_response,
            _dropdown_choices(current_session_id),
            "No saved session selected.",
        )

    old_data = _read_json(path) or {}
    title = str(old_data.get("title") or path.stem)

    try:
        path.unlink()
    except Exception as exc:
        return (
            _history_snapshot(normalize_history(history)),
            current_session_id or "",
            pending_response,
            _dropdown_choices(current_session_id),
            f"Delete failed: {exc}",
        )

    if target == current_session_id:
        return (
            [],
            "",
            False,
            _dropdown_choices(None),
            f'Deleted "{title}" and started a new chat.',
        )

    return (
        _history_snapshot(normalize_history(history)),
        current_session_id or "",
        pending_response,
        _dropdown_choices(current_session_id),
        f'Deleted "{title}".',
    )


def refresh_sessions(current_session_id: Optional[str] = None):
    """
    Refresh the dropdown list.
    """
    path = _safe_path(current_session_id)
    value = current_session_id if path is not None and path.exists() else None
    return _dropdown_choices(value)
```

## 3. What gets sent to llama.cpp

For example, if the UI settings are:

```text
temperature = 0.7
top_p = 0.95
top_k = 40
min_p = 0.1
repeat_penalty = 1.1
max_tokens = 1024
system prompt = "You are a helpful assistant."
```

The request body sent to llama.cpp will conceptually look like this:

```json
{
  "model": "your-detected-model",
  "stream": true,
  "messages": [
    {
      "role": "system",
      "content": "You are a helpful assistant."
    },
    {
      "role": "user",
      "content": "Hello"
    }
  ],
  "temperature": 0.7,
  "top_p": 0.95,
  "max_tokens": 1024,
  "top_k": 40,
  "min_p": 0.1,
  "repeat_penalty": 1.1
}
```

The important part is that `top_k`, `min_p`, and `repeat_penalty` are included in the request through:

```python
client.chat.completions.create(
    ...,
    extra_body={
        "top_k": 40,
        "min_p": 0.1,
        "repeat_penalty": 1.1,
    },
)
```
