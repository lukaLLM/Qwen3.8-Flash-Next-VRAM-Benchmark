# [1] build app

## 1. Create a separate venv with `uv`

From a terminal:

```bash
mkdir llama-gradio-chat
cd llama-gradio-chat

# Create a separate virtual environment named .venv
uv venv .venv --python 3.11

# Activate it

# Linux/macOS
source .venv/bin/activate

# Windows PowerShell
# .venv\Scripts\Activate.ps1

# Install packages into the same venv using uv
uv pip install "gradio>=5" openai
```

If you do not want to activate the venv, you can also do:

```bash
uv pip install --python .venv/bin/python "gradio>=5" openai
.venv/bin/python app.py
```

On Windows:

```powershell
uv pip install --python .venv\Scripts\python.exe "gradio>=5" openai
.venv\Scripts\python.exe app.py
```

## 2. Create `app.py`

Save this as `app.py` in the same folder:

```python
#!/usr/bin/env python3
import os

import gradio as gr
from openai import OpenAI


# llama.cpp OpenAI-compatible server
BASE_URL = "http://localhost:8001/v1"
API_KEY = "not-needed"

# Prefer an explicit model name if provided:
#   LLAMA_MODEL="my-model-name" python app.py
# or:
#   MODEL="my-model-name" python app.py
MODEL = os.getenv("LLAMA_MODEL") or os.getenv("MODEL")

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)

# If no model name is provided, try to read the first model exposed by llama.cpp.
if not MODEL:
    try:
        listed = client.models.list()
        model_list = getattr(listed, "data", None)
        if model_list is None:
            model_list = list(listed)
        MODEL = model_list[0].id if model_list else "local-model"
    except Exception:
        MODEL = "local-model"

print(f"Using llama.cpp model: {MODEL}")


def respond(message: str, history: list, busy: bool):
    """
    Handles one chat turn.

    - Adds the user message to the Gradio Chatbot message list.
    - Sends the full conversation history to llama.cpp each time.
    - Streams assistant tokens back into the UI.
    """
    history = list(history or [])

    # Ignore queued requests while one generation is already running.
    if busy:
        yield history.copy(), gr.update(), gr.update(), True
        return

    # Ignore empty input.
    if not message or not message.strip():
        yield (
            history.copy(),
            gr.update(value="", interactive=True),
            gr.update(interactive=True),
            False,
        )
        return

    # Add user message to the visible conversation.
    history.append({"role": "user", "content": message.strip()})
    yield (
        history.copy(),
        gr.update(value="", interactive=False),
        gr.update(interactive=False),
        True,
    )

    # Build the OpenAI-compatible request payload from the current conversation.
    # Do not include the placeholder assistant message below.
    request_messages = []
    for msg in history:
        if isinstance(msg, dict) and msg.get("role") in {"system", "user", "assistant"}:
            content = msg.get("content", "")
            request_messages.append(
                {
                    "role": msg["role"],
                    "content": "" if content is None else content,
                }
            )

    # Add a placeholder assistant message for streaming output.
    history.append({"role": "assistant", "content": ""})
    yield (
        history.copy(),
        gr.update(interactive=False),
        gr.update(interactive=False),
        True,
    )

    assistant_text = ""

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
        )

        for chunk in stream:
            if getattr(chunk, "choices", None):
                delta = chunk.choices[0].delta.content
                if delta:
                    assistant_text += delta
                    history[-1] = {
                        "role": "assistant",
                        "content": assistant_text,
                    }
                    yield (
                        history.copy(),
                        gr.update(interactive=False),
                        gr.update(interactive=False),
                        True,
                    )

    except Exception as exc:
        # Preserve partial output if it exists, then append the error.
        error_text = f"\n\n⚠️ llama.cpp request failed: {exc}"
        history[-1] = {
            "role": "assistant",
            "content": (assistant_text + error_text).strip(),
        }
        yield (
            history.copy(),
            gr.update(interactive=False),
            gr.update(interactive=False),
            True,
        )

    # Re-enable UI input after the response finishes.
    yield (
        history.copy(),
        gr.update(value="", interactive=True),
        gr.update(interactive=True),
        False,
    )


with gr.Blocks(title="llama.cpp Chat") as demo:
    gr.Markdown(
        f"## llama.cpp chat\n\n"
        f"Endpoint: `{BASE_URL}`  \n"
        f"Model: `{MODEL}`"
    )

    chatbot = gr.Chatbot(
        value=[],
        type="messages",
        height=600,
    )

    # State to prevent overlapping/streaming requests from clobbering each other.
    busy = gr.State(False)

    with gr.Row():
        user_input = gr.Textbox(
            lines=1,
            placeholder="Type a message...",
            show_label=False,
            scale=8,
        )
        send_btn = gr.Button("Send", variant="primary", scale=1)

    inputs = [user_input, chatbot, busy]
    outputs = [chatbot, user_input, send_btn, busy]

    send_btn.click(respond, inputs=inputs, outputs=outputs)
    user_input.submit(respond, inputs=inputs, outputs=outputs)


if __name__ == "__main__":
    # queue() is required for streaming with generator events.
    demo.queue(default_concurrency_limit=1).launch(
        server_name="0.0.0.0",
        server_port=7860,
    )
```

## 3. Start a llama.cpp server on port 8001

In another terminal, start your `llama.cpp` OpenAI-compatible server, for example:

```bash
llama-server -m /path/to/model.gguf --port 8001
```

Or, if you built `llama.cpp` from source:

```bash
./build/bin/llama-server -m /path/to/model.gguf --port 8001
```

Make sure it exposes an OpenAI-compatible endpoint at:

```text
http://localhost:8001/v1/chat/completions
```

---

## 4. Run the Gradio app

With the venv active:

```bash
python app.py
```

Then open:

```text
http://localhost:7860
```

---

## 5. Optional: set the model name manually

If the app cannot auto-detect the model, or if llama.cpp rejects the detected model name, list the models:

```bash
curl http://localhost:8001/v1/models
```

Then set `LLAMA_MODEL` or `MODEL` when launching:

```bash
LLAMA_MODEL="your-model-id" python app.py
```

or:

```bash
MODEL="your-model-id" python app.py
```
