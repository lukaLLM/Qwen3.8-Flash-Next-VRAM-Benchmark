# [2] chat history

Below is a complete updated single-file app with persistent sessions stored as JSON on disk.

No extra packages are needed beyond `gradio` and `openai`.

## 1. Create a separate venv with `uv`

```bash
mkdir llama-gradio-chat
cd llama-gradio-chat

uv venv .venv --python 3.11

# Linux/macOS
source .venv/bin/activate

# Windows PowerShell
# .venv\Scripts\Activate.ps1

uv pip install "gradio>=5" openai
```

## 2. Save this as `app.py`

```python
#!/usr/bin/env python3
import os
import json
import uuid
import datetime
import pathlib
import traceback

import gradio as gr
from openai import OpenAI


# -----------------------------------------------------------------------------
# llama.cpp / OpenAI-compatible client
# -----------------------------------------------------------------------------

BASE_URL = "http://localhost:8001/v1"
API_KEY = "not-needed"

# Prefer an explicit model name if provided:
#   LLAMA_MODEL="my-model-name" python app.py
# or:
#   MODEL="my-model-name" python app.py
MODEL = os.getenv("LLAMA_MODEL") or os.getenv("MODEL")

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)

if not MODEL:
    try:
        listed = client.models.list()
        model_list = getattr(listed, "data", None)
        if model_list is None:
            model_list = list(listed)
        MODEL = model_list[0].id if model_list else "local-model"
    except Exception:
        MODEL = "local-model"

print(f"Using llama.cpp model: {MODEL}")


# -----------------------------------------------------------------------------
# Persistent session storage
# -----------------------------------------------------------------------------

SESSIONS_DIR = pathlib.Path(os.getenv("CHAT_SESSIONS_DIR", "./chat_sessions"))
SESSIONS_DIR.mkdir(parents=True, exist_ok=True)

VALID_ROLES = {"system", "user", "assistant"}
ERROR_PREFIX = "⚠️ llama.cpp request failed:"
EMPTY_ASSISTANT_TEXTS = {"", "(no response)"}


def now_iso():
    """Return current UTC timestamp in ISO 8601 format."""
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def format_time(iso_str):
    """Human-friendly timestamp for the dropdown."""
    if not iso_str:
        return "unknown"

    try:
        dt = datetime.datetime.fromisoformat(str(iso_str))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=datetime.timezone.utc)
        return dt.astimezone().strftime("%Y-%m-%d %H:%M")
    except Exception:
        return str(iso_str)


def title_from_text(text):
    """Create a short title from the first user message."""
    if not text or not text.strip():
        return "New chat"

    first_line = text.strip().splitlines()[0]
    first_line = " ".join(first_line.split())

    if len(first_line) > 50:
        first_line = first_line[:47] + "..."

    return first_line or "New chat"


def session_path(session_id):
    return SESSIONS_DIR / f"{session_id}.json"


def normalize_messages(raw_messages):
    """
    Keep messages compatible with gr.Chatbot(type="messages") and OpenAI chat format.
    """
    normalized = []

    if not isinstance(raw_messages, list):
        return normalized

    for msg in raw_messages:
        if not isinstance(msg, dict):
            continue

        role = msg.get("role")
        content = msg.get("content")

        if role not in VALID_ROLES:
            continue

        normalized.append(
            {
                "role": role,
                "content": "" if content is None else str(content),
            }
        )

    return normalized


def new_session(title="New chat", messages=None):
    """Create a new in-memory session object."""
    ts = now_iso()

    return {
        "id": uuid.uuid4().hex,
        "title": title,
        "created_at": ts,
        "updated_at": ts,
        "messages": normalize_messages(messages),
    }


def save_session(session):
    """
    Persist a session as JSON.

    JSON shape:
    {
        "id": "...",
        "title": "...",
        "created_at": "...",
        "updated_at": "...",
        "messages": [
            {"role": "user", "content": "..."},
            {"role": "assistant", "content": "..."}
        ]
    }
    """
    if not isinstance(session, dict):
        return None

    if not session.get("id"):
        return None

    session["updated_at"] = now_iso()
    session.setdefault("title", "New chat")
    session.setdefault("created_at", session["updated_at"])
    session["messages"] = normalize_messages(session.get("messages", []))

    if not session.get("title") or session.get("title") == "New chat":
        for msg in session["messages"]:
            if msg["role"] == "user" and msg["content"].strip():
                session["title"] = title_from_text(msg["content"])
                break

    path = session_path(session["id"])
    tmp_path = path.with_name(path.name + ".tmp")

    try:
        with tmp_path.open("w", encoding="utf-8") as f:
            json.dump(session, f, ensure_ascii=False, indent=2, sort_keys=False)

        os.replace(tmp_path, path)
    except Exception as exc:
        print(f"Failed to save session {session.get('id')}: {exc}")

    return session


def load_session(session_id):
    """Load a session from disk by id."""
    if not session_id:
        return None

    path = session_path(session_id)

    if not path.exists():
        return None

    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception:
        return None

    if not isinstance(data, dict):
        return None

    if data.get("id") != session_id:
        return None

    data.setdefault("title", "New chat")
    data.setdefault("created_at", data.get("updated_at") or now_iso())
    data.setdefault("updated_at", data["created_at"])
    data["messages"] = normalize_messages(data.get("messages", []))

    if not data.get("title") or data.get("title") == "New chat":
        for msg in data["messages"]:
            if msg["role"] == "user" and msg["content"].strip():
                data["title"] = title_from_text(msg["content"])
                break

    return data


def list_sessions():
    """
    List all saved sessions sorted by most recently updated first.
    """
    sessions = []

    for path in SESSIONS_DIR.glob("*.json"):
        if path.name.endswith(".tmp"):
            continue

        try:
            with path.open("r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        session_id = str(data.get("id", path.stem))

        data["id"] = session_id
        data.setdefault("title", "New chat")
        data.setdefault("created_at", data.get("updated_at") or now_iso())
        data.setdefault("updated_at", data["created_at"])
        data["messages"] = normalize_messages(data.get("messages", []))

        sessions.append(data)

    sessions.sort(key=lambda s: str(s.get("updated_at") or ""), reverse=True)
    return sessions


def session_label(session):
    """Label shown in the dropdown."""
    title = session.get("title") or "New chat"
    title = " ".join(str(title).split())

    if len(title) > 42:
        title = title[:39] + "..."

    updated = format_time(session.get("updated_at"))
    return f"{title} — {updated}"


def session_choices():
    sessions = list_sessions()
    return [(session_label(s), str(s["id"])) for s in sessions]


def dropdown_selected(session_id):
    """
    Update dropdown choices and select a session.
    """
    choices = session_choices()
    valid_values = {value for _, value in choices}

    if session_id not in valid_values:
        session_id = None

    return gr.update(
        choices=choices,
        value=session_id,
    )


# -----------------------------------------------------------------------------
# OpenAI / llama.cpp request helpers
# -----------------------------------------------------------------------------

def completion_kwargs():
    """
    Optional generation parameters.

    You can set environment variables:
      TEMPERATURE=0.7
      MAX_TOKENS=1024
    """
    kwargs = {}

    temperature = os.getenv("TEMPERATURE")
    if temperature:
        try:
            kwargs["temperature"] = float(temperature)
        except ValueError:
            pass

    max_tokens = os.getenv("MAX_TOKENS")
    if max_tokens:
        try:
            kwargs["max_tokens"] = int(max_tokens)
        except ValueError:
            pass

    return kwargs


def build_openai_messages(history):
    """
    Convert Gradio Chatbot messages to OpenAI chat messages.

    Empty assistant placeholders and request errors are skipped so they do not
    pollute future prompts.
    """
    out = []

    for msg in history:
        if not isinstance(msg, dict):
            continue

        role = msg.get("role")
        if role not in VALID_ROLES:
            continue

        content = msg.get("content")
        text = "" if content is None else str(content)

        if role == "assistant":
            stripped = text.strip()

            if (
                stripped in EMPTY_ASSISTANT_TEXTS
                or stripped.startswith(ERROR_PREFIX)
            ):
                continue

        out.append({"role": role, "content": text})

    return out


# -----------------------------------------------------------------------------
# Gradio handlers
# -----------------------------------------------------------------------------

def respond(message, history, busy, session_id):
    """
    Main chat handler.

    - Adds the user message to the Gradio message list.
    - Creates a session if needed.
    - Sends the full conversation to llama.cpp.
    - Streams assistant tokens into the UI.
    - Saves messages to disk.
    """
    history = list(history or [])

    # Do not allow overlapping generations.
    if busy:
        yield (
            history.copy(),
            gr.update(),
            gr.update(),
            True,
            session_id,
            gr.update(),
        )
        return

    if not message or not message.strip():
        yield (
            history.copy(),
            gr.update(value="", interactive=True),
            gr.update(interactive=True),
            False,
            session_id,
            gr.update(),
        )
        return

    user_text = message.strip()

    # Add user message to visible history.
    history.append({"role": "user", "content": user_text})

    # Load existing session or create a new one.
    session = load_session(session_id) if session_id else None

    if session is None:
        session = new_session(
            title=title_from_text(user_text),
            messages=history,
        )
        session_id = session["id"]
        save_session(session)
        dropdown_update = dropdown_selected(session_id)
    else:
        session["messages"] = history.copy()
        session["title"] = session.get("title") or title_from_text(user_text)
        save_session(session)
        dropdown_update = gr.update()

    # Build request messages BEFORE adding the empty assistant placeholder.
    request_messages = build_openai_messages(history)

    # Add assistant placeholder for streamed output.
    history.append({"role": "assistant", "content": ""})

    yield (
        history.copy(),
        gr.update(value="", interactive=False),
        gr.update(interactive=False),
        True,
        session_id,
        dropdown_update,
    )

    assistant_text = ""
    delta_count = 0

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
            **completion_kwargs(),
        )

        for chunk in stream:
            choices = getattr(chunk, "choices", None)

            if not choices:
                continue

            choice = choices[0]
            delta = getattr(getattr(choice, "delta", None), "content", None)

            if not delta:
                continue

            assistant_text += delta

            history[-1] = {
                "role": "assistant",
                "content": assistant_text,
            }

            yield (
                history.copy(),
                gr.update(),
                gr.update(),
                True,
                session_id,
                gr.update(),
            )

            delta_count += 1

            # Periodically persist partial response so a crash does not lose it.
            if delta_count % 50 == 0:
                session["messages"] = history.copy()
                save_session(session)

    except Exception as exc:
        traceback.print_exc()

        if assistant_text and not assistant_text.endswith("\n"):
            error_text = f"\n\n{ERROR_PREFIX} {exc}"
        elif assistant_text:
            error_text = f"\n{ERROR_PREFIX} {exc}"
        else:
            error_text = f"{ERROR_PREFIX} {exc}"

        assistant_text = (assistant_text + error_text).strip()

        history[-1] = {
            "role": "assistant",
            "content": assistant_text,
        }

        yield (
            history.copy(),
            gr.update(),
            gr.update(),
            True,
            session_id,
            gr.update(),
        )

    # If the model produced no visible content, keep a small placeholder.
    if not assistant_text.strip():
        history[-1] = {
            "role": "assistant",
            "content": "(no response)",
        }
    else:
        history[-1] = {
            "role": "assistant",
            "content": assistant_text,
        }

    # Final save.
    session["messages"] = history.copy()
    save_session(session)

    yield (
        history.copy(),
        gr.update(value="", interactive=True),
        gr.update(interactive=True),
        False,
        session_id,
        dropdown_selected(session_id),
    )


def load_from_dropdown(selected_id, busy, current_session_id):
    """
    Load a session when the user selects it from the dropdown.
    """
    # While a generation is running, ignore manual session changes.
    if busy:
        return (
            gr.update(),
            current_session_id,
            dropdown_selected(current_session_id),
            gr.update(),
            True,
        )

    session = load_session(selected_id)

    if session is not None:
        return (
            session["messages"],
            str(session["id"]),
            gr.update(),
            gr.update(interactive=True),
            False,
        )

    # If the selected session no longer exists, revert to the current valid one.
    if current_session_id:
        current = load_session(current_session_id)

        if current is not None:
            return (
                current["messages"],
                str(current["id"]),
                dropdown_selected(current["id"]),
                gr.update(interactive=True),
                False,
            )

    return (
        [],
        None,
        dropdown_selected(None),
        gr.update(interactive=True),
        False,
    )


def new_chat(busy, current_session_id):
    """
    Start a new chat.

    The session is not written to disk until the user sends the first message.
    """
    if busy:
        return (
            gr.update(),
            current_session_id,
            dropdown_selected(current_session_id),
            gr.update(),
            True,
        )

    return (
        [],
        None,
        dropdown_selected(None),
        gr.update(interactive=True),
        False,
    )


def delete_current(session_id, busy):
    """
    Delete the current session from disk.

    Then load the most recent remaining session, if any.
    """
    if busy:
        return (
            gr.update(),
            session_id,
            dropdown_selected(session_id),
            gr.update(),
            True,
        )

    if session_id:
        path = session_path(session_id)

        try:
            if path.exists():
                path.unlink()
        except Exception as exc:
            print(f"Failed to delete session {session_id}: {exc}")

    sessions = list_sessions()

    if sessions:
        nxt = sessions[0]
        return (
            nxt["messages"],
            str(nxt["id"]),
            dropdown_selected(nxt["id"]),
            gr.update(interactive=True),
            False,
        )

    return (
        [],
        None,
        dropdown_selected(None),
        gr.update(interactive=True),
        False,
    )


# -----------------------------------------------------------------------------
# Initial UI state
# -----------------------------------------------------------------------------

initial_sessions = list_sessions()
initial_session = initial_sessions[0] if initial_sessions else None

initial_session_id = str(initial_session["id"]) if initial_session else None
initial_messages = initial_session["messages"] if initial_session else []
initial_choices = [(session_label(s), str(s["id"])) for s in initial_sessions]


# -----------------------------------------------------------------------------
# Gradio UI
# -----------------------------------------------------------------------------

with gr.Blocks(title="llama.cpp Chat") as demo:
    busy = gr.State(False)
    current_session = gr.State(initial_session_id)

    with gr.Row():
        # Sidebar
        with gr.Column(scale=1, min_width=260):
            gr.Markdown("### Chats")

            session_dropdown = gr.Dropdown(
                label="",
                choices=initial_choices,
                value=initial_session_id,
                interactive=True,
            )

            new_btn = gr.Button("New Chat", variant="secondary")
            delete_btn = gr.Button("Delete Chat", variant="stop")

            gr.Markdown(
                "Sessions are saved as JSON in:\n\n"
                f"`{SESSIONS_DIR.resolve()}`"
            )

        # Main chat area
        with gr.Column(scale=4):
            gr.Markdown(
                f"## llama.cpp chat\n\n"
                f"Endpoint: `{BASE_URL}`  \n"
                f"Model: `{MODEL}`"
            )

            chatbot = gr.Chatbot(
                value=initial_messages,
                type="messages",
                height=620,
            )

            with gr.Row():
                user_input = gr.Textbox(
                    lines=1,
                    placeholder="Type a message...",
                    show_label=False,
                    scale=8,
                )
                send_btn = gr.Button("Send", variant="primary", scale=1)

    # Respond event: button click and Enter key.
    respond_inputs = [user_input, chatbot, busy, current_session]
    respond_outputs = [chatbot, user_input, send_btn, busy, current_session, session_dropdown]

    send_btn.click(
        respond,
        inputs=respond_inputs,
        outputs=respond_outputs,
    )

    user_input.submit(
        respond,
        inputs=respond_inputs,
        outputs=respond_outputs,
    )

    # Session dropdown events.
    load_inputs = [session_dropdown, busy, current_session]
    load_outputs = [chatbot, current_session, session_dropdown, send_btn, busy]

    session_dropdown.select(
        load_from_dropdown,
        inputs=load_inputs,
        outputs=load_outputs,
    )

    new_btn.click(
        new_chat,
        inputs=[busy, current_session],
        outputs=load_outputs,
    )

    delete_btn.click(
        delete_current,
        inputs=[current_session, busy],
        outputs=load_outputs,
    )


if __name__ == "__main__":
    # queue() is required for streaming responses.
    demo.queue(default_concurrency_limit=1).launch(
        server_name="0.0.0.0",
        server_port=7860,
    )
```
