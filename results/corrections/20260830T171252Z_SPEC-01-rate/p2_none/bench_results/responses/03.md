# [3] settings panel

## What changed

- Added a **Settings** panel (an `Accordion` in the sidebar) with sliders for `temperature`, `top_p`, `top_k`, `min_p`, `repeat_penalty`, `max_tokens` and a `system prompt` textbox.
- The OpenAI-standard params (`temperature`, `top_p`, `max_tokens`) go through the normal `chat.completions.create` arguments.
- The llama.cpp-only params (`top_k`, `min_p`, `repeat_penalty`) are passed via `extra_body={...}`, which the `openai` client forwards to the request body.
- The system prompt is prepended as a `{"role":"system"}` message on **every** request but is **not** stored in the chat history, so it never shows up in the transcript or gets re-sent as a user bubble.
- These settings are **global** controls (like llama.cpp's own WebUI) rather than per-session, so they apply to the next request from any session.

## 2. Save this as `app.py`

```python
#!/usr/bin/env python3
import os
import json
import uuid
import datetime
import pathlib
import traceback

import gradio as gr
from openai import OpenAI


# -----------------------------------------------------------------------------
# llama.cpp / OpenAI-compatible client
# -----------------------------------------------------------------------------

BASE_URL = "http://localhost:8001/v1"
API_KEY = "not-needed"

# Prefer an explicit model name if provided:
#   LLAMA_MODEL="my-model-name" python app.py
MODEL = os.getenv("LLAMA_MODEL") or os.getenv("MODEL")

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)

if not MODEL:
    try:
        listed = client.models.list()
        model_list = getattr(listed, "data", None)
        if model_list is None:
            model_list = list(listed)
        MODEL = model_list[0].id if model_list else "local-model"
    except Exception:
        MODEL = "local-model"

print(f"Using llama.cpp model: {MODEL}")


# -----------------------------------------------------------------------------
# Settings defaults (overridable via env vars)
# -----------------------------------------------------------------------------

def _f_env(name, default):
    try:
        return float(os.getenv(name))
    except (TypeError, ValueError):
        return default


def _i_env(name, default):
    try:
        return int(float(os.getenv(name)))
    except (TypeError, ValueError):
        return default


DEFAULT_TEMPERATURE = _f_env("TEMPERATURE", 0.8)
DEFAULT_TOP_P = _f_env("TOP_P", 0.95)
DEFAULT_TOP_K = _i_env("TOP_K", 40)
DEFAULT_MIN_P = _f_env("MIN_P", 0.0)
DEFAULT_REPEAT_PENALTY = _f_env("REPEAT_PENALTY", 1.10)
DEFAULT_MAX_TOKENS = _i_env("MAX_TOKENS", 1024)
DEFAULT_SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")


# -----------------------------------------------------------------------------
# Persistent session storage
# -----------------------------------------------------------------------------

SESSIONS_DIR = pathlib.Path(os.getenv("CHAT_SESSIONS_DIR", "./chat_sessions"))
SESSIONS_DIR.mkdir(parents=True, exist_ok=True)

VALID_ROLES = {"system", "user", "assistant"}
ERROR_PREFIX = "⚠️ llama.cpp request failed:"
EMPTY_ASSISTANT_TEXTS = {"", "(no response)"}


def now_iso():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def format_time(iso_str):
    if not iso_str:
        return "unknown"
    try:
        dt = datetime.datetime.fromisoformat(str(iso_str))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=datetime.timezone.utc)
        return dt.astimezone().strftime("%Y-%m-%d %H:%M")
    except Exception:
        return str(iso_str)


def title_from_text(text):
    if not text or not text.strip():
        return "New chat"
    first_line = " ".join(text.strip().splitlines()[0].split())
    if len(first_line) > 50:
        first_line = first_line[:47] + "..."
    return first_line or "New chat"


def session_path(session_id):
    return SESSIONS_DIR / f"{session_id}.json"


def normalize_messages(raw_messages):
    """Keep messages compatible with Chatbot(type='messages') and OpenAI format."""
    normalized = []
    if not isinstance(raw_messages, list):
        return normalized
    for msg in raw_messages:
        if not isinstance(msg, dict):
            continue
        role = msg.get("role")
        content = msg.get("content")
        if role not in VALID_ROLES:
            continue
        normalized.append({"role": role, "content": "" if content is None else str(content)})
    return normalized


def new_session(title="New chat", messages=None):
    ts = now_iso()
    return {
        "id": uuid.uuid4().hex,
        "title": title,
        "created_at": ts,
        "updated_at": ts,
        "messages": normalize_messages(messages),
    }


def save_session(session):
    """Persist a session as JSON: {id, title, created_at, updated_at, messages}."""
    if not isinstance(session, dict) or not session.get("id"):
        return None

    session["updated_at"] = now_iso()
    session.setdefault("title", "New chat")
    session.setdefault("created_at", session["updated_at"])
    session["messages"] = normalize_messages(session.get("messages", []))

    if not session.get("title") or session.get("title") == "New chat":
        for msg in session["messages"]:
            if msg["role"] == "user" and msg["content"].strip():
                session["title"] = title_from_text(msg["content"])
                break

    path = session_path(session["id"])
    tmp_path = path.with_name(path.name + ".tmp")
    try:
        with tmp_path.open("w", encoding="utf-8") as f:
            json.dump(session, f, ensure_ascii=False, indent=2)
        os.replace(tmp_path, path)
    except Exception as exc:
        print(f"Failed to save session {session.get('id')}: {exc}")
    return session


def load_session(session_id):
    if not session_id:
        return None
    path = session_path(session_id)
    if not path.exists():
        return None
    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception:
        return None
    if not isinstance(data, dict) or data.get("id") != session_id:
        return None

    data.setdefault("title", "New chat")
    data.setdefault("created_at", data.get("updated_at") or now_iso())
    data.setdefault("updated_at", data["created_at"])
    data["messages"] = normalize_messages(data.get("messages", []))
    if not data.get("title") or data.get("title") == "New chat":
        for msg in data["messages"]:
            if msg["role"] == "user" and msg["content"].strip():
                data["title"] = title_from_text(msg["content"])
                break
    return data


def list_sessions():
    sessions = []
    for path in SESSIONS_DIR.glob("*.json"):
        if path.name.endswith(".tmp"):
            continue
        try:
            with path.open("r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            continue
        if not isinstance(data, dict):
            continue
        data["id"] = str(data.get("id", path.stem))
        data.setdefault("title", "New chat")
        data.setdefault("created_at", data.get("updated_at") or now_iso())
        data.setdefault("updated_at", data["created_at"])
        data["messages"] = normalize_messages(data.get("messages", []))
        sessions.append(data)
    sessions.sort(key=lambda s: str(s.get("updated_at") or ""), reverse=True)
    return sessions


def session_label(session):
    title = " ".join(str(session.get("title") or "New chat").split())
    if len(title) > 42:
        title = title[:39] + "..."
    return f"{title} — {format_time(session.get('updated_at'))}"


def session_choices():
    return [(session_label(s), str(s["id"])) for s in list_sessions()]


def dropdown_selected(session_id):
    choices = session_choices()
    if session_id not in {v for _, v in choices}:
        session_id = None
    return gr.update(choices=choices, value=session_id)


# -----------------------------------------------------------------------------
# Request construction (OpenAI schema + llama.cpp extra_body)
# -----------------------------------------------------------------------------

def build_request_messages(history, system_prompt):
    """
    Prepend the system prompt (kept OUT of the saved transcript) and convert
    Gradio messages to OpenAI chat messages, skipping empty/error placeholders.
    """
    messages = []
    if system_prompt and system_prompt.strip():
        messages.append({"role": "system", "content": system_prompt.strip()})

    for msg in history:
        if not isinstance(msg, dict):
            continue
        role = msg.get("role")
        if role not in VALID_ROLES:
            continue
        text = "" if msg.get("content") is None else str(msg.get("content"))
        if role == "assistant":
            stripped = text.strip()
            if stripped in EMPTY_ASSISTANT_TEXTS or stripped.startswith(ERROR_PREFIX):
                continue
        messages.append({"role": role, "content": text})
    return messages


def split_settings(temperature, top_p, top_k, min_p, repeat_penalty, max_tokens):
    """
    Split UI values into (openai_kwargs, extra_body).

    - temperature, top_p, max_tokens are standard OpenAI chat.completions params.
    - top_k, min_p, repeat_penalty are llama.cpp-only and go into extra_body.
    """
    openai_kwargs = {
        "temperature": float(temperature),
        "top_p": float(top_p),
        "max_tokens": int(max_tokens),
    }
    extra_body = {
        "top_k": int(top_k),
        "min_p": float(min_p),
        "repeat_penalty": float(repeat_penalty),
    }
    return openai_kwargs, extra_body


# -----------------------------------------------------------------------------
# Gradio handlers
# -----------------------------------------------------------------------------

def respond(message, history, busy, session_id,
            temperature, top_p, top_k, min_p, repeat_penalty, max_tokens,
            system_prompt):
    history = list(history or [])

    if busy:
        yield history.copy(), gr.update(), gr.update(), True, session_id, gr.update()
        return

    if not message or not message.strip():
        yield history.copy(), gr.update(value="", interactive=True), \
              gr.update(interactive=True), False, session_id, gr.update()
        return

    user_text = message.strip()
    history.append({"role": "user", "content": user_text})

    # Load or create the session.
    session = load_session(session_id) if session_id else None
    if session is None:
        session = new_session(title=title_from_text(user_text), messages=history)
        session_id = session["id"]
        save_session(session)
        dropdown_update = dropdown_selected(session_id)
    else:
        session["messages"] = history.copy()
        session["title"] = session.get("title") or title_from_text(user_text)
        save_session(session)
        dropdown_update = gr.update()

    # Build the request (with system prompt) BEFORE the streaming placeholder.
    request_messages = build_request_messages(history, system_prompt)
    openai_kwargs, extra_body = split_settings(
        temperature, top_p, top_k, min_p, repeat_penalty, max_tokens
    )

    history.append({"role": "assistant", "content": ""})
    yield history.copy(), gr.update(value="", interactive=False), \
          gr.update(interactive=False), True, session_id, dropdown_update

    assistant_text = ""
    delta_count = 0

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
            extra_body=extra_body,   # llama.cpp-only sampling params
            **openai_kwargs,         # standard OpenAI sampling params
        )

        for chunk in stream:
            choices = getattr(chunk, "choices", None)
            if not choices:
                continue
            delta = getattr(getattr(choices[0], "delta", None), "content", None)
            if not delta:
                continue

            assistant_text += delta
            history[-1] = {"role": "assistant", "content": assistant_text}
            yield history.copy(), gr.update(), gr.update(), True, session_id, gr.update()

            delta_count += 1
            if delta_count % 50 == 0:
                session["messages"] = history.copy()
                save_session(session)

    except Exception as exc:
        traceback.print_exc()
        sep = "\n\n" if assistant_text else ""
        error_text = (assistant_text + sep + f"{ERROR_PREFIX} {exc}").strip()
        history[-1] = {"role": "assistant", "content": error_text}
        yield history.copy(), gr.update(), gr.update(), True, session_id, gr.update()
    finally:
        # Finalize and persist.
        final_text = assistant_text if assistant_text.strip() else "(no response)"
        # If an error already wrote text, keep it; otherwise keep final_text.
        if history and history[-1]["role"] == "assistant":
            if not history[-1]["content"].strip():
                history[-1] = {"role": "assistant", "content": final_text}

        session["messages"] = history.copy()
        save_session(session)

    yield history.copy(), gr.update(value="", interactive=True), \
          gr.update(interactive=True), False, session_id, dropdown_selected(session_id)


def load_from_dropdown(selected_id, busy, current_session_id):
    if busy:
        return gr.update(), current_session_id, dropdown_selected(current_session_id), \
               gr.update(), True

    session = load_session(selected_id)
    if session is not None:
        return session["messages"], str(session["id"]), gr.update(), \
               gr.update(interactive=True), False

    if current_session_id:
        current = load_session(current_session_id)
        if current is not None:
            return current["messages"], str(current["id"]), \
                   dropdown_selected(current["id"]), gr.update(interactive=True), False

    return [], None, dropdown_selected(None), gr.update(interactive=True), False


def new_chat(busy, current_session_id):
    if busy:
        return gr.update(), current_session_id, dropdown_selected(current_session_id), \
               gr.update(), True
    return [], None, dropdown_selected(None), gr.update(interactive=True), False


def delete_current(session_id, busy):
    if busy:
        return gr.update(), session_id, dropdown_selected(session_id), \
               gr.update(), True

    if session_id:
        path = session_path(session_id)
        try:
            if path.exists():
                path.unlink()
        except Exception as exc:
            print(f"Failed to delete session {session_id}: {exc}")

    sessions = list_sessions()
    if sessions:
        nxt = sessions[0]
        return nxt["messages"], str(nxt["id"]), dropdown_selected(nxt["id"]), \
               gr.update(interactive=True), False

    return [], None, dropdown_selected(None), gr.update(interactive=True), False


# -----------------------------------------------------------------------------
# Initial UI state
# -----------------------------------------------------------------------------

initial_sessions = list_sessions()
initial_session = initial_sessions[0] if initial_sessions else None
initial_session_id = str(initial_session["id"]) if initial_session else None
initial_messages = initial_session["messages"] if initial_session else []
initial_choices = [(session_label(s), str(s["id"])) for s in initial_sessions]


# -----------------------------------------------------------------------------
# Gradio UI
# -----------------------------------------------------------------------------

with gr.Blocks(title="llama.cpp Chat") as demo:
    busy = gr.State(False)
    current_session = gr.State(initial_session_id)

    with gr.Row():
        # ---- Sidebar ----
        with gr.Column(scale=1, min_width=300):
            gr.Markdown("### Chats")
            session_dropdown = gr.Dropdown(
                label="", choices=initial_choices,
                value=initial_session_id, interactive=True,
            )
            new_btn = gr.Button("New Chat", variant="secondary")
            delete_btn = gr.Button("Delete Chat", variant="stop")

            gr.Markdown("Sessions are saved as JSON in:\n\n"
                        f"`{SESSIONS_DIR.resolve()}`")

            # ---- Settings panel ----
            with gr.Accordion("Settings", open=False):
                gr.Markdown("### Generation (llama.cpp)")
                temperature = gr.Slider(0, 2, value=DEFAULT_TEMPERATURE, step=0.05,
                                        label="temperature")
                top_p = gr.Slider(0, 1, value=DEFAULT_TOP_P, step=0.01, label="top_p")
                top_k = gr.Slider(1, 200, value=DEFAULT_TOP_K, step=1, label="top_k")
                min_p = gr.Slider(0, 1, value=DEFAULT_MIN_P, step=0.01, label="min_p")
                repeat_penalty = gr.Slider(1, 2, value=DEFAULT_REPEAT_PENALTY, step=0.01,
                                           label="repeat_penalty")
                max_tokens = gr.Slider(1, 8192, value=DEFAULT_MAX_TOKENS, step=1,
                                       label="max_tokens")

                gr.Markdown("---\n### System prompt")
                system_prompt = gr.Textbox(
                    value=DEFAULT_SYSTEM_PROMPT,
                    lines=4,
                    label="System prompt",
                    placeholder="Applied to every request, never shown in history.",
                )
                gr.Markdown(
                    "*`temperature`, `top_p`, `max_tokens` go through the standard "
                    "OpenAI schema. `top_k`, `min_p`, `repeat_penalty` are sent in "
                    "`extra_body` for llama.cpp.*"
                )

        # ---- Main chat area ----
        with gr.Column(scale=4):
            gr.Markdown(f"## llama.cpp chat\n\nEndpoint: `{BASE_URL}`  \nModel: `{MODEL}`")
            chatbot = gr.Chatbot(value=initial_messages, type="messages", height=620)
            with gr.Row():
                user_input = gr.Textbox(lines=1, placeholder="Type a message...",
                                        show_label=False, scale=8)
                send_btn = gr.Button("Send", variant="primary", scale=1)

    # Settings values flow into every request as extra inputs.
    setting_inputs = [temperature, top_p, top_k, min_p, repeat_penalty,
                      max_tokens, system_prompt]

    respond_inputs = [user_input, chatbot, busy, current_session] + setting_inputs
    respond_outputs = [chatbot, user_input, send_btn, busy, current_session,
                       session_dropdown]

    send_btn.click(respond, inputs=respond_inputs, outputs=respond_outputs)
    user_input.submit(respond, inputs=respond_inputs, outputs=respond_outputs)

    load_inputs = [session_dropdown, busy, current_session]
    load_outputs = [chatbot, current_session, session_dropdown, send_btn, busy]

    session_dropdown.select(load_from_dropdown, inputs=load_inputs, outputs=load_outputs)
    new_btn.click(new_chat, inputs=[busy, current_session], outputs=load_outputs)
    delete_btn.click(delete_current, inputs=[current_session, busy], outputs=load_outputs)


if __name__ == "__main__":
    demo.queue(default_concurrency_limit=1).launch(
        server_name="0.0.0.0", server_port=7860,
    )
```

## 5. Notes on how the params are routed

| UI control | Where it goes | Why |
|---|---|---|
| `temperature` | top-level kwarg | in OpenAI chat.completions schema |
| `top_p` | top-level kwarg | in OpenAI chat.completions schema |
| `max_tokens` | top-level kwarg | in OpenAI chat.completions schema |
| `top_k` | `extra_body["top_k"]` | llama.cpp-only |
| `min_p` | `extra_body["min_p"]` | llama.cpp-only |
| `repeat_penalty` | `extra_body["repeat_penalty"]` | llama.cpp-only |
| `system prompt` | prepended `{"role":"system"}` message | not standard sampling; kept out of stored history |

This works because the `openai` client forwards `extra_body` keys into the raw JSON request body, and llama.cpp's OpenAI-compatible server reads them. If you later add more llama.cpp-specific knobs (e.g. `mirostat`, `seed`, `penalty_last_n`), just extend the `extra_body` dict in `split_settings` and add a matching slider.

Optional env overrides for the slider defaults:

```bash
TEMPERATURE=0.7 TOP_K=20 MAX_TOKENS=2048 \
SYSTEM_PROMPT="You are terse." python app.py
```
