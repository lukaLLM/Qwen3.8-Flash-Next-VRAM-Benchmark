# [1] build app

Here is a complete single-file Gradio chat app that talks to a llama.cpp server via the OpenAI-compatible endpoint, streams tokens into the UI, and keeps the visible `gr.Chatbot(type="messages")` list as the conversation history sent on every request.

## 1. Create a separate venv and install packages with `uv`

Run these commands in the directory where you want the app:

```bash
mkdir llama-gradio-chat
cd llama-gradio-chat

# Create a separate virtual environment
uv venv

# Activate it
# macOS/Linux:
source .venv/bin/activate

# Windows PowerShell:
# .venv\Scripts\Activate.ps1

# Windows cmd:
# .venv\Scripts\activate.bat

# Install packages
uv pip install "gradio>=5" "openai>=1"
```

You can also run the app without manually activating the venv:

```bash
uv run app.py
```

## 2. Save this as `app.py`

```python
#!/usr/bin/env python3
"""
Single-file Gradio chat UI for a llama.cpp OpenAI-compatible server.

Run:
    uv run app.py

The app connects by default to:
    http://localhost:8001/v1
"""

import os
import traceback

import gradio as gr
from openai import OpenAI

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("OPENAI_API_KEY", "not-needed")  # llama.cpp server may ignore this.
MODEL = os.getenv("LLAMA_CPP_MODEL", "local-model")
SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")
TEMPERATURE = float(os.getenv("TEMPERATURE", "0.7"))
TIMEOUT_SECONDS = float(os.getenv("OPENAI_TIMEOUT", "120"))

ERROR_TAG = "[llm-request-failed]"
ALLOWED_ROLES = {"system", "user", "assistant"}

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=TIMEOUT_SECONDS,
)


# -----------------------------------------------------------------------------
# Message helpers
# -----------------------------------------------------------------------------


def content_to_text(content):
    """
    Gradio Chatbot content can sometimes be multimodal.
    This flattens text-like content into a plain string for llama.cpp.
    """
    if content is None:
        return ""

    if isinstance(content, list):
        parts = []
        for part in content:
            if isinstance(part, dict):
                if part.get("type") == "text":
                    parts.append(str(part.get("text", "")))
                elif "text" in part:
                    parts.append(str(part["text"]))
            elif isinstance(part, str):
                parts.append(part)
            else:
                parts.append(str(part))
        return "".join(parts)

    return str(content)


def to_api_messages(ui_messages):
    """
    Convert Gradio Chatbot messages to OpenAI chat messages.
    The full visible user/assistant history is sent on every request.
    """
    api_messages = []

    if SYSTEM_PROMPT:
        api_messages.append(
            {
                "role": "system",
                "content": SYSTEM_PROMPT,
            }
        )

    for msg in ui_messages:
        role = msg.get("role")
        if role not in ALLOWED_ROLES:
            continue

        content = content_to_text(msg.get("content"))

        if not content.strip():
            continue

        # Keep UI error placeholders out of future model prompts.
        if content.strip().startswith(ERROR_TAG):
            continue

        if ERROR_TAG in content:
            content = content.split(ERROR_TAG, 1)[0].rstrip()

        if content.strip():
            api_messages.append(
                {
                    "role": role,
                    "content": content,
                }
            )

    return api_messages


# -----------------------------------------------------------------------------
# Chat handler
# -----------------------------------------------------------------------------


def respond(user_text, history):
    """
    Generator handler:
      - appends the user's message,
      - appends an assistant placeholder,
      - streams tokens into the assistant message,
      - yields updated Chatbot history to the UI.
    """
    user_text = (user_text or "").strip()
    messages = list(history or [])

    if not user_text:
        yield messages.copy(), ""
        return

    # Add user message.
    messages.append(
        {
            "role": "user",
            "content": user_text,
        }
    )
    yield messages.copy(), ""

    # Add assistant placeholder.
    assistant_msg = {
        "role": "assistant",
        "content": "",
    }
    messages.append(assistant_msg)
    yield messages.copy(), ""

    try:
        # Send full previous conversation plus the new user message.
        # Exclude the empty assistant placeholder from the request.
        api_messages = to_api_messages(messages[:-1])

        stream = client.chat.completions.create(
            model=MODEL,
            messages=api_messages,
            temperature=TEMPERATURE,
            stream=True,
        )

        got_text = False

        for chunk in stream:
            if not chunk.choices:
                continue

            choice = chunk.choices[0]
            delta = getattr(choice.delta, "content", None)

            if delta:
                assistant_msg["content"] += delta
                got_text = True
                yield messages.copy(), ""

        if not got_text:
            assistant_msg["content"] = "(The model returned no visible text.)"
            yield messages.copy(), ""

    except Exception as exc:
        traceback.print_exc()

        if assistant_msg["content"].strip():
            assistant_msg["content"] += f"\n\n{ERROR_TAG} {exc}"
        else:
            assistant_msg["content"] = f"{ERROR_TAG} {exc}"

        yield messages.copy(), ""


# -----------------------------------------------------------------------------
# Gradio UI
# -----------------------------------------------------------------------------

with gr.Blocks(title="llama.cpp Gradio Chat") as demo:
    gr.Markdown("## Local llama.cpp Chat")
    gr.Markdown(
        f"**Base URL:** `{BASE_URL}`  \n"
        f"**Model:** `{MODEL}`  \n"
        "This UI keeps the visible Chatbot message list as conversation history "
        "and sends that full history on every request."
    )

    chatbot = gr.Chatbot(
        type="messages",
        label="Conversation",
        height=600,
    )

    user_input = gr.Textbox(
        placeholder="Type a message...",
        show_label=False,
        lines=1,
    )

    with gr.Row():
        send_button = gr.Button("Send", variant="primary")
        clear_button = gr.ClearButton(
            [chatbot, user_input],
            value="Clear conversation",
        )

    def handle_submit(text, chat_history):
        yield from respond(text, chat_history)

    send_button.click(
        fn=handle_submit,
        inputs=[user_input, chatbot],
        outputs=[chatbot, user_input],
    )

    user_input.submit(
        fn=handle_submit,
        inputs=[user_input, chatbot],
        outputs=[chatbot, user_input],
    )


if __name__ == "__main__":
    demo.queue().launch()
```

## 3. Start llama.cpp server

Make sure llama.cpp is serving an OpenAI-compatible endpoint on port `8001`.

Example:

```bash
llama-server -m ./model.gguf --host 0.0.0.0 --port 8001
```

Check the available model name:

```bash
curl http://localhost:8001/v1/models
```

If llama.cpp expects a specific model name, run the Gradio app with:

```bash
LLAMA_CPP_MODEL=your-model-name uv run app.py
```

Example:

```bash
LLAMA_CPP_MODEL=local-model uv run app.py
```

Or:

```bash
LLAMA_CPP_MODEL=qwen2.5-coder-14b-instruct uv run app.py
```

The Gradio app should then be available at:

```text
http://127.0.0.1:7860
```
