# [2] chat history

Below is a complete single-file Gradio app with persistent chat sessions.

It saves each conversation to disk as JSON, lets you load previous chats from a dropdown, create a new chat, and delete the selected chat. Each saved session stores:

- `id`
- `title`
- `created_at`
- `updated_at`
- `messages`

The visible Gradio `Chatbot(type="messages")` list is still the multi-turn history sent to llama.cpp on every request.

## 2. Save this as `app.py`

```python
#!/usr/bin/env python3
"""
Gradio chat UI for a llama.cpp OpenAI-compatible server with persistent chat history.

Run:
    python app.py

Optional environment variables:
    LLAMA_CPP_BASE_URL=http://localhost:8001/v1
    LLAMA_CPP_MODEL=local-model
    OPENAI_API_KEY=not-needed
    SYSTEM_PROMPT="You are a helpful assistant."
    TEMPERATURE=0.7
    OPENAI_TIMEOUT=120
    CHAT_SESSION_DIR=chat_sessions
"""

import json
import os
import re
import traceback
import uuid
from datetime import datetime, timezone
from pathlib import Path

import gradio as gr
from openai import OpenAI

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("OPENAI_API_KEY", "not-needed")
MODEL = os.getenv("LLAMA_CPP_MODEL", "local-model")
SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")
TEMPERATURE = float(os.getenv("TEMPERATURE", "0.7"))
TIMEOUT_SECONDS = float(os.getenv("OPENAI_TIMEOUT", "120"))

SESSION_DIR = Path(os.getenv("CHAT_SESSION_DIR", "chat_sessions"))
SESSION_DIR.mkdir(parents=True, exist_ok=True)

ERROR_TAG = "[llm-request-failed]"
UNTITLED = "Untitled chat"
ALLOWED_ROLES = {"system", "user", "assistant"}

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=TIMEOUT_SECONDS,
)


# -----------------------------------------------------------------------------
# Utilities
# -----------------------------------------------------------------------------


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def content_to_text(content) -> str:
    """
    Gradio chatbot content can sometimes be multimodal.
    Flatten text-like content into a plain string for llama.cpp.
    """
    if content is None:
        return ""

    if isinstance(content, list):
        parts = []
        for part in content:
            if isinstance(part, dict):
                if part.get("type") == "text":
                    parts.append(str(part.get("text", "")))
                elif "text" in part:
                    parts.append(str(part["text"]))
            elif isinstance(part, str):
                parts.append(part)
            else:
                parts.append(str(part))
        return "".join(parts)

    return str(content)


def title_from_messages(messages) -> str:
    """
    Create a short title from the first user message.
    """
    for msg in messages:
        if isinstance(msg, dict) and msg.get("role") == "user":
            text = content_to_text(msg.get("content")).strip()
            if not text:
                continue
            text = text.replace("\n", " ").strip()
            if len(text) > 60:
                return text[:60].rstrip() + "…"
            return text

    return UNTITLED


# -----------------------------------------------------------------------------
# Session persistence
# -----------------------------------------------------------------------------


def session_path(session_id: str) -> Path:
    safe_id = re.sub(r"[^a-zA-Z0-9_.-]", "_", str(session_id))
    return SESSION_DIR / f"{safe_id}.json"


def write_session(session: dict) -> None:
    session_id = session.get("id")
    if not session_id:
        return

    path = session_path(session_id)
    tmp_path = path.with_suffix(".json.tmp")

    with open(tmp_path, "w", encoding="utf-8") as f:
        json.dump(session, f, ensure_ascii=False, indent=2)

    tmp_path.replace(path)


def read_session(session_id: str):
    if not session_id:
        return None

    path = session_path(session_id)
    if not path.exists():
        return None

    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception:
        return None

    if not isinstance(data, dict):
        return None

    # Normalize fields.
    data["id"] = session_id

    if not isinstance(data.get("messages"), list):
        data["messages"] = []

    if not data.get("created_at"):
        data["created_at"] = utc_now()

    if not data.get("updated_at"):
        data["updated_at"] = data.get("created_at", utc_now())

    if not str(data.get("title") or "").strip():
        data["title"] = title_from_messages(data["messages"])

    return data


def create_session(messages=None) -> str:
    """
    Create a new session file and return its ID.
    """
    messages = messages or []
    session_id = uuid.uuid4().hex

    session = {
        "id": session_id,
        "title": title_from_messages(messages),
        "created_at": utc_now(),
        "updated_at": utc_now(),
        "messages": messages,
    }

    write_session(session)
    return session_id


def save_session(session_id: str, messages):
    """
    Update an existing session, preserving created_at and existing title.
    """
    if not session_id:
        return None

    session = read_session(session_id)

    if session is None:
        session = {
            "id": session_id,
            "title": title_from_messages(messages),
            "created_at": utc_now(),
        }

    session["messages"] = messages
    session["updated_at"] = utc_now()

    # Keep manually-created/derived title if it is not still generic.
    current_title = str(session.get("title") or "").strip()
    if not current_title or current_title == UNTITLED:
        session["title"] = title_from_messages(messages)

    write_session(session)
    return session


def list_sessions():
    sessions = []

    for path in SESSION_DIR.glob("*.json"):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)

            if isinstance(data, dict) and data.get("id"):
                sessions.append(data)
        except Exception:
            continue

    sessions.sort(key=lambda s: s.get("updated_at", ""), reverse=True)
    return sessions


def choices_for_sessions(sessions, value=None):
    choices = []

    for session in sessions:
        session_id = session.get("id")
        title = session.get("title") or UNTITLED
        updated = str(session.get("updated_at", ""))[:16].replace("T", " ")
        label = f"{title} · {updated}"
        choices.append((label, session_id))

    valid_ids = {s.get("id") for s in sessions}

    if value not in valid_ids:
        value = sessions[0].get("id") if sessions else None

    return gr.update(choices=choices, value=value)


def remove_empty_sessions(keep_id=None):
    """
    Optional cleanup: remove JSON sessions with no messages.
    The new/current empty chat can be kept by passing keep_id.
    """
    for path in list(SESSION_DIR.glob("*.json")):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        sid = data.get("id") or path.stem

        if keep_id and sid == keep_id:
            continue

        if not data.get("messages"):
            try:
                path.unlink()
            except Exception:
                pass


# -----------------------------------------------------------------------------
# Chat conversion for OpenAI-compatible API
# -----------------------------------------------------------------------------


def to_api_messages(ui_messages):
    """
    Convert Gradio Chatbot messages to OpenAI-style messages.

    We send the full previous conversation every time, including system prompt.
    UI error placeholders are excluded from future prompts.
    """
    api_messages = []

    if SYSTEM_PROMPT:
        api_messages.append(
            {
                "role": "system",
                "content": SYSTEM_PROMPT,
            }
        )

    for msg in ui_messages:
        if not isinstance(msg, dict):
            continue

        role = msg.get("role")
        if role not in ALLOWED_ROLES:
            continue

        content = content_to_text(msg.get("content"))

        if not content.strip():
            continue

        # Do not send error placeholders to the model.
        if content.strip().startswith(ERROR_TAG):
            continue

        if ERROR_TAG in content:
            content = content.split(ERROR_TAG, 1)[0].rstrip()

        if content.strip():
            api_messages.append(
                {
                    "role": role,
                    "content": content,
                }
            )

    return api_messages


# -----------------------------------------------------------------------------
# UI event handlers
# -----------------------------------------------------------------------------


def on_app_load():
    """
    When the app loads, select the most recent existing session if one exists.
    If none exists, start with an empty session state and no active session ID.
    """
    sessions = list_sessions()

    if not sessions:
        return None, [], gr.update(choices=[], value=None)

    session = sessions[0]
    session_id = session.get("id")
    messages = session.get("messages", [])

    return session_id, messages, choices_for_sessions(sessions, session_id)


def new_chat():
    """
    Create a fresh chat session.
    """
    session_id = uuid.uuid4().hex
    now = utc_now()

    write_session(
        {
            "id": session_id,
            "title": UNTITLED,
            "created_at": now,
            "updated_at": now,
            "messages": [],
        }
    )

    # Clean up abandoned empty chats, except the one just created.
    remove_empty_sessions(keep_id=session_id)

    sessions = list_sessions()
    return (
        session_id,
        [],
        "",
        choices_for_sessions(sessions, session_id),
    )


def delete_session(selected_id, current_id, current_messages):
    """
    Delete the currently selected session from disk.

    If the deleted session is the active one, load the newest remaining session.
    If no sessions remain, reset to an empty state.
    """
    target_id = selected_id or current_id

    if not target_id:
        return None, [], gr.update(choices=[], value=None)

    path = session_path(target_id)

    try:
        if path.exists():
            path.unlink()
    except Exception:
        pass

    # Also remove a stale temp file if present.
    tmp_path = path.with_suffix(".json.tmp")
    try:
        if tmp_path.exists():
            tmp_path.unlink()
    except Exception:
        pass

    keep_current = bool(
        current_id
        and current_id != target_id
        and session_path(current_id).exists()
    )

    remove_empty_sessions(keep_id=current_id if keep_current else None)

    sessions = list_sessions()

    if not sessions:
        return None, [], gr.update(choices=[], value=None)

    if keep_current:
        return (
            current_id,
            current_messages or [],
            choices_for_sessions(sessions, current_id),
        )

    next_id = sessions[0].get("id")
    next_session = read_session(next_id) or {"messages": []}

    return (
        next_id,
        next_session.get("messages", []),
        choices_for_sessions(sessions, next_id),
    )


def select_from_dropdown(selected_id):
    """
    Load a session when the user selects one from the dropdown.
    """
    if not selected_id:
        return None, [], ""

    session = read_session(selected_id)
    messages = session.get("messages", []) if session else []

    return selected_id, messages, ""


def respond(user_text, history, session_id):
    """
    Streaming chat handler.

    Outputs:
      - chatbot history
      - user input
      - current session ID state
      - session dropdown
    """
    messages = list(history or [])
    user_text = (user_text or "").strip()

    if not user_text:
        yield messages, "", session_id, gr.update()
        return

    # Add the user's message to the visible history.
    messages.append(
        {
            "role": "user",
            "content": user_text,
        }
    )

    # Ensure we have a valid session file.
    if not session_id or not session_path(session_id).exists():
        session_id = create_session(messages)
        dropdown_update = choices_for_sessions(list_sessions(), session_id)
    else:
        save_session(session_id, messages)
        dropdown_update = choices_for_sessions(list_sessions(), session_id)

    # Show the user message immediately and update dropdown/state.
    yield messages, "", session_id, dropdown_update

    # Add assistant placeholder.
    assistant_msg = {
        "role": "assistant",
        "content": "",
    }
    messages.append(assistant_msg)

    yield messages.copy(), "", session_id, gr.update()

    try:
        # Send the full conversation except the empty assistant placeholder.
        api_messages = to_api_messages(messages[:-1])

        stream = client.chat.completions.create(
            model=MODEL,
            messages=api_messages,
            temperature=TEMPERATURE,
            stream=True,
        )

        got_text = False

        for chunk in stream:
            if not chunk.choices:
                continue

            choice = chunk.choices[0]
            delta = getattr(choice.delta, "content", None)

            if delta:
                assistant_msg["content"] += delta
                got_text = True
                yield messages.copy(), "", session_id, gr.update()

        if not got_text:
            assistant_msg["content"] = "(The model returned no visible text.)"
            yield messages.copy(), "", session_id, gr.update()

    except Exception as exc:
        traceback.print_exc()

        error_text = f"{ERROR_TAG} {exc}"

        if assistant_msg["content"].strip():
            assistant_msg["content"] += f"\n\n{error_text}"
        else:
            assistant_msg["content"] = error_text

        yield messages.copy(), "", session_id, gr.update()

    # Persist final assistant message, title, and timestamp.
    try:
        save_session(session_id, messages)
    except Exception:
        traceback.print_exc()

    final_dropdown_update = choices_for_sessions(list_sessions(), session_id)
    yield messages.copy(), "", session_id, final_dropdown_update


# -----------------------------------------------------------------------------
# Gradio UI
# -----------------------------------------------------------------------------

with gr.Blocks(title="llama.cpp Gradio Chat") as demo:
    gr.Markdown("## Local llama.cpp Chat with Persistent Sessions")

    gr.Markdown(
        f"**Base URL:** `{BASE_URL}`  \n"
        f"**Model:** `{MODEL}`  \n"
        f"**Session directory:** `{SESSION_DIR}`"
    )

    session_id_state = gr.State(value=None)

    with gr.Row():
        # Sidebar / session list
        with gr.Column(scale=1):
            session_dropdown = gr.Dropdown(
                label="Chats",
                choices=[],
                value=None,
                allow_custom_value=False,
                placeholder="Select a chat",
            )

            with gr.Row():
                new_button = gr.Button("New Chat", variant="primary")
                delete_button = gr.Button("Delete", variant="stop")

            gr.Markdown(
                "Sessions are stored locally as JSON files.\n\n"
                "`chat_sessions/<session-id>.json`"
            )

        # Main chat area
        with gr.Column(scale=3):
            chatbot = gr.Chatbot(
                type="messages",
                label="Conversation",
                height=650,
            )

            user_input = gr.Textbox(
                placeholder="Type a message...",
                show_label=False,
                lines=1,
            )

            send_button = gr.Button("Send", variant="primary")

    # Events
    new_button.click(
        fn=new_chat,
        outputs=[
            session_id_state,
            chatbot,
            user_input,
            session_dropdown,
        ],
    )

    delete_button.click(
        fn=delete_session,
        inputs=[
            session_dropdown,
            session_id_state,
            chatbot,
        ],
        outputs=[
            session_id_state,
            chatbot,
            session_dropdown,
        ],
    )

    session_dropdown.select(
        fn=select_from_dropdown,
        inputs=session_dropdown,
        outputs=[
            session_id_state,
            chatbot,
            user_input,
        ],
    )

    send_button.click(
        fn=respond,
        inputs=[
            user_input,
            chatbot,
            session_id_state,
        ],
        outputs=[
            chatbot,
            user_input,
            session_id_state,
            session_dropdown,
        ],
    )

    user_input.submit(
        fn=respond,
        inputs=[
            user_input,
            chatbot,
            session_id_state,
        ],
        outputs=[
            chatbot,
            user_input,
            session_id_state,
            session_dropdown,
        ],
    )

    demo.load(
        fn=on_app_load,
        outputs=[
            session_id_state,
            chatbot,
            session_dropdown,
        ],
    )


if __name__ == "__main__":
    demo.queue().launch()
```

## 3. Run llama.cpp server

Example:

```bash
llama-server -m ./model.gguf --host 0.0.0.0 --port 8001
```

Check the model name expected by the server:

```bash
curl http://localhost:8001/v1/models
```

If llama.cpp expects a specific model name, run the Gradio app with:

```bash
LLAMA_CPP_MODEL=your-model-name python app.py
```

Example:

```bash
LLAMA_CPP_MODEL=local-model python app.py
```

## 4. Run the Gradio app

With the virtual environment activated:

```bash
python app.py
```

Or, depending on your `uv` setup:

```bash
uv run app.py
```

The UI should appear at:

```text
http://127.0.0.1:7860
```

Sessions are saved in:

```text
chat_sessions/
```

Each file contains something like:

```json
{
  "id": "abc123...",
  "title": "Hello there",
  "created_at": "2025-01-01T00:00:00+00:00",
  "updated_at": "2025-01-01T00:01:00+00:00",
  "messages": [
    {
      "role": "user",
      "content": "Hello there"
    },
    {
      "role": "assistant",
      "content": "Hello! How can I help you today?"
    }
  ]
}
```
