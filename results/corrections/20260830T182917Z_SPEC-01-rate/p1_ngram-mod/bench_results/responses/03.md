def clean_number(value, default, min_value=None, max_value=None, integer=False):
    """
    Convert values coming from Gradio/env/file into a clean int or float.
    """
    if value is None:
        numeric = float(default)
    else:
        try:
            numeric = float(value)
        except (TypeError, ValueError):
            numeric = float(default)

    if math.isnan(numeric) or math.isinf(numeric):
        numeric = float(default)

    if min_value is not None:
        numeric = max(numeric, float(min_value))

    if max_value is not None:
        numeric = min(numeric, float(max_value))

    if integer:
        return int(round(numeric))

    return numeric

def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")

def clean_settings(
    temperature,
    top_p,
    top_k,
    min_p,
    repeat_penalty,
    max_tokens,
    system_prompt,
):
    cleaned_system = "" if system_prompt is None else str(system_prompt)
    cleaned_system = cleaned_system.strip()

    return {
        "temperature": clean_number(temperature, DEFAULT_TEMPERATURE, 0.0, 2.0),
        "top_p": clean_number(top_p, DEFAULT_TOP_P, 0.0, 1.0),
        "top_k": clean_number(top_k, DEFAULT_TOP_K, 1, 200, integer=True),
        "min_p": clean_number(min_p, DEFAULT_MIN_P, 0.0, 1.0),
        "repeat_penalty": clean_number(repeat_penalty, DEFAULT_REPEAT_PENALTY, 1.0, 2.0),
        "max_tokens": clean_number(max_tokens, DEFAULT_MAX_TOKENS, 1, 4096, integer=True),
        "system_prompt": cleaned_system,
    }


def default_settings_dict():
    return clean_settings(
        DEFAULT_TEMPERATURE,
        DEFAULT_TOP_P,
        DEFAULT_TOP_K,
        DEFAULT_MIN_P,
        DEFAULT_REPEAT_PENALTY,
        DEFAULT_MAX_TOKENS,
        DEFAULT_SYSTEM_PROMPT,
    )


def default_settings_tuple():
    s = default_settings_dict()
    return (
        s["temperature"],
        s["top_p"],
        s["top_k"],
        s["min_p"],
        s["repeat_penalty"],
        s["max_tokens"],
        s["system_prompt"],
    )


def settings_tuple_from_dict(settings):
    if not isinstance(settings, dict):
        return default_settings_tuple()

    system_prompt = settings.get("system_prompt")
    if system_prompt is None:
        system_prompt = DEFAULT_SYSTEM_PROMPT

    return (

        clean_number(settings.get("repeat_penalty"), DEFAULT_REPEAT_PENALTY, 1.0, 2.0),
        clean_number(settings.get("max_tokens"), DEFAULT_MAX_TOKENS, 1, 4096, integer=True),
        str(system_prompt),
    )

def content_to_text(content) -> str:
    """
    Gradio chatbot content can sometimes be multimodal.
    This flattens text-like content into a plain string.
    """
    if content is None:
        return ""

    if isinstance(content, list):
        parts = []

        for part in content:
            if isinstance(part, dict):
                if part.get("type") == "text":
                    parts.append(str(part.get("text", "")))
                elif "text" in part:
                    parts.append(str(part["text"]))
            elif isinstance(part, str):
                parts.append(part)
            else:
                parts.append(str(part))

        return "".join(parts)

    return str(content)


def title_from_messages(messages) -> str:
    """
    Make a short title from the first user message.
    """
    for msg in messages:
        if isinstance(msg, dict) and msg.get("role") == "user":
            text = content_to_text(msg.get("content")).strip()

            if not text:
                continue

            text = text.replace("\n", " ").strip()

            if len(text) > 60:
                return text[:60].rstrip() + "…"

            return text

    return UNTITLED


def to_api_messages(ui_messages, system_prompt: str):
    """
    Convert Gradio chatbot messages into OpenAI-style chat messages.

    The full previous visible user/assistant conversation is sent on every request.
    The system prompt comes from the Settings textbox, not from saved chat history,
    to avoid accidentally duplicating system messages.
    """
    api_messages = []

    if system_prompt:
        api_messages.append(
            {
                "role": "system",
                "content": system_prompt,
            }
        )

    for msg in ui_messages:
        if not isinstance(msg, dict):
            continue

        role = msg.get("role")

        if role not in SENDABLE_ROLES:
            continue

        content = content_to_text(msg.get("content"))

        if not content.strip():
            continue

        # Do not send UI error placeholders to the model.
        if content.strip().startswith(ERROR_TAG):
            continue

        if ERROR_TAG in content:
            content = content.split(ERROR_TAG, 1)[0].rstrip()

        if content.strip():
            api_messages.append(
                {
                    "role": role,
                    "content": content,
                }
            )

    return api_messages

def session_path(session_id: str) -> Path:
    safe_id = re.sub(r"[^a-zA-Z0-9_.-]", "_", str(session_id))
    return SESSION_DIR / f"{safe_id}.json"


def write_session(session: dict) -> None:
    session_id = session.get("id")

    if not session_id:
        return

    path = session_path(session_id)
    tmp_path = path.with_suffix(".json.tmp")

    with open(tmp_path, "w", encoding="utf-8") as f:
        json.dump(session, f, ensure_ascii=False, indent=2)

    tmp_path.replace(path)


def read_session(session_id: str):
    if not session_id:
        return None

    path = session_path(session_id)

    if not path.exists():
        return None

    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
    except Exception:
        return None

    if not isinstance(data, dict):
        return None

    data["id"] = session_id

    if not isinstance(data.get("messages"), list):
        data["messages"] = []

    if not data.get("created_at"):
        data["created_at"] = utc_now()

    if not data.get("updated_at"):
        data["updated_at"] = data.get("created_at", utc_now())

    if not isinstance(data.get("settings"), dict):
        data["settings"] = default_settings_dict()

    if not str(data.get("title") or "").strip():
        data["title"] = title_from_messages(data["messages"])

    return data


def create_session(messages=None, settings=None) -> str:
    messages = messages or []

    if settings is None:
        settings = default_settings_dict()

    session_id = uuid.uuid4().hex
    now = utc_now()

    session = {
        "id": session_id,
        "title": title_from_messages(messages),
        "created_at": now,
        "updated_at": now,
        "messages": messages,
        "settings": settings,
    }

    write_session(session)
    return session_id


def save_session(session_id: str, messages, settings=None):
    if not session_id:
        return None

    session = read_session(session_id)

    if session is None:
        return create_session(messages, settings)

    session["messages"] = messages
    session["updated_at"] = utc_now()

    current_title = str(session.get("title") or "").strip()
    if not current_title or current_title == UNTITLED:
        session["title"] = title_from_messages(messages)

    if settings is not None:
        session["settings"] = settings

    write_session(session)
    return session


def list_sessions():
    sessions = []

    for path in SESSION_DIR.glob("*.json"):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)

            if isinstance(data, dict) and data.get("id"):
                sessions.append(data)
        except Exception:
            continue

    sessions.sort(key=lambda s: str(s.get("updated_at", "")), reverse=True)
    return sessions


def choices_for_sessions(sessions, value=None):
    choices = []

    for session in sessions:
        session_id = session.get("id")
        title = session.get("title") or UNTITLED
        updated = str(session.get("updated_at", ""))[:16].replace("T", " ")
        label = f"{title} · {updated}"
        choices.append((label, session_id))

    valid_ids = {s.get("id") for s in sessions}

    if value not in valid_ids:
        value = sessions[0].get("id") if sessions else None

    return gr.update(choices=choices, value=value)


def remove_empty_sessions(keep_id=None):
    """
    Delete abandoned empty sessions from disk.

    keep_id lets you keep the current/new empty chat alive.
    """
    for path in list(SESSION_DIR.glob("*.json")):
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        sid = data.get("id") or path.stem

        if keep_id and sid == keep_id:
            continue

        if not data.get("messages"):
            try:
                path.unlink()
            except Exception:
                pass

def on_app_load():
    sessions = list_sessions()

    if not sessions:
        return (
            None,
            [],
            gr.update(choices=[], value=None),
            *default_settings_tuple(),
        )

    session_id = sessions[0].get("id")
    session = read_session(session_id) or sessions[0]

    return (
        session_id,
        session.get("messages", []),
        choices_for_sessions(sessions, session_id),
        *settings_tuple_from_dict(session.get("settings")),
    )


def new_chat(
    temperature,
    top_p,
    top_k,
    min_p,
    repeat_penalty,
    max_tokens,
    system_prompt,
):
    settings = clean_settings(
        temperature,
        top_p,
        top_k,
        min_p,
        repeat_penalty,
        max_tokens,
        system_prompt,
    )

    session_id = create_session([], settings)

    remove_empty_sessions(keep_id=session_id)

    sessions = list_sessions()

    return (
        session_id,
        [],
        "",
        choices_for_sessions(sessions, session_id),
    )


def select_from_dropdown(selected_id):
    if not selected_id:
        return (
            None,
            [],
            "",
            *default_settings_tuple(),
        )

    session = read_session(selected_id)

    if not session:
        return (
            None,
            [],
            "",
            *default_settings_tuple(),
        )

    return (
        selected_id,
        session.get("messages", []),
        "",
        *settings_tuple_from_dict(session.get("settings")),
    )


def delete_session(
    selected_id,
    current_id,
    current_messages,
    temperature,
    top_p,
    top_k,
    min_p,
    repeat_penalty,
    max_tokens,
    system_prompt,
):
    target_id = selected_id or current_id

    sessions = list_sessions()

    # If nothing is selected, load the newest session instead of crashing.
    if not target_id:
        if not sessions:
            return (
                None,
                [],
                gr.update(choices=[], value=None),
                *default_settings_tuple(),
            )

        next_id = sessions[0].get("id")
        session = read_session(next_id) or sessions[0]

        return (
            next_id,
            session.get("messages", []),
            choices_for_sessions(sessions, next_id),
            *settings_tuple_from_dict(session.get("settings")),
        )

    path = session_path(target_id)

    try:
        if path.exists():
            path.unlink()
    except Exception:
        pass

    tmp_path = path.with_suffix(".json.tmp")
    try:
        if tmp_path.exists():
            tmp_path.unlink()
    except Exception:
        pass

    keep_current = bool(
        current_id
        and current_id != target_id
        and session_path(current_id).exists()
    )

    remove_empty_sessions(keep_id=current_id if keep_current else None)

    sessions = list_sessions()

    if not sessions:
        return (
            None,
            [],
            gr.update(choices=[], value=None),
            *default_settings_tuple(),
        )

    if keep_current:
        return (
            current_id,
            current_messages or [],
            choices_for_sessions(sessions, current_id),
            temperature,
            top_p,
            top_k,
            min_p,
            repeat_penalty,
            max_tokens,
            system_prompt,
        )

    next_id = sessions[0].get("id")
    session = read_session(next_id) or sessions[0]

    return (
        next_id,
        session.get("messages", []),
        choices_for_sessions(sessions, next_id),
        *settings_tuple_from_dict(session.get("settings")),
    )


def respond(
    user_text,
    history,
    session_id,
    temperature,
    top_p,
    top_k,
    min_p,
    repeat_penalty,
    max_tokens,
    system_prompt,
):
    """
    Streaming chat handler.

    Outputs:
      - Chatbot history
      - User input
      - Session ID state
      - Session dropdown update
    """
    messages = list(history or [])
    user_text = (user_text or "").strip()

    settings = clean_settings(
        temperature,
        top_p,
        top_k,
        min_p,
        repeat_penalty,
        max_tokens,
        system_prompt,
    )

    if not user_text:
        yield messages, "", session_id, gr.update()
        return

    # Add the new user message.
    messages.append(
        {
            "role": "user",
            "content": user_text,
        }
    )

    # Ensure a valid session file exists.
    if not session_id or not session_path(session_id).exists():
        session_id = create_session(messages, settings)
    else:
        save_session(session_id, messages, settings)

    dropdown_update = choices_for_sessions(list_sessions(), session_id)

    # Show user message immediately.
    yield messages.copy(), "", session_id, dropdown_update

    # Add assistant placeholder.
    assistant_msg = {
        "role": "assistant",
        "content": "",
    }
    messages.append(assistant_msg)

    yield messages.copy(), "", session_id, gr.update()

    try:
        # Exclude the empty assistant placeholder from the request.
        api_messages = to_api_messages(messages[:-1], settings["system_prompt"])

        # Standard OpenAI-compatible params:
        #   temperature, top_p, max_tokens
        #
        # llama.cpp-specific params:
        #   top_k, min_p, repeat_penalty
        extra_body = {
            "top_k": settings["top_k"],
            "min_p": settings["min_p"],
            "repeat_penalty": settings["repeat_penalty"],
        }

        stream = client.chat.completions.create(
            model=MODEL,
            messages=api_messages,
            temperature=settings["temperature"],
            top_p=settings["top_p"],
            max_tokens=settings["max_tokens"],
            stream=True,
            extra_body=extra_body,
        )

        got_text = False

        for chunk in stream:
            if not chunk.choices:
                continue

            choice = chunk.choices[0]
            delta = getattr(choice.delta, "content", None)

            if delta:
                assistant_msg["content"] += delta
                got_text = True
                yield messages.copy(), "", session_id, gr.update()

        if not got_text:
            assistant_msg["content"] = "(The model returned no visible text.)"
            yield messages.copy(), "", session_id, gr.update()

    except Exception as exc:
        traceback.print_exc()

        error_text = f"{ERROR_TAG} {exc}"

        if assistant_msg["content"].strip():
            assistant_msg["content"] += f"\n\n{error_text}"
        else:
            assistant_msg["content"] = error_text

        yield messages.copy(), "", session_id, gr.update()

    # Persist final assistant response, updated title, timestamp, and settings.
    try:
        save_session(session_id, messages, settings)
    except Exception:
        traceback.print_exc()

    final_dropdown_update = choices_for_sessions(list_sessions(), session_id)

    yield messages.copy(), "", session_id, final_dropdown_update
