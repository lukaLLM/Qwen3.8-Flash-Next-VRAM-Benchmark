# [1] build app

Below is a complete single-file Gradio app that uses the `openai` Python client against a local llama.cpp OpenAI-compatible server and streams responses token-by-token.

## 1. Create a separate venv and install packages with `uv`

```bash
uv init llama-chat-gradio
cd llama-chat-gradio
uv add gradio openai
```

If `uv init` creates a `main.py`, you can ignore it or delete it.

## 2. Create `app.py`

```python
import os
import gradio as gr
from openai import OpenAI

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("LLAMA_CPP_API_KEY", "not-needed")
MODEL = os.getenv("LLAMA_CPP_MODEL", "local")
SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)


def _attr(obj, key, default=None):
    if isinstance(obj, dict):
        return obj.get(key, default)
    return getattr(obj, key, default)


def to_message_dict(message):
    if isinstance(message, dict):
        return {
            "role": message.get("role", "user"),
            "content": message.get("content", "") or "",
        }
    return {
        "role": getattr(message, "role", "user"),
        "content": getattr(message, "content", "") or "",
    }


def stream_reply(user_message, chat_history):
    user_message = (user_message or "").strip()

    if not user_message:
        yield chat_history or [], gr.update()
        return

    # Keep multi-turn history in the message list.
    history = [] if not chat_history else [to_message_dict(m) for m in chat_history]
    history.append({"role": "user", "content": user_message})

    # Send the full conversation on every request.
    request_messages = []
    if SYSTEM_PROMPT:
        request_messages.append({"role": "system", "content": SYSTEM_PROMPT})

    for msg in history:
        # Avoid sending an empty assistant placeholder from a previous failed stream.
        if msg.get("role") == "assistant" and not msg.get("content"):
            continue
        request_messages.append(msg)

    # Add an assistant placeholder that will be streamed into.
    history.append({"role": "assistant", "content": ""})

    # Show the user message immediately and disable input while generating.
    yield history, gr.update(value="", interactive=False)

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
        )

        for chunk in stream:
            choices = _attr(chunk, "choices", None)
            if not choices:
                continue

            delta = _attr(choices[0], "delta", None)
            token = _attr(delta, "content", None)

            if token:
                history[-1]["content"] += token if isinstance(token, str) else str(token)
                yield history, gr.update(value="", interactive=False)

    except Exception as exc:
        if history and history[-1].get("role") == "assistant":
            history[-1]["content"] = (
                (history[-1]["content"] or "") + f"\n\nRequest failed: {exc}"
            ).strip()
        else:
            history.append({"role": "assistant", "content": f"Request failed: {exc}"})

    yield history, gr.update(value="", interactive=True)


with gr.Blocks(title="llama.cpp Chat") as demo:
    gr.Markdown(
        f"""
        ## llama.cpp local chat

        Connecting to `{BASE_URL}` using an OpenAI-compatible client.
        """
    )

    chatbot = gr.Chatbot(type="messages", label="Conversation", height=600)

    with gr.Row():
        text_box = gr.Textbox(
            placeholder="Type a message and press Enter...",
            show_label=False,
            scale=10,
        )
        send_button = gr.Button("Send", variant="primary", scale=1)

    clear_button = gr.Button("Clear chat", variant="secondary")

    chat_inputs = [text_box, chatbot]
    chat_outputs = [chatbot, text_box]

    send_button.click(stream_reply, inputs=chat_inputs, outputs=chat_outputs)
    text_box.submit(stream_reply, inputs=chat_inputs, outputs=chat_outputs)
    clear_button.click(
        lambda: ([], gr.update(value="", interactive=True)),
        outputs=chat_outputs,
    )

if __name__ == "__main__":
    demo.queue().launch(server_name="127.0.0.1", server_port=7860)
```

## 3. Start llama.cpp server

Make sure your llama.cpp OpenAI-compatible server is running on `http://localhost:8001/v1`.

Example:

```bash
llama-server -m /path/to/model.gguf --host 127.0.0.1 --port 8001
```

Check available models if needed:

```bash
curl http://localhost:8001/v1/models
```

If the server requires a specific model name, set it:

```bash
LLAMA_CPP_MODEL=model-name uv run python app.py
```

## 4. Run the Gradio app

```bash
uv run python app.py
```

Then open:

```text
http://127.0.0.1:7860
```
