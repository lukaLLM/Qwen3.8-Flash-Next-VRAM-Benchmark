# [2] chat history

Here is a complete single-file Gradio app with persistent JSON chat history, a session dropdown, **New Chat**, **Delete**, and streaming from a local llama.cpp OpenAI-compatible server.

## 1. Create a separate venv with `uv`

```bash
uv init llama-chat-gradio
cd llama-chat-gradio
uv add gradio openai
```

`uv` creates and manages the project virtual environment in `.venv`.

You can remove `main.py` if `uv init` created one.

---

## 2. `app.py`

```python
import os
import json
import re
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import gradio as gr
from openai import OpenAI


# -----------------------------
# llama.cpp / OpenAI client
# -----------------------------

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("LLAMA_CPP_API_KEY", "not-needed")
MODEL = os.getenv("LLAMA_CPP_MODEL", "local")
SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)


# -----------------------------
# Session storage
# -----------------------------

SESSIONS_DIR = Path(os.getenv("CHAT_SESSIONS_DIR", "chats"))


def ensure_sessions_dir() -> None:
    SESSIONS_DIR.mkdir(parents=True, exist_ok=True)


ensure_sessions_dir()


def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def valid_session_id(session_id: Any) -> bool:
    if session_id is None:
        return False
    return bool(re.fullmatch(r"[A-Za-z0-9_-]+", str(session_id)))


def normalize_message(message: Any) -> dict[str, str]:
    if isinstance(message, dict):
        role = message.get("role") or "user"
        content = message.get("content", "")
    else:
        role = getattr(message, "role", "user")
        content = getattr(message, "content", "")

    role = str(role).lower()

    if content is None:
        content = ""
    elif not isinstance(content, str):
        content = str(content)

    return {"role": role, "content": content}


def to_conversation_messages(messages: Any, include_empty_assistant: bool = False) -> list[dict[str, str]]:
    """
    Convert Gradio / chat messages into simple role/content dicts.

    We intentionally store only user and assistant messages in session files.
    The system prompt is added again on each request.
    """
    out: list[dict[str, str]] = []

    for message in messages or []:
        msg = normalize_message(message)

        if msg["role"] not in {"user", "assistant"}:
            continue

        if not include_empty_assistant and msg["role"] == "assistant" and not msg["content"].strip():
            continue

        out.append(msg)

    return out


def has_messages(messages: Any) -> bool:
    return bool(to_conversation_messages(messages))


def generate_title(messages: list[dict[str, str]], fallback: str = "New Chat") -> str:
    for msg in messages:
        if msg.get("role") != "user":
            continue

        text = " ".join((msg.get("content") or "").split())
        if not text:
            continue

        if len(text) <= 60:
            return text
        return text[:59] + "…"

    return fallback


def list_sessions() -> list[tuple[str, str]]:
    """
    Returns dropdown choices as (title, session_id), sorted newest first.
    """
    ensure_sessions_dir()

    entries: list[tuple[str, str, str]] = []

    for path in SESSIONS_DIR.glob("*.json"):
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        session_id = data.get("id") or path.stem
        if not valid_session_id(session_id):
            continue

        messages = to_conversation_messages(data.get("messages", []))
        if not messages:
            continue

        title = data.get("title") or generate_title(messages)
        updated_at = data.get("updated_at") or data.get("created_at") or ""

        entries.append((str(updated_at), title, str(session_id)))

    entries.sort(key=lambda item: item[0], reverse=True)

    return [(title, session_id) for _updated_at, title, session_id in entries]


def load_session(session_id: Any) -> list[dict[str, str]]:
    if not valid_session_id(session_id):
        return []

    path = SESSIONS_DIR / f"{session_id}.json"

    if not path.exists():
        return []

    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except Exception:
        return []

    if not isinstance(data, dict):
        return []

    return to_conversation_messages(data.get("messages", []))


def save_session(session_id: Any, messages: Any) -> str | None:
    """
    Save a session to disk.

    Session JSON schema:
        {
          "id": "...",
          "title": "...",
          "created_at": "...",
          "updated_at": "...",
          "messages": [{"role": "user", "content": "..."}]
        }

    Returns the session id, or None if nothing was saved.
    """
    ensure_sessions_dir()

    if not valid_session_id(session_id):
        session_id = uuid.uuid4().hex

    messages_to_save = to_conversation_messages(messages)
    path = SESSIONS_DIR / f"{session_id}.json"

    # Do not keep empty sessions around.
    if not messages_to_save:
        if path.exists():
            try:
                path.unlink(missing_ok=True)
            except OSError:
                pass
        return None

    created_at = now_iso()
    title = None

    if path.exists():
        try:
            existing = json.loads(path.read_text(encoding="utf-8"))
            if isinstance(existing, dict):
                created_at = existing.get("created_at") or created_at
                title = existing.get("title")
        except Exception:
            pass

    # Re-title untitled sessions automatically.
    if not title or title in {"", "New Chat"}:
        title = generate_title(messages_to_save)

    payload = {
        "id": session_id,
        "title": title,
        "created_at": created_at,
        "updated_at": now_iso(),
        "messages": messages_to_save,
    }

    # Atomic-ish write.
    tmp_path = path.with_suffix(".tmp")
    tmp_path.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
    os.replace(tmp_path, path)

    return session_id


# -----------------------------
# Helper for streaming chunks
# -----------------------------

def _attr(obj: Any, key: str, default: Any = None) -> Any:
    if isinstance(obj, dict):
        return obj.get(key, default)
    return getattr(obj, key, default)


# -----------------------------
# Gradio callbacks
# -----------------------------

def load_selected(selected: Any, state_session_id: Any, current_messages: Any):
    """
    Dropdown change handler.

    Inputs:
        selected: selected session id
        state_session_id: current session id
        current_messages: current chatbot messages

    Outputs:
        chatbot
        session_id state
        textbox
    """
    if not valid_session_id(selected):
        return [], None, gr.update(value="")

    # If the user re-selects the current session, preserve the visible chat.
    # This also avoids clobbering a streaming response.
    if selected == state_session_id and current_messages:
        return current_messages, selected, gr.update(value="")

    loaded = load_session(selected)
    return loaded, selected, gr.update(value="")


def new_chat(chat_messages: Any, session_id: Any):
    """
    Save the current session if it has content, then clear the UI.

    Outputs:
        chatbot
        session_id
        textbox
        dropdown
    """
    if valid_session_id(session_id) and has_messages(chat_messages):
        save_session(session_id, chat_messages)

    return (
        [],
        None,
        gr.update(value="", interactive=True),
        gr.update(choices=list_sessions(), value=None, interactive=True),
    )


def delete_chat(session_id: Any):
    """
    Delete the current session file and clear the UI.

    Outputs:
        chatbot
        session_id
        textbox
        dropdown
    """
    ensure_sessions_dir()

    if valid_session_id(session_id):
        path = SESSIONS_DIR / f"{session_id}.json"
        try:
            path.unlink(missing_ok=True)
        except OSError:
            pass

    return (
        [],
        None,
        gr.update(value="", interactive=True),
        gr.update(choices=list_sessions(), value=None, interactive=True),
    )


def refresh_dropdown(current_value: Any):
    """
    Refresh dropdown choices while trying to keep the current selection.
    """
    choices = list_sessions()
    values = {value for _label, value in choices}

    new_value = current_value if valid_session_id(current_value) and str(current_value) in values else None

    return gr.update(
        choices=choices,
        value=new_value,
        interactive=True,
    )


def stream_reply(user_text: str, chat_messages: Any, session_id: Any):
    """
    Streams a response from llama.cpp and yields updates to the UI.

    Keeps multi-turn history in the message list and sends the full conversation
    on every request.
    """
    user_text = (user_text or "").strip()

    # If the user clicks Send with no text, just clean up the textbox.
    if not user_text:
        yield (
            chat_messages or [],
            gr.update(value=""),
            session_id,
            gr.skip(),
            gr.update(interactive=True),
        )
        return

    history = to_conversation_messages(chat_messages)
    history.append({"role": "user", "content": user_text})

    # Build the request message list: system prompt + full conversation so far.
    request_messages = []

    if SYSTEM_PROMPT:
        request_messages.append({"role": "system", "content": SYSTEM_PROMPT})

    request_messages.extend(to_conversation_messages(history))

    # Save the user message before generating.
    session_id = save_session(session_id, history)

    # Assistant placeholder that will be filled token-by-token.
    history.append({"role": "assistant", "content": ""})

    text_disabled = gr.update(value="", interactive=False)
    text_enabled = gr.update(value="", interactive=True)

    dropdown_initial = gr.update(
        choices=list_sessions(),
        value=session_id,
        interactive=False,
    )

    send_disabled = gr.update(interactive=False)
    send_enabled = gr.update(interactive=True)

    # Show the user message immediately and block further input.
    yield history, text_disabled, session_id, dropdown_initial, send_disabled

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
        )

        for chunk in stream:
            choices = _attr(chunk, "choices", None)

            if not choices:
                continue

            delta = _attr(choices[0], "delta", None)
            token = _attr(delta, "content", None)

            if token:
                history[-1]["content"] += token if isinstance(token, str) else str(token)
                yield history, text_disabled, session_id, gr.skip(), gr.skip()

    except Exception as exc:
        if history and history[-1].get("role") == "assistant":
            history[-1]["content"] = (
                (history[-1]["content"] or "") + f"\n\nRequest failed: {exc}"
            ).strip()
        else:
            history.append({"role": "assistant", "content": f"Request failed: {exc}"})

        session_id = save_session(session_id, history)

        yield (
            history,
            text_enabled,
            session_id,
            gr.update(choices=list_sessions(), value=session_id, interactive=True),
            send_enabled,
        )
        return

    # Save final assistant message.
    session_id = save_session(session_id, history)

    yield (
        history,
        text_enabled,
        session_id,
        gr.update(choices=list_sessions(), value=session_id, interactive=True),
        send_enabled,
    )


# -----------------------------
# UI
# -----------------------------

with gr.Blocks(title="llama.cpp Local Chat") as demo:
    session_id = gr.State(value=None)

    gr.Markdown(
        f"""
        ## llama.cpp local chat

        Connecting to `{BASE_URL}` with an OpenAI-compatible client.

        Sessions are saved as JSON files under `{SESSIONS_DIR.resolve()}`.
        """
    )

    with gr.Row():
        with gr.Column(scale=1, min_width=280):
            gr.Markdown("### Sessions")

            dropdown = gr.Dropdown(
                choices=list_sessions(),
                label="Saved chats",
                value=None,
                interactive=True,
                allow_custom_value=False,
            )

            with gr.Row():
                new_button = gr.Button("New Chat", variant="primary")
                delete_button = gr.Button("Delete", variant="stop")

            refresh_button = gr.Button("Refresh")

        with gr.Column(scale=4):
            chatbot = gr.Chatbot(
                label="Conversation",
                type="messages",
                height=620,
                show_copy_button=True,
                autoscroll=True,
            )

            with gr.Row():
                textbox = gr.Textbox(
                    placeholder="Type a message and press Enter...",
                    show_label=False,
                    scale=10,
                )
                send_button = gr.Button("Send", variant="primary", scale=1)

    # Streaming chat outputs.
    chat_outputs = [
        chatbot,
        textbox,
        session_id,
        dropdown,
        send_button,
    ]

    send_button.click(
        stream_reply,
        inputs=[textbox, chatbot, session_id],
        outputs=chat_outputs,
    )

    textbox.submit(
        stream_reply,
        inputs=[textbox, chatbot, session_id],
        outputs=chat_outputs,
    )

    # Session management.
    dropdown.change(
        load_selected,
        inputs=[dropdown, session_id, chatbot],
        outputs=[chatbot, session_id, textbox],
    )

    new_button.click(
        new_chat,
        inputs=[chatbot, session_id],
        outputs=[chatbot, session_id, textbox, dropdown],
    )

    delete_button.click(
        delete_chat,
        inputs=[session_id],
        outputs=[chatbot, session_id, textbox, dropdown],
    )

    refresh_button.click(
        refresh_dropdown,
        inputs=[dropdown],
        outputs=[dropdown],
    )

    # Useful if another tab/process created sessions.
    demo.load(
        refresh_dropdown,
        inputs=[dropdown],
        outputs=[dropdown],
    )


if __name__ == "__main__":
    demo.queue().launch(server_name="127.0.0.1", server_port=7860)
```

---

## 3. Start llama.cpp

Make sure llama.cpp is running on:

```bash
http://localhost:8001/v1
```

Example:

```bash
llama-server -m /path/to/model.gguf --host 127.0.0.1 --port 8001
```

Check available models if needed:

```bash
curl http://localhost:8001/v1/models
```

If llama.cpp requires a specific model name, set it:

```bash
LLAMA_CPP_MODEL=model-name-here uv run python app.py
```

Other optional environment variables:

```bash
LLAMA_CPP_BASE_URL=http://localhost:8001/v1
LLAMA_CPP_API_KEY=not-needed
LLAMA_CPP_MODEL=local
SYSTEM_PROMPT=You are a helpful assistant.
CHAT_SESSIONS_DIR=chats
```

---

## 4. Run the app

```bash
uv run python app.py
```

Then open:

```text
http://127.0.0.1:7860
```

Each conversation is saved as a JSON file in `./chats/`, containing:

```json
{
  "id": "...",
  "title": "...",
  "created_at": "...",
  "updated_at": "...",
  "messages": [
    {
      "role": "user",
      "content": "..."
    },
    {
      "role": "assistant",
      "content": "..."
    }
  ]
}
```
