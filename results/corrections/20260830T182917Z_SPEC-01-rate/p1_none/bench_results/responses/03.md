# [3] settings panel

Below is a complete single-file version with:

- persistent JSON session history
- sidebar dropdown to load past sessions
- **New Chat**
- **Delete**
- settings panel with sliders for:
  - `temperature`
  - `top_p`
  - `top_k`
  - `min_p`
  - `repeat_penalty`
  - `max_tokens`
- a `system_prompt` textbox
- llama.cpp-specific parameters sent through `extra_body`

## 1. Project setup with `uv`

```bash
uv init llama-chat-gradio
cd llama-chat-gradio
uv add gradio openai
```

You can delete `main.py` if it was created.

## 2. `app.py`

```python
import os
import json
import re
import uuid
from datetime import datetime, timezone
from pathlib import Path

import gradio as gr
from openai import OpenAI


# -----------------------------
# llama.cpp / OpenAI client
# -----------------------------

BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("LLAMA_CPP_API_KEY", "not-needed")
MODEL = os.getenv("LLAMA_CPP_MODEL", "local")
DEFAULT_SYSTEM_PROMPT = os.getenv("SYSTEM_PROMPT", "You are a helpful assistant.")

DEFAULT_SETTINGS = {
    "temperature": float(os.getenv("DEFAULT_TEMPERATURE", "0.7")),
    "top_p": float(os.getenv("DEFAULT_TOP_P", "0.95")),
    "top_k": int(os.getenv("DEFAULT_TOP_K", "40")),
    "min_p": float(os.getenv("DEFAULT_MIN_P", "0.05")),
    "repeat_penalty": float(os.getenv("DEFAULT_REPEAT_PENALTY", "1.05")),
    "max_tokens": int(os.getenv("DEFAULT_MAX_TOKENS", "512")),
    "system_prompt": DEFAULT_SYSTEM_PROMPT,
}

client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=600.0,
    max_retries=0,
)


# -----------------------------
# Session storage
# -----------------------------

SESSIONS_DIR = Path(os.getenv("CHAT_SESSIONS_DIR", "chats"))


def ensure_sessions_dir():
    SESSIONS_DIR.mkdir(parents=True, exist_ok=True)


ensure_sessions_dir()


def now_iso():
    return datetime.now(timezone.utc).isoformat()


def valid_session_id(session_id):
    if session_id is None:
        return False
    return bool(re.fullmatch(r"[A-Za-z0-9_-]+", str(session_id)))


def normalize_message(message):
    if isinstance(message, dict):
        role = message.get("role", "user")
        content = message.get("content", "")
    else:
        role = getattr(message, "role", "user")
        content = getattr(message, "content", "")

    role = str(role).lower()

    if content is None:
        content = ""
    elif not isinstance(content, str):
        content = str(content)

    return {"role": role, "content": content}


def to_conversation_messages(messages, include_empty_assistant=False):
    out = []

    for message in messages or []:
        msg = normalize_message(message)

        if msg["role"] not in ("user", "assistant"):
            continue

        if (
            not include_empty_assistant
            and msg["role"] == "assistant"
            and not msg["content"].strip()
        ):
            continue

        out.append(msg)

    return out


def has_messages(messages):
    return bool(to_conversation_messages(messages))


def generate_title(messages, fallback="New Chat"):
    for msg in messages:
        if msg.get("role") != "user":
            continue

        text = " ".join((msg.get("content") or "").split())
        if not text:
            continue

        if len(text) <= 60:
            return text
        return text[:59] + "…"

    return fallback


def list_sessions():
    """
    Returns dropdown choices as (title, session_id), sorted newest first.
    """
    ensure_sessions_dir()

    entries = []

    for path in SESSIONS_DIR.glob("*.json"):
        try:
            data = json.loads(path.read_text(encoding="utf-8"))
        except Exception:
            continue

        if not isinstance(data, dict):
            continue

        session_id = data.get("id") or path.stem
        if not valid_session_id(session_id):
            continue

        messages = to_conversation_messages(data.get("messages", []))
        if not messages:
            continue

        title = data.get("title") or generate_title(messages)
        updated_at = data.get("updated_at") or data.get("created_at") or ""

        entries.append((str(updated_at), str(title), str(session_id)))

    entries.sort(key=lambda item: item[0], reverse=True)

    return [(title, session_id) for _, title, session_id in entries]


def load_session(session_id):
    if not valid_session_id(session_id):
        return []

    selected = str(session_id)
    path = SESSIONS_DIR / f"{selected}.json"

    if not path.exists():
        return []

    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except Exception:
        return []

    if not isinstance(data, dict):
        return []

    return to_conversation_messages(data.get("messages", []))


def save_session(session_id, messages):
    """
    Save a session to disk.

    Session JSON schema:
        {
          "id": "...",
          "title": "...",
          "created_at": "...",
          "updated_at": "...",
          "messages": [{"role": "user", "content": "..."}]
        }
    """
    ensure_sessions_dir()

    if valid_session_id(session_id):
        session_id = str(session_id)
    else:
        session_id = uuid.uuid4().hex

    messages_to_save = to_conversation_messages(messages)
    path = SESSIONS_DIR / f"{session_id}.json"

    # Do not keep empty sessions around.
    if not messages_to_save:
        if path.exists():
            try:
                path.unlink(missing_ok=True)
            except OSError:
                pass
        return None

    created_at = now_iso()
    title = None

    if path.exists():
        try:
            existing = json.loads(path.read_text(encoding="utf-8"))
            if isinstance(existing, dict):
                created_at = existing.get("created_at") or created_at
                title = existing.get("title")
        except Exception:
            pass

    # Re-title untitled sessions automatically.
    if not title or title in ("", "New Chat"):
        title = generate_title(messages_to_save)

    payload = {
        "id": session_id,
        "title": title,
        "created_at": created_at,
        "updated_at": now_iso(),
        "messages": messages_to_save,
    }

    # Atomic-ish write.
    tmp_path = path.with_suffix(".tmp")
    tmp_path.write_text(
        json.dumps(payload, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )
    os.replace(tmp_path, path)

    return session_id


# -----------------------------
# Helper for streaming chunks
# -----------------------------

def _attr(obj, key, default=None):
    if isinstance(obj, dict):
        return obj.get(key, default)
    return getattr(obj, key, default)


def output_tuple(
    history,
    textbox_value,
    textbox_interactive,
    session_id,
    dropdown_value,
    dropdown_interactive,
    button_interactive,
    dropdown_choices=None,
):
    """
    Helper for stream_reply() outputs.

    Output order must match:
        [chatbot, textbox, session_id, dropdown,
         send_button, new_button, delete_button, refresh_button]
    """
    return (
        history,
        gr.update(value=textbox_value, interactive=textbox_interactive),
        session_id,
        gr.update(
            choices=dropdown_choices if dropdown_choices is not None else list_sessions(),
            value=dropdown_value,
            interactive=dropdown_interactive,
        ),
        gr.update(interactive=button_interactive),
        gr.update(interactive=button_interactive),
        gr.update(interactive=button_interactive),
        gr.update(interactive=button_interactive),
    )


# -----------------------------
# Gradio callbacks
# -----------------------------

def stream_reply(
    user_text,
    chat_messages,
    session_id,
    temperature,
    top_p,
    top_k,
    min_p,
    repeat_penalty,
    max_tokens,
    system_prompt,
):
    if valid_session_id(session_id):
        session_id = str(session_id)
    else:
        session_id = None

    user_text = (user_text or "").strip()

    # If the user clicks Send with no text, just clean up the textbox.
    if not user_text:
        yield output_tuple(
            chat_messages or [],
            "",
            True,
            session_id,
            session_id,
            True,
            True,
        )
        return

    # Normalize slider values before using them.
    try:
        temperature = (
            DEFAULT_SETTINGS["temperature"] if temperature is None else float(temperature)
        )
        top_p = DEFAULT_SETTINGS["top_p"] if top_p is None else float(top_p)
        top_k = DEFAULT_SETTINGS["top_k"] if top_k is None else int(top_k)
        min_p = DEFAULT_SETTINGS["min_p"] if min_p is None else float(min_p)
        repeat_penalty = (
            DEFAULT_SETTINGS["repeat_penalty"]
            if repeat_penalty is None
            else float(repeat_penalty)
        )
        max_tokens = (
            DEFAULT_SETTINGS["max_tokens"] if max_tokens is None else int(max_tokens)
        )
    except Exception:
        # Do not start a request with invalid settings.
        yield output_tuple(
            chat_messages or [],
            "",
            True,
            session_id,
            session_id,
            True,
            True,
        )
        return

    history = to_conversation_messages(chat_messages)
    history.append({"role": "user", "content": user_text})

    # Build the request message list: system prompt + full conversation so far.
    request_messages = []

    system_text = (system_prompt or "").strip()
    if system_text:
        request_messages.append({"role": "system", "content": system_text})

    request_messages.extend(history)

    # Save the user message before generating.
    try:
        saved = save_session(session_id, history)
        if saved:
            session_id = saved
    except Exception as exc:
        history.append({"role": "assistant", "content": f"Could not save chat: {exc}"})
        yield output_tuple(
            history,
            "",
            True,
            session_id,
            session_id,
            True,
            True,
        )
        return

    # Assistant placeholder that will be filled token-by-token.
    history.append({"role": "assistant", "content": ""})

    # Refresh dropdown immediately so the new session is visible.
    choices = list_sessions()

    # Disable UI while generating.
    yield output_tuple(
        history,
        "",
        False,
        session_id,
        session_id,
        False,
        False,
        dropdown_choices=choices,
    )

    try:
        stream = client.chat.completions.create(
            model=MODEL,
            messages=request_messages,
            stream=True,
            temperature=temperature,
            top_p=top_p,
            max_tokens=max_tokens,
            # Llama.cpp-specific parameters that are not in the standard
            # OpenAI request schema are merged into the top-level JSON body
            # by the OpenAI Python client when passed through extra_body.
            extra_body={
                "top_k": top_k,
                "min_p": min_p,
                "repeat_penalty": repeat_penalty,
            },
        )

        for chunk in stream:
            choices_from_chunk = _attr(chunk, "choices", None)

            if not choices_from_chunk:
                continue

            delta = _attr(choices_from_chunk[0], "delta", None)
            token = _attr(delta, "content", None)

            if token:
                history[-1]["content"] += token if isinstance(token, str) else str(token)

                yield output_tuple(
                    history,
                    "",
                    False,
                    session_id,
                    session_id,
                    False,
                    False,
                    dropdown_choices=choices,
                )

    except Exception as exc:
        if history and history[-1].get("role") == "assistant":
            history[-1]["content"] = (
                (history[-1]["content"] or "") + f"\n\nRequest failed: {exc}"
            ).strip()
        else:
            history.append({"role": "assistant", "content": f"Request failed: {exc}"})

        try:
            saved = save_session(session_id, history)
            if saved:
                session_id = saved
        except Exception:
            pass

        yield output_tuple(
            history,
            "",
            True,
            session_id,
            session_id,
            True,
            True,
        )
        return

    # Some models/servers may finish without returning visible text.
    if (
        history
        and history[-1].get("role") == "assistant"
        and not history[-1]["content"].strip()
    ):
        history[-1]["content"] = "(No content received.)"

    # Save final assistant message.
    try:
        saved = save_session(session_id, history)
        if saved:
            session_id = saved
    except Exception:
        pass

    yield output_tuple(
        history,
        "",
        True,
        session_id,
        session_id,
        True,
        True,
    )


def load_selected(selected, state_session_id, current_messages):
    """
    Dropdown selection handler.

    Inputs:
        selected: selected session id
        state_session_id: current session id
        current_messages: current chatbot messages

    Outputs:
        chatbot
        session_id state
        textbox
    """
    if not valid_session_id(selected):
        return [], None, gr.update(value="")

    selected = str(selected)
    current_id = str(state_session_id) if valid_session_id(state_session_id) else None

    # If the user re-selects the current session, preserve the visible chat.
    if selected == current_id and current_messages:
        return current_messages, selected, gr.update(value="")

    loaded = load_session(selected)
    return loaded, selected, gr.update(value="")


def new_chat(chat_messages, session_id):
    """
    Save the current session if it has content, then clear the UI.

    Outputs:
        chatbot
        session_id
        textbox
        dropdown
    """
    current_id = str(session_id) if valid_session_id(session_id) else None

    if current_id and has_messages(chat_messages):
        save_session(current_id, chat_messages)

    return (
        [],
        None,
        gr.update(value="", interactive=True),
        gr.update(choices=list_sessions(), value=None, interactive=True),
    )


def delete_chat(session_id):
    """
    Delete the current session file and clear the UI.

    Outputs:
        chatbot
        session_id
        textbox
        dropdown
    """
    ensure_sessions_dir()

    if valid_session_id(session_id):
        path = SESSIONS_DIR / f"{str(session_id)}.json"
        try:
            path.unlink(missing_ok=True)
        except OSError:
            pass

    return (
        [],
        None,
        gr.update(value="", interactive=True),
        gr.update(choices=list_sessions(), value=None, interactive=True),
    )


def refresh_dropdown(current_value):
    """
    Refresh dropdown choices while trying to keep the current selection.
    """
    choices = list_sessions()
    values = {value for _label, value in choices}

    current_value = str(current_value) if valid_session_id(current_value) else None
    new_value = current_value if current_value in values else None

    return gr.update(
        choices=choices,
        value=new_value,
        interactive=True,
    )


# -----------------------------
# UI
# -----------------------------

with gr.Blocks(title="llama.cpp Local Chat") as demo:
    session_id = gr.State(value=None)

    gr.Markdown(
        f"""
        ## llama.cpp local chat

        Connecting to `{BASE_URL}` with an OpenAI-compatible client.

        Sessions are saved as JSON files under `{SESSIONS_DIR.resolve()}`.
        """
    )

    with gr.Accordion("Generation settings", open=False):
        gr.Markdown(
            """
            These values are sent on each request.

            Standard OpenAI parameters:
            - `temperature`
            - `top_p`
            - `max_tokens`

            Llama.cpp-specific parameters sent through `extra_body`:
            - `top_k`
            - `min_p`
            - `repeat_penalty`
            """
        )

        with gr.Row():
            temp = gr.Slider(
                minimum=0.0,
                maximum=2.0,
                value=DEFAULT_SETTINGS["temperature"],
                step=0.01,
                label="temperature",
            )
            top_p = gr.Slider(
                minimum=0.0,
                maximum=1.0,
                value=DEFAULT_SETTINGS["top_p"],
                step=0.01,
                label="top_p",
            )

        with gr.Row():
            top_k = gr.Slider(
                minimum=0,
                maximum=200,
                value=DEFAULT_SETTINGS["top_k"],
                step=1,
                label="top_k",
            )
            min_p = gr.Slider(
                minimum=0.0,
                maximum=1.0,
                value=DEFAULT_SETTINGS["min_p"],
                step=0.05,
                label="min_p",
            )

        with gr.Row():
            repeat_penalty = gr.Slider(
                minimum=0.0,
                maximum=2.0,
                value=DEFAULT_SETTINGS["repeat_penalty"],
                step=0.01,
                label="repeat_penalty",
            )
            max_tokens = gr.Slider(
                minimum=1,
                maximum=8192,
                value=DEFAULT_SETTINGS["max_tokens"],
                step=1,
                label="max_tokens",
            )

        system_prompt_box = gr.Textbox(
            label="System prompt",
            value=DEFAULT_SETTINGS["system_prompt"],
            lines=3,
            placeholder="You are a helpful assistant.",
        )

    with gr.Row():
        with gr.Column(scale=1, min_width=280):
            gr.Markdown("### Sessions")

            dropdown = gr.Dropdown(
                choices=list_sessions(),
                label="Saved chats",
                value=None,
                interactive=True,
                allow_custom_value=False,
            )

            with gr.Row():
                new_button = gr.Button("New Chat", variant="primary")
                delete_button = gr.Button("Delete", variant="stop")

            refresh_button = gr.Button("Refresh")

        with gr.Column(scale=4):
            chatbot = gr.Chatbot(
                label="Conversation",
                type="messages",
                height=620,
            )

            with gr.Row():
                textbox = gr.Textbox(
                    placeholder="Type a message and press Enter...",
                    show_label=False,
                    scale=10,
                )
                send_button = gr.Button("Send", variant="primary", scale=1)

    # Streaming chat inputs.
    chat_inputs = [
        textbox,
        chatbot,
        session_id,
        temp,
        top_p,
        top_k,
        min_p,
        repeat_penalty,
        max_tokens,
        system_prompt_box,
    ]

    # Streaming chat outputs.
    chat_outputs = [
        chatbot,
        textbox,
        session_id,
        dropdown,
        send_button,
        new_button,
        delete_button,
        refresh_button,
    ]

    send_button.click(
        stream_reply,
        inputs=chat_inputs,
        outputs=chat_outputs,
    )

    textbox.submit(
        stream_reply,
        inputs=chat_inputs,
        outputs=chat_outputs,
    )

    # Session management.
    # Use .select() to avoid triggering load on programmatic dropdown updates.
    dropdown.select(
        load_selected,
        inputs=[dropdown, session_id, chatbot],
        outputs=[chatbot, session_id, textbox],
    )

    new_button.click(
        new_chat,
        inputs=[chatbot, session_id],
        outputs=[chatbot, session_id, textbox, dropdown],
    )

    delete_button.click(
        delete_chat,
        inputs=[session_id],
        outputs=[chatbot, session_id, textbox, dropdown],
    )

    refresh_button.click(
        refresh_dropdown,
        inputs=[dropdown],
        outputs=[dropdown],
    )

    # Useful if another tab/process created sessions.
    demo.load(
        refresh_dropdown,
        inputs=[dropdown],
        outputs=[dropdown],
    )


if __name__ == "__main__":
    demo.queue().launch(server_name="127.0.0.1", server_port=7860)
```

## 3. Start llama.cpp

Make sure llama.cpp is running on:

```bash
http://localhost:8001/v1
```

Example:

```bash
llama-server -m /path/to/model.gguf --host 127.0.0.1 --port 8001
```

Check available models if needed:

```bash
curl http://localhost:8001/v1/models
```

If llama.cpp requires a specific model name, set it:

```bash
LLAMA_CPP_MODEL=model-name-here uv run python app.py
```

## Notes

The generated request will include parameters like this conceptually:

```json
{
  "model": "local",
  "messages": [...],
  "stream": true,
  "temperature": 0.7,
  "top_p": 0.95,
  "max_tokens": 512,
  "top_k": 40,
  "min_p": 0.05,
  "repeat_penalty": 1.05
}
```

`extra_body` does not appear literally in the final HTTP JSON body. The OpenAI Python client merges its contents into the top-level request body, which is what llama.cpp sees.
