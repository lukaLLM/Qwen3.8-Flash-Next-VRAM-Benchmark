# [2] chat history

Here is a complete updated single-file app that adds persistent JSON chat history, a sidebar/dropdown to load sessions, a **New Chat** button, and a **Delete** button. Each conversation is stored on disk with:

- `title`
- `created_at`
- `updated_at`
- `messages`
- optional `model`

Sessions are saved in a local folder named `chat_history/`.

---

## 1. Create a separate project/venv with `uv`

If you already created the previous project, you can skip to step 2 and replace `app.py`.

```bash
uv init llama-chat --python 3.12
cd llama-chat
uv add "gradio>=5" "openai>=1"
```

`uv` will create and manage the project virtual environment, usually in `.venv/`.

## 2. Replace `app.py` with this

```python
from __future__ import annotations

import json
import os
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import gradio as gr
from openai import OpenAI


# ----------------------------------------------------------------------------
# llama.cpp server configuration
# ----------------------------------------------------------------------------
BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("OPENAI_API_KEY", "EMPTY")
MODEL_ENV = os.getenv("LLAMA_CPP_MODEL", "")

# Folder where chat sessions are saved as JSON files.
CHAT_HISTORY_DIR = Path(os.getenv("CHAT_HISTORY_DIR", "chat_history"))


client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=300,
)

# Shorter timeout client used only for trying to discover a model name.
probe_client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=5,
)


def detect_default_model() -> str:
    """
    Try to read a model name from llama.cpp's /v1/models endpoint.

    llama.cpp's OpenAI-compatible server usually serves the loaded model id
    there. If the server is not running, fall back to a generic name.
    """
    if MODEL_ENV:
        return MODEL_ENV

    try:
        response = probe_client.models.list()
        if getattr(response, "data", None):
            model_id = getattr(response.data[0], "id", None)
            if model_id:
                return str(model_id)
    except Exception:
        pass

    return "llama.cpp"


DEFAULT_MODEL = detect_default_model()


# ----------------------------------------------------------------------------
# Session storage helpers
# ----------------------------------------------------------------------------
def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def ensure_history_dir() -> None:
    CHAT_HISTORY_DIR.mkdir(parents=True, exist_ok=True)


def make_session_id() -> str:
    """
    Generate a filesystem-friendly id that is also roughly chronologically
    sortable.
    """
    timestamp = datetime.now(timezone.utc).strftime("%Y%m%d-%H%M%S-%f")
    return f"{timestamp}-{uuid.uuid4().hex[:8]}"


def session_path(session_id: str) -> Path:
    """
    Convert a session id to a JSON file path.

    The id is sanitized so that odd characters cannot escape the history dir.
    """
    safe = "".join(ch for ch in session_id if ch.isalnum() or ch in "-_.")
    if not safe:
        safe = uuid.uuid4().hex
    return CHAT_HISTORY_DIR / f"{safe}.json"


def content_to_text(content: Any) -> str:
    """
    Convert common Gradio/OpenAI message content shapes to plain text.

    This keeps the example simple and text-only. Multimodal support can be
    added here later if needed.
    """
    if isinstance(content, str):
        return content

    if isinstance(content, list):
        parts = []
        for part in content:
            if isinstance(part, str):
                parts.append(part)
            elif isinstance(part, dict) and part.get("type") == "text":
                parts.append(str(part.get("text", "")))
        return "\n".join(parts)

    return str(content)


def make_title_from_messages(messages: list[dict[str, Any]]) -> str:
    """
    Use the first non-empty user message as the session title.
    """
    for message in messages:
        if not isinstance(message, dict):
            continue

        if message.get("role") != "user":
            continue

        text = content_to_text(message.get("content", "")).strip()
        if not text:
            continue

        first_line = text.splitlines()[0].strip()
        if len(first_line) > 80:
            return first_line[:79] + "…"
        return first_line

    return "New chat"


def safe_session_data(data: Any, fallback_id: str | None = None) -> dict[str, Any]:
    """
    Normalize loaded JSON so the rest of the app can rely on required fields.
    """
    if not isinstance(data, dict):
        data = {}

    messages = data.get("messages")
    if not isinstance(messages, list):
        messages = []

    title = data.get("title")
    if not isinstance(title, str) or not title.strip():
        title = make_title_from_messages(messages)

    created_at = data.get("created_at")
    if not isinstance(created_at, str) or not created_at.strip():
        created_at = now_iso()

    updated_at = data.get("updated_at")
    if not isinstance(updated_at, str) or not updated_at.strip():
        updated_at = now_iso()

    model = data.get("model")
    if not isinstance(model, str) or not model.strip():
        model = DEFAULT_MODEL

    session_id = data.get("id")
    if not isinstance(session_id, str) or not session_id.strip():
        session_id = fallback_id or make_session_id()

    return {
        "id": session_id,
        "title": title,
        "created_at": created_at,
        "updated_at": updated_at,
        "model": model,
        "messages": messages,
    }


def load_session(session_id: str | None) -> dict[str, Any] | None:
    """
    Load one session JSON file.
    """
    if not session_id:
        return None

    ensure_history_dir()
    path = session_path(session_id)

    if not path.exists():
        return None

    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)
        return safe_session_data(data, fallback_id=session_id)
    except Exception:
        return None


def save_session(
    session_id: str | None,
    messages: list[dict[str, Any]],
    model: str | None = None,
) -> str:
    """
    Save or update one session JSON file.

    A new id is created if session_id is None.
    """
    ensure_history_dir()

    messages = [m for m in (messages or []) if isinstance(m, dict)]
    session_id = session_id or make_session_id()

    existing = load_session(session_id)
    if existing and existing.get("id"):
        session_id = str(existing["id"])

    path = session_path(session_id)

    data = {
        "id": session_id,
        "title": make_title_from_messages(messages),
        "created_at": existing.get("created_at") if existing else now_iso(),
        "updated_at": now_iso(),
        "model": (model or "").strip() or (existing.get("model") if existing else DEFAULT_MODEL),
        "messages": messages,
    }

    tmp_path = path.with_name(path.name + ".tmp")
    with tmp_path.open("w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2, default=str)

    tmp_path.replace(path)
    return session_id


def list_session_metadata() -> list[dict[str, Any]]:
    """
    Load all session files and sort newest first.
    """
    ensure_history_dir()

    sessions: list[dict[str, Any]] = []

    for path in CHAT_HISTORY_DIR.glob("*.json"):
        session = load_session(path.stem)
        if session:
            sessions.append(session)

    sessions.sort(key=lambda s: s.get("updated_at", ""), reverse=True)
    return sessions


def humanize_timestamp(timestamp: str) -> str:
    try:
        dt = datetime.fromisoformat(timestamp)
        return dt.strftime("%Y-%m-%d %H:%M")
    except Exception:
        return timestamp if isinstance(timestamp, str) else ""


def session_choices() -> list[tuple[str, str]]:
    """
    Return dropdown choices: label, id.
    """
    choices: list[tuple[str, str]] = []

    for session in list_session_metadata():
        session_id = str(session.get("id", ""))
        if not session_id:
            continue

        title = str(session.get("title") or "New chat")
        timestamp = humanize_timestamp(str(session.get("updated_at", "")))
        label = f"{title} — {timestamp}" if timestamp else title

        choices.append((label, session_id))

    return choices


def dropdown_update(selected: str | None) -> dict:
    """
    Build a Gradio dropdown update that preserves the selected session if valid.
    """
    choices = session_choices()
    valid = {value for _, value in choices}

    if selected not in valid:
        selected = None

    return gr.update(choices=choices, value=selected)


# ----------------------------------------------------------------------------
# Chat logic
# ----------------------------------------------------------------------------
def build_api_messages(messages: list[dict[str, Any]]) -> list[dict[str, str]]:
    """
    Convert Gradio message history into OpenAI-compatible messages.

    The full conversation is sent on every request.
    """
    api_messages: list[dict[str, str]] = []

    for message in messages:
        if not isinstance(message, dict):
            continue

        role = message.get("role")
        if role not in {"system", "user", "assistant"}:
            continue

        content = content_to_text(message.get("content", "")).strip()
        if not content:
            continue

        api_messages.append(
            {
                "role": role,
                "content": content,
            }
        )

    return api_messages


def respond(
    message: str,
    history: list[dict[str, Any]],
    model_name: str,
    session_id: str | None,
):
    """
    Generator handler for sending a message and streaming tokens.
    """
    if not isinstance(message, str) or not message.strip():
        yield gr.update(value=""), history, session_id, gr.skip()
        return

    user_text = message.strip()
    history = list(history or [])

    # Append the user's message and save it before making the API request.
    history.append({"role": "user", "content": user_text})
    session_id = save_session(session_id, history, model_name)

    # Clear input, show user message, refresh dropdown if this is a new chat.
    yield gr.update(value=""), history, session_id, dropdown_update(session_id)

    # Add assistant placeholder that will be filled by streamed tokens.
    history.append({"role": "assistant", "content": ""})
    yield gr.skip(), history, session_id, gr.skip()

    # Exclude the empty assistant placeholder from the API request.
    api_messages = build_api_messages(history[:-1])
    model = (model_name or "").strip() or DEFAULT_MODEL

    assistant_text = ""
    last_saved_len = 0

    try:
        if not api_messages:
            raise RuntimeError("No messages to send.")

        stream = client.chat.completions.create(
            model=model,
            messages=api_messages,
            stream=True,
        )

        for chunk in stream:
            choices = getattr(chunk, "choices", None)
            if not choices:
                continue

            delta = getattr(choices[0], "delta", None)
            piece = getattr(delta, "content", None)

            if isinstance(piece, str) and piece:
                assistant_text += piece
                history[-1] = {"role": "assistant", "content": assistant_text}

                # Periodically save during long streams so partial answers
                # survive crashes or browser reloads.
                if len(assistant_text) - last_saved_len >= 512:
                    save_session(session_id, history, model)
                    last_saved_len = len(assistant_text)

                yield gr.skip(), history, session_id, gr.skip()

        if not assistant_text:
            history[-1] = {"role": "assistant", "content": "(empty response)"}

        save_session(session_id, history, model)
        yield gr.skip(), history, session_id, dropdown_update(session_id)

    except Exception as exc:
        prefix = assistant_text + ("\n\n" if assistant_text else "")
        history[-1] = {
            "role": "assistant",
            "content": f"{prefix}Request failed: {exc}",
        }
        save_session(session_id, history, model)
        yield gr.skip(), history, session_id, dropdown_update(session_id)


# ----------------------------------------------------------------------------
# Session UI actions
# ----------------------------------------------------------------------------
def initialize():
    """
    Load the most recent chat when the app starts, if any exist.
    """
    sessions = list_session_metadata()

    if not sessions:
        return [], None, DEFAULT_MODEL, dropdown_update(None)

    first = sessions[0]
    return (
        first.get("messages", []),
        first.get("id"),
        first.get("model", DEFAULT_MODEL),
        dropdown_update(first.get("id")),
    )


def load_selected(
    selected_id: str | None,
    current_id: str | None,
    history: list[dict[str, Any]],
    model: str,
):
    """
    Load messages from the selected dropdown session.
    """
    data = load_session(selected_id)

    if not data:
        # If the selected id disappeared, don't destroy the current UI.
        if current_id and current_id == selected_id:
            return [], None, model
        return history, current_id, model

    return data.get("messages", []), data.get("id"), data.get("model", DEFAULT_MODEL)


def new_chat():
    """
    Start a new empty chat. No file is written until the first message is sent.
    """
    return (
        gr.update(value=""),
        [],
        None,
        DEFAULT_MODEL,
        dropdown_update(None),
    )


def delete_current(
    selected_id: str | None,
    current_id: str | None,
    history: list[dict[str, Any]],
    model: str,
):
    """
    Delete the selected session.

    If the selected session is also the current one, the UI starts a new chat.
    """
    target = selected_id or current_id

    if not target:
        return history, current_id, model, dropdown_update(None)

    path = session_path(target)
    try:
        path.unlink()
    except FileNotFoundError:
        pass
    except Exception:
        pass

    if current_id == target:
        current_id = None
        history = []
        model = DEFAULT_MODEL

    return history, current_id, model, dropdown_update(current_id)


# ----------------------------------------------------------------------------
# UI
# ----------------------------------------------------------------------------
with gr.Blocks(title="llama.cpp Gradio Chat") as demo:
    current_session_id = gr.State(value=None)

    # Sidebar layout if available; otherwise fall back to a normal column.
    if hasattr(gr, "Sidebar"):
        sidebar_context = gr.Sidebar(label="Conversations")
    else:
        sidebar_context = gr.Column(scale=1, min_width=320)

    with sidebar_context:
        gr.Markdown("### Chats")

        dropdown = gr.Dropdown(
            label="Select a session",
            choices=[],
            value=None,
            placeholder="No saved chats",
        )

        with gr.Row():
            new_button = gr.Button("New Chat", variant="primary")
            delete_button = gr.Button("Delete Chat", variant="stop")

    # Main chat UI.
    gr.Markdown(
        f"""
        ## Gradio ↔ llama.cpp

        Connects to `{BASE_URL}` using the OpenAI-compatible API.

        The full message list is sent on every request.
        """
    )

    model_box = gr.Textbox(
        label="Model",
        value=DEFAULT_MODEL,
        info="OpenAI-compatible model id served by llama.cpp.",
    )

    chatbot = gr.Chatbot(
        type="messages",
        height=520,
        label="",
    )

    with gr.Row():
        msg = gr.Textbox(
            label="Message",
            placeholder="Type a message and press Enter...",
            scale=10,
            lines=1,
        )
        send_button = gr.Button("Send", variant="primary", scale=1)
        stop_button = gr.Button("Stop", variant="stop", scale=1)

    # Events.
    enter_event = msg.submit(
        respond,
        inputs=[msg, chatbot, model_box, current_session_id],
        outputs=[msg, chatbot, current_session_id, dropdown],
    )

    send_event = send_button.click(
        respond,
        inputs=[msg, chatbot, model_box, current_session_id],
        outputs=[msg, chatbot, current_session_id, dropdown],
    )

    new_event = new_button.click(
        new_chat,
        outputs=[msg, chatbot, current_session_id, model_box, dropdown],
    )

    delete_event = delete_button.click(
        delete_current,
        inputs=[dropdown, current_session_id, chatbot, model_box],
        outputs=[chatbot, current_session_id, model_box, dropdown],
    )

    dropdown.select(
        load_selected,
        inputs=[dropdown, current_session_id, chatbot, model_box],
        outputs=[chatbot, current_session_id, model_box],
    )

    stop_button.click(
        lambda: None,
        None,
        None,
        cancels=[send_event, enter_event],
        show_progress="hidden",
    )

    demo.load(
        initialize,
        outputs=[chatbot, current_session_id, model_box, dropdown],
    )


if __name__ == "__main__":
    demo.queue().launch(show_api=False)
```

## 5. Example saved session JSON

Files are saved in `chat_history/` like:

```json
{
  "id": "20260127-123456-123456-abcd1234",
  "title": "Hello",
  "created_at": "2026-01-27T12:34:56.123456+00:00",
  "updated_at": "2026-01-27T12:34:59.987654+00:00",
  "model": "llama.cpp",
  "messages": [
    {
      "role": "user",
      "content": "Hello"
    },
    {
      "role": "assistant",
      "content": "Hi! How can I help?"
    }
  ]
}
```
