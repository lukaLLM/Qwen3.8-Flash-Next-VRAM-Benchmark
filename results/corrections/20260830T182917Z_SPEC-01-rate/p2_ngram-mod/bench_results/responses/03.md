# [3] settings panel

Replace `app.py` with the version below.

It keeps the same core requirements:

- `gr.Blocks`
- `gr.Chatbot(type="messages")`
- `openai` client pointed at `http://localhost:8001/v1`
- streaming responses
- full chat history sent each request
- persistent JSON session history
- load/new/delete sessions

It also adds a settings panel with:

- system prompt
- `temperature`
- `top_p`
- `top_k`
- `min_p`
- `repeat_penalty`
- `max_tokens`

The llama.cpp-specific parameters that are not part of the standard OpenAI chat completion schema are sent using the OpenAI client’s `extra_body` option. In practice, `extra_body` is flattened into the top-level JSON request body, so the server receives:

```json
{
  "model": "...",
  "messages": [...],
  "temperature": 0.8,
  "top_p": 0.95,
  "max_tokens": 2048,
  "top_k": 40,
  "min_p": 0,
  "repeat_penalty": 1.1
}
```

where `top_k`, `min_p`, and `repeat_penalty` came from `extra_body`.

from __future__ import annotations

import json
import os
import uuid
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import gradio as gr
from openai import OpenAI


# ---------------------------------------------------------------------------
# llama.cpp server configuration
# ---------------------------------------------------------------------------
BASE_URL = os.getenv("LLAMA_CPP_BASE_URL", "http://localhost:8001/v1")
API_KEY = os.getenv("OPENAI_API_KEY", "EMPTY")
MODEL_ENV = os.getenv("LLAMA_CPP_MODEL", "")

# Where JSON chat sessions are stored.
CHAT_HISTORY_DIR = Path(os.getenv("CHAT_HISTORY_DIR", "chat_history"))


client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=300,
)

# Short timeout client used only for model discovery.
probe_client = OpenAI(
    base_url=BASE_URL,
    api_key=API_KEY,
    timeout=5,
)


def detect_default_model() -> str:
    """
    Try to read the model name from llama.cpp's /v1/models endpoint.

    If the server is not running or the endpoint is unavailable, fall back
    to a generic default.
    """
    if MODEL_ENV:
        return MODEL_ENV

    try:
        response = probe_client.models.list()
        if getattr(response, "data", None):
            model_id = getattr(response.data[0], "id", None)
            if model_id:
                return str(model_id)
    except Exception:
        pass

    return "llama.cpp"


DEFAULT_MODEL = detect_default_model()


# ---------------------------------------------------------------------------
# Generation settings
# ---------------------------------------------------------------------------
DEFAULT_SETTINGS = {
    "system_prompt": "You are a helpful assistant.",
    "temperature": 0.8,
    "top_p": 0.95,
    "top_k": 40,
    "min_p": 0.0,
    "repeat_penalty": 1.1,
    "max_tokens": 2048,
}

SETTINGS_KEYS = tuple(DEFAULT_SETTINGS.keys())


def clamp_number(
    value: Any,
    minimum: float,
    maximum: float,
    default: float,
    integer: bool = False,
) -> Any:
    """
    Convert a value to float, clamp it to range, and optionally return int.
    """
    try:
        x = float(value)
    except (TypeError, ValueError):
        x = float(default)

    if x < minimum:
        x = float(minimum)
    elif x > maximum:
        x = float(maximum)

    if integer:
        return int(round(x))

    return float(x)


def normalize_settings(settings: Any = None) -> dict[str, Any]:
    """
    Ensure settings values exist and are within safe ranges.
    """
    if not isinstance(settings, dict):
        settings = {}

    return {
        "system_prompt": str(
            settings.get("system_prompt", DEFAULT_SETTINGS["system_prompt"]) or ""
        ),
        "temperature": clamp_number(
            settings.get("temperature", DEFAULT_SETTINGS["temperature"]),
            0.0,
            2.0,
            DEFAULT_SETTINGS["temperature"],
        ),
        "top_p": clamp_number(
            settings.get("top_p", DEFAULT_SETTINGS["top_p"]),
            0.0,
            1.0,
            DEFAULT_SETTINGS["top_p"],
        ),
        "top_k": clamp_number(
            settings.get("top_k", DEFAULT_SETTINGS["top_k"]),
            1.0,
            200.0,
            DEFAULT_SETTINGS["top_k"],
            integer=True,
        ),
        "min_p": clamp_number(
            settings.get("min_p", DEFAULT_SETTINGS["min_p"]),
            0.0,
            1.0,
            DEFAULT_SETTINGS["min_p"],
        ),
        "repeat_penalty": clamp_number(
            settings.get("repeat_penalty", DEFAULT_SETTINGS["repeat_penalty"]),
            1.0,
            2.0,
            DEFAULT_SETTINGS["repeat_penalty"],
        ),
        "max_tokens": clamp_number(
            settings.get("max_tokens", DEFAULT_SETTINGS["max_tokens"]),
            1.0,
            8192.0,
            DEFAULT_SETTINGS["max_tokens"],
            integer=True,
        ),
    }


def settings_tuple(settings: Any = None) -> tuple[Any, ...]:
    """
    Return settings in the order expected by the Gradio controls.
    """
    s = normalize_settings(settings)

    return (
        s["system_prompt"],
        s["temperature"],
        s["top_p"],
        s["top_k"],
        s["min_p"],
        s["repeat_penalty"],
        s["max_tokens"],
    )


# ---------------------------------------------------------------------------
# Session storage helpers
# ---------------------------------------------------------------------------
def now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def ensure_history_dir() -> None:
    CHAT_HISTORY_DIR.mkdir(parents=True, exist_ok=True)


def make_session_id() -> str:
    """
    Filesystem-friendly, roughly chronologically sortable session id.
    """
    timestamp = datetime.now(timezone.utc).strftime("%Y%m%d-%H%M%S-%f")
    return f"{timestamp}-{uuid.uuid4().hex[:8]}"


def session_path(session_id: str) -> Path:
    """
    Convert a session id into a JSON file path.

    The id is sanitized so odd characters cannot escape the history directory.
    """
    safe = "".join(ch for ch in str(session_id) if ch.isalnum() or ch in "-_.")
    if not safe:
        safe = uuid.uuid4().hex

    return CHAT_HISTORY_DIR / f"{safe}.json"


def content_to_text(content: Any) -> str:
    """
    Convert common Gradio/OpenAI message content shapes to plain text.
    """
    if isinstance(content, str):
        return content

    if isinstance(content, list):
        parts = []

        for part in content:
            if isinstance(part, str):
                parts.append(part)
            elif isinstance(part, dict) and part.get("type") == "text":
                parts.append(str(part.get("text", "")))

        return "\n".join(parts)

    return str(content)


def make_title_from_messages(messages: list[dict[str, Any]]) -> str:
    """
    Use the first non-empty user message as the session title.
    """
    for message in messages:
        if not isinstance(message, dict):
            continue

        if message.get("role") != "user":
            continue

        text = content_to_text(message.get("content", "")).strip()
        if not text:
            continue

        first_line = text.splitlines()[0].strip()

        if len(first_line) > 80:
            return first_line[:79] + "…"

        return first_line

    return "New chat"


def safe_session_data(data: Any, fallback_id: str | None = None) -> dict[str, Any]:
    """
    Normalize loaded JSON session data.
    """
    if not isinstance(data, dict):
        data = {}

    messages = data.get("messages")
    if not isinstance(messages, list):
        messages = []

    messages = [m for m in messages if isinstance(m, dict)]

    title = data.get("title")
    if not isinstance(title, str) or not title.strip():
        title = make_title_from_messages(messages)

    created_at = data.get("created_at")
    if not isinstance(created_at, str) or not created_at.strip():
        created_at = now_iso()

    updated_at = data.get("updated_at")
    if not isinstance(updated_at, str) or not updated_at.strip():
        updated_at = now_iso()

    model = data.get("model")
    if not isinstance(model, str) or not model.strip():
        model = DEFAULT_MODEL

    session_id = data.get("id")
    if not isinstance(session_id, str) or not session_id.strip():
        session_id = fallback_id or make_session_id()

    return {
        "id": session_id,
        "title": title,
        "created_at": created_at,
        "updated_at": updated_at,
        "model": model,
        "settings": normalize_settings(data.get("settings")),
        "messages": messages,
    }


def load_session(session_id: str | None) -> dict[str, Any] | None:
    """
    Load one session JSON file.
    """
    if not session_id:
        return None

    ensure_history_dir()
    path = session_path(session_id)

    if not path.exists():
        return None

    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)

        return safe_session_data(data, fallback_id=session_id)
    except Exception:
        return None


def save_session(
    session_id: str | None,
    messages: list[dict[str, Any]],
    model: str | None = None,
    settings: Any = None,
) -> str:
    """
    Save or update one session JSON file.

    If session_id is None, create a new session id.
    """
    ensure_history_dir()

    settings = normalize_settings(settings)
    messages = [m for m in (messages or []) if isinstance(m, dict)]
    session_id = session_id or make_session_id()

    existing = load_session(session_id)
    if existing and existing.get("id"):
        session_id = str(existing["id"])

    path = session_path(session_id)

    data = {
        "id": session_id,
        "title": make_title_from_messages(messages),
        "created_at": existing.get("created_at") if existing else now_iso(),
        "updated_at": now_iso(),
        "model": (model or "").strip()
        or (existing.get("model") if existing else DEFAULT_MODEL),
        "settings": settings,
        "messages": messages,
    }

    tmp_path = path.with_name(path.name + ".tmp")

    with tmp_path.open("w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2, default=str)

    tmp_path.replace(path)
    return session_id


def list_session_metadata() -> list[dict[str, Any]]:
    """
    Load all saved sessions and sort newest first.
    """
    ensure_history_dir()

    sessions: list[dict[str, Any]] = []

    for path in CHAT_HISTORY_DIR.glob("*.json"):
        session = load_session(path.stem)

        if session:
            sessions.append(session)

    sessions.sort(key=lambda s: s.get("updated_at", ""), reverse=True)
    return sessions


def humanize_timestamp(timestamp: str) -> str:
    """
    Render ISO timestamps more compactly for the dropdown.
    """
    try:
        dt = datetime.fromisoformat(timestamp)
        return dt.strftime("%Y-%m-%d %H:%M")
    except Exception:
        return timestamp if isinstance(timestamp, str) else ""


def session_choices() -> list[tuple[str, str]]:
    """
    Return dropdown choices as (label, session_id).
    """
    choices: list[tuple[str, str]] = []

    for session in list_session_metadata():
        session_id = str(session.get("id", ""))

        if not session_id:
            continue

        title = str(session.get("title") or "New chat")
        timestamp = humanize_timestamp(str(session.get("updated_at", "")))
        label = f"{title} — {timestamp}" if timestamp else title

        choices.append((label, session_id))

    return choices


def dropdown_update(selected: str | None) -> dict:
    """
    Build a Gradio dropdown update while preserving valid selection.
    """
    choices = session_choices()
    valid_ids = {value for _, value in choices}

    if selected not in valid_ids:
        selected = None

    return gr.update(choices=choices, value=selected)


# ---------------------------------------------------------------------------
# Chat completion request helpers
# ---------------------------------------------------------------------------
def build_api_messages(
    history: list[dict[str, Any]],
    system_prompt: str,
) -> list[dict[str, str]]:
    """
    Convert Gradio message history into OpenAI-compatible messages.

    The system prompt from the settings panel is always prepended if non-empty.
    The full user/assistant conversation is included.
    """
    messages: list[dict[str, str]] = []

    prompt = (system_prompt or "").strip()
    if prompt:
        messages.append(
            {
                "role": "system",
                "content": prompt,
            }
        )

    for message in history:
        if not isinstance(message, dict):
            continue

        role = message.get("role")

        if role not in {"system", "user", "assistant"}:
            continue

        content = content_to_text(message.get("content", "")).strip()

        if not content:
            continue

        messages.append(
            {
                "role": role,
                "content": content,
            }
        )

    return messages


def completion_kwargs(
    model: str,
    settings: Any,
    api_messages: list[dict[str, str]],
) -> dict[str, Any]:
    """
    Build keyword arguments for client.chat.completions.create().

    Standard OpenAI schema params:
      - temperature
      - top_p
      - max_tokens

    llama.cpp-specific params sent via extra_body:
      - top_k
      - min_p
      - repeat_penalty
    """
    s = normalize_settings(settings)

    kwargs: dict[str, Any] = {
        "model": model,
        "messages": api_messages,
        "stream": True,
        "temperature": s["temperature"],
        "top_p": s["top_p"],
        "max_tokens": s["max_tokens"],
        "extra_body": {
            "top_k": s["top_k"],
            "min_p": s["min_p"],
            "repeat_penalty": s["repeat_penalty"],
        },
    }

    return kwargs


# ---------------------------------------------------------------------------
# Chat handler
# ---------------------------------------------------------------------------
def respond(
    message: str,
    history: list[dict[str, Any]],
    model_name: str,
    session_id: str | None,
    system_prompt: str,
    temperature: float,
    top_p: float,
    top_k: int,
    min_p: float,
    repeat_penalty: float,
    max_tokens: int,
):
    """
    Gradio event handler.

    Sends the full visible conversation history, plus the current system prompt,
    to llama.cpp every time.
    """
    settings = normalize_settings(
        {
            "system_prompt": system_prompt,
            "temperature": temperature,
            "top_p": top_p,
            "top_k": top_k,
            "min_p": min_p,
            "repeat_penalty": repeat_penalty,
            "max_tokens": max_tokens,
        }
    )

    if not isinstance(message, str) or not message.strip():
        yield gr.update(value=""), history, session_id, gr.skip()
        return

    user_text = message.strip()
    history = list(history or [])

    # Append user message to UI history and save immediately.
    history.append({"role": "user", "content": user_text})
    session_id = save_session(session_id, history, model_name, settings)

    # Clear input, show user message, refresh session dropdown if needed.
    yield gr.update(value=""), history, session_id, dropdown_update(session_id)

    # Add assistant placeholder for streaming.
    history.append({"role": "assistant", "content": ""})
    yield gr.skip(), history, session_id, gr.skip()

    # Exclude empty assistant placeholder from API request.
    api_messages = build_api_messages(history[:-1], settings["system_prompt"])
    model = (model_name or "").strip() or DEFAULT_MODEL

    assistant_text = ""
    last_saved_len = 0

    try:
        if not api_messages:
            raise RuntimeError("No messages to send.")

        kwargs = completion_kwargs(
            model=model,
            settings=settings,
            api_messages=api_messages,
        )

        stream = client.chat.completions.create(**kwargs)

        for chunk in stream:
            choices = getattr(chunk, "choices", None)

            if not choices:
                continue

            delta = getattr(choices[0], "delta", None)
            piece = getattr(delta, "content", None)

            if isinstance(piece, str) and piece:
                assistant_text += piece

                history[-1] = {
                    "role": "assistant",
                    "content": assistant_text,
                }

                # Periodically save during long generations.
                if len(assistant_text) - last_saved_len >= 512:
                    save_session(session_id, history, model, settings)
                    last_saved_len = len(assistant_text)

                yield gr.skip(), history, session_id, gr.skip()

        if not assistant_text:
            history[-1] = {
                "role": "assistant",
                "content": "(empty response)",
            }

        save_session(session_id, history, model, settings)
        yield gr.skip(), history, session_id, dropdown_update(session_id)

    except Exception as exc:
        prefix = assistant_text + ("\n\n" if assistant_text else "")

        history[-1] = {
            "role": "assistant",
            "content": f"{prefix}Request failed: {exc}",
        }

        save_session(session_id, history, model, settings)
        yield gr.skip(), history, session_id, dropdown_update(session_id)

# Session UI actions
# ---------------------------------------------------------------------------
def initialize():
    """
    Load the most recent saved chat when the app starts, if one exists.
    """
    sessions = list_session_metadata()

    if not sessions:
        return (
            [],
            None,
            dropdown_update(None),
            DEFAULT_MODEL,
            *settings_tuple(DEFAULT_SETTINGS),
        )

    first = sessions[0]

    return (
        first.get("messages", []),
        first.get("id"),
        dropdown_update(first.get("id")),
        first.get("model", DEFAULT_MODEL),
        *settings_tuple(first.get("settings")),
    )


def load_selected(
    selected_id: str | None,
    current_id: str | None,
    history: list[dict[str, Any]],
    model: str,
    *current_settings,
):
    """
    Load messages and settings from the selected dropdown session.
    """
    settings = normalize_settings(
        dict(zip(SETTINGS_KEYS, current_settings))
    )

    data = load_session(selected_id)

    if not data:
        return (
            history,
            current_id,
            model,
            *settings_tuple(settings),
        )

    return (
        data.get("messages", []),
        data.get("id"),
        data.get("model", DEFAULT_MODEL),
        *settings_tuple(data.get("settings")),
    )


def new_chat():
    """
    Start a new empty chat.

    No file is written until the first message is sent.
    Current generation settings are intentionally kept, because users usually
    want those to persist across new chats.
    """
    return (
        gr.update(value=""),
        [],
        None,
        dropdown_update(None),
    )


def delete_current(
    selected_id: str | None,
    current_id: str | None,
    history: list[dict[str, Any]],
):
    """
    Delete the selected session.

    If the selected session is also the current one, the UI starts a new chat.
    If it is a different session, the current chat is left untouched.
    """
    target = selected_id or current_id

    if not target:
        return history, current_id, dropdown_update(None)

    path = session_path(target)

    try:
        path.unlink()
    except FileNotFoundError:
        pass
    except Exception:
        pass

    if current_id == target:
        current_id = None
        history = []

    return history, current_id, dropdown_update(current_id)

---

## 4. Optional overrides

You can set the model explicitly if `/v1/models` does not return a good model name:

```bash
uv run env LLAMA_CPP_MODEL=my-model python app.py
```

You can also change the history directory:

```bash
uv run env CHAT_HISTORY_DIR=./my_chats python app.py
```

---

## 5. Notes on parameters

The request is built like this:

```python
client.chat.completions.create(
    model=model,
    messages=messages,
    stream=True,
    temperature=...,
    top_p=...,
    max_tokens=...,
    extra_body={
        "top_k": ...,
        "min_p": ...,
        "repeat_penalty": ...,
    },
)
```

The OpenAI client sends `extra_body` fields as normal top-level JSON fields, not inside a literal `"extra_body"` object.
