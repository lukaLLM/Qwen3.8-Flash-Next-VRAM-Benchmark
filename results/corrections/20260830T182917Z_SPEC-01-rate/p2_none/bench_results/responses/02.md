# [2] chat history

A chat that forgets everything is just a notepad. Here's **Ledger** — the whole thing runs on Node with zero dependencies: every session is its own pretty-printed JSON file on disk, the sidebar is a live index of those files, and there's a JSON rail on the right that shows the exact bytes being written as you type.

```
ledger/
├── server.js          ← zero-dep HTTP server + JSON file store
├── public/
│   ├── index.html
│   ├── styles.css
│   └── app.js
├── package.json
└── data/sessions/     ← created on boot, one .json per conversation
```

/* ---------------------------------------------------------------------------
   Ledger — chat with persistent session history.
   Storage model: data/sessions/<id>.json, one human-readable file per chat.
   Zero dependencies: node:http + node:fs is all it needs.
--------------------------------------------------------------------------- */

const http = require('http');
const fs = require('fs/promises');
const path = require('path');
const crypto = require('crypto');

const ROOT = __dirname;
const PUBLIC_DIR = path.join(ROOT, 'public');
const DATA_DIR = path.join(ROOT, 'data', 'sessions');
const PORT = Number(process.env.PORT || 5173);
const MAX_BODY = 1.5 * 1024 * 1024;

const ID_RE = /^[A-Za-z0-9_-]{3,64}$/;
const ROLES = new Set(['user', 'assistant', 'system']);

const c = {
  dim: (s) => `\x1b[2m${s}\x1b[22m`,
  amber: (s) => `\x1b[38;5;215m${s}\x1b[0m`,
  green: (s) => `\x1b[32m${s}\x1b[0m`,
  teal: (s) => `\x1b[36m${s}\x1b[0m`,
  red: (s) => `\x1b[31m${s}\x1b[0m`,
  bold: (s) => `\x1b[1m${s}\x1b[0m`,
};
const stamp = () => new Date().toTimeString().slice(0, 8);
const log = (verb, msg, color = c.teal) =>
  console.log(`${c.dim(stamp())} ${color(verb.padEnd(6))} ${msg}`);

/* ------------------------------ model stub ------------------------------- */
/* Swap generateReply() for a real provider call — everything else (history,
   persistence, streaming to the client) stays exactly the same.

   const r = await fetch('https://api.anthropic.com/v1/messages', { ... })
   return (await r.json()).content.map(b => b.text).join('');
--------------------------------------------------------------------------- */
async function generateReply({ message, session, stats }) {
  const t = String(message).toLowerCase();
  const words = message.trim().split(/\s+/).filter(Boolean).length;
  const mine = (session.messages || []).filter((m) => m.role === 'assistant').length;
  const hour = new Date().getHours();
  const when = hour < 5 ? 'a late one' : hour < 11 ? 'a morning one'
    : hour < 18 ? 'an afternoon one' : 'an evening one';

  const header = {
    id: session.id,
    title: session.title,
    createdAt: session.createdAt,
    updatedAt: new Date().toISOString(),
    messageCount: (session.messages || []).length + 2,
  };

  let body;
  if (/(json|disk|persist|save|file|store|export)/.test(t)) {
    body = [
      `This conversation isn't in memory — it's a file. Here is the record the moment before I answered:`,
      '',
      '```json',
      JSON.stringify(header, null, 2),
      '```',
      '',
      `Total store: **${stats.count} session file${stats.count === 1 ? '' : 's'}**, ${stats.readable} on disk at \`${stats.dir}\`. Reload the page whenever you like; it's all still there.`,
    ].join('\n');
  } else if (/(delete|wipe|remove|trash|clear)/.test(t)) {
    body = `Deletion is destructive and final on purpose — the file is unlinked from \`data/sessions/\`. Use the two-step confirm on a session row, or the header's trash button. If you delete by mistake, the toast gives you one **Undo** before the JSON is gone for good.`;
  } else if (/(help|what can|how do|capabilit)/.test(t)) {
    body = [
      `You're in session **${session.title}** (${(session.messages || []).length + 1} messages, ${mine} of them mine). Things worth knowing:`,
      '',
      `- **⌘/Ctrl + N** starts a new file`,
      `- **⌘/Ctrl + K** jumps to search — it reads titles *and* message bodies`,
      `- Double-click any session title to rename it`,
      `- The right-hand rail shows the literal bytes being written`,
    ].join('\n');
  } else if (/(code|function|example|snippet)/.test(t)) {
    body = [
      'Every write goes through a temp file and an atomic rename, so a crash mid-save can never leave you with half a conversation:',
      '',
      '```javascript',
      'const tmp = `${file}.${process.pid}.${Date.now()}.tmp`;',
      'await fs.writeFile(tmp, JSON.stringify(session, null, 2));',
      'await fs.rename(tmp, file); // atomic on POSIX',
      '```',
    ].join('\n');
  } else {
    const echo = message.trim().replace(/\s+/g, ' ').slice(0, 110);
    body = `I've got ${stats.count} file${stats.count === 1 ? '' : 's'} of history behind me, and this one goes back ${(session.messages || []).length + 1} messages — ${when}. You wrote ${words} word${words === 1 ? '' : 's'}: _"${echo}"_. Ask me about the **JSON on disk**, **deleting** sessions, or your **options**, or just keep writing — the autosave indicator up top is already ticking.`;
  }

  const openers = ['Filed.', 'Noted.', 'Written down.', 'Logged.', 'Kept.'];
  const opener = mine === 0 ? 'First message in the file —' : openers[mine % openers.length];
  return `${opener} ${body}`;
}

/* ------------------------------- storage -------------------------------- */

const nowISO = () => new Date().toISOString();
const newId = () =>
  `${new Date().toISOString().slice(0, 10).replace(/-/g, '')}-${crypto.randomBytes(4).toString('hex')}`;
const mid = () => `m_${Date.now().toString(36)}${crypto.randomBytes(3).toString('hex')}`;
const cleanTitle = (v, fallback = 'Untitled session') => {
  const s = String(v || '').replace(/\s+/g, ' ').trim();
  return s ? s.slice(0, 120) : fallback;
};

function normalize(raw) {
  if (!raw || typeof raw !== 'object') throw new Error('session must be an object');
  const id = ID_RE.test(String(raw.id || '')) ? String(raw.id) : newId();
  const messages = Array.isArray(raw.messages)
    ? raw.messages
        .filter((m) => m && typeof m === 'object' && ROLES.has(m.role))
        .slice(0, 4000)
        .map((m) => ({
          id: String(m.id || mid()),
          role: m.role,
          content: String(m.content ?? '').slice(0, 40000),
          ts: typeof m.ts === 'string' && !Number.isNaN(Date.parse(m.ts)) ? m.ts : nowISO(),
        }))
    : [];
  const parseDate = (v) =>
    typeof v === 'string' && !Number.isNaN(Date.parse(v)) ? new Date(v).toISOString() : nowISO();
  return {
    id,
    title: cleanTitle(raw.title),
    createdAt: parseDate(raw.createdAt),
    updatedAt: parseDate(raw.updatedAt),
    messages,
  };
}

function summarize(s, bytes) {
  const firstUser = s.messages.find((m) => m.role === 'user');
  return {
    id: s.id,
    title: s.title,
    createdAt: s.createdAt,
    updatedAt: s.updatedAt,
    messageCount: s.messages.length,
    bytes,
    preview: firstUser ? firstUser.content.replace(/\s+/g, ' ').slice(0, 90) : 'No messages yet',
  };
}

function sessionPath(id) {
  if (!ID_RE.test(String(id))) {
    const e = new Error(`invalid session id: ${id}`);
    e.status = 400;
    throw e;
  }
  return path.join(DATA_DIR, `${id}.json`);
}

async function writeSession(file, session) {
  const tmp = `${file}.${process.pid}.${Date.now().toString(36)}.tmp`;
  await fs.writeFile(tmp, JSON.stringify(session, null, 2), 'utf8');
  await fs.rename(tmp, file);
}

async function readSession(id) {
  const file = sessionPath(id);
  const raw = await fs.readFile(file, 'utf8');
  return { session: normalize(JSON.parse(raw)), file, bytes: Buffer.byteLength(raw, 'utf8') };
}

async function listSessions() {
  const names = await fs.readdir(DATA_DIR).catch(() => []);
  const out = [];
  for (const name of names) {
    if (!name.endsWith('.json')) continue;
    const file = path.join(DATA_DIR, name);
    try {
      const raw = await fs.readFile(file, 'utf8');
      out.push(summarize(normalize(JSON.parse(raw)), Buffer.byteLength(raw, 'utf8')));
    } catch (err) {
      console.warn(`${c.dim(stamp())} ${c.red('SKIP ')} ${name} — ${err.message}`);
    }
  }
  return out;
}

async function storeStats() {
  const list = await listSessions();
  const bytes = list.reduce((n, s) => n + s.bytes, 0);
  return { count: list.length, bytes, dir: path.relative(ROOT, DATA_DIR) || DATA_DIR, readable: humanBytes(bytes) };
}

/* ------------------------------- helpers -------------------------------- */

function humanBytes(n) {
  if (!n) return '0 B';
  const u = ['B', 'KB', 'MB', 'GB'];
  const i = Math.min(Math.floor(Math.log(n) / Math.log(1024)), u.length - 1);
  return `${(n / 1024 ** i).toFixed(i === 0 ? 0 : 1)} ${u[i]}`;
}

function send(res, status, body, headers = {}) {
  res.writeHead(status, headers);
  res.end(body);
}
const sendJSON = (res, status, obj) =>
  send(res, status, JSON.stringify(obj), {
    'Content-Type': 'application/json; charset=utf-8',
    'Cache-Control': 'no-store',
  });

function readBody(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    let len = 0;
    req.on('data', (d) => {
      len += d.length;
      if (len > MAX_BODY) {
        const e = new Error('payload too large');
        e.status = 413;
        reject(e);
        req.destroy();
        return;
      }
      chunks.push(d);
    });
    req.on('end', () => {
      if (!chunks.length) return resolve({});
      try {
        resolve(JSON.parse(Buffer.concat(chunks).toString('utf8')));
      } catch {
        const e = new Error('body must be valid JSON');
        e.status = 400;
        reject(e);
      }
    });
    req.on('error', reject);
  });
}

const MIME = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.ico': 'image/x-icon',
  '.woff2': 'font/woff2',
};

async function serveStatic(res, pathname) {
  const rel = pathname === '/' ? '/index.html' : pathname;
  const file = path.normalize(path.join(PUBLIC_DIR, rel));
  if (!file.startsWith(PUBLIC_DIR)) return send(res, 403, 'Forbidden');
  try {
    const data = await fs.readFile(file);
    send(res, 200, data, {
      'Content-Type': MIME[path.extname(file)] || 'application/octet-stream',
      'Cache-Control': 'no-cache',
    });
  } catch {
    send(res, 404, 'Not found', { 'Content-Type': 'text/plain' });
  }
}

/* ------------------------------- API ------------------------------------ */

async function api(req, res, url) {
  const p = url.pathname.replace(/\/+$/, '') || '/';
  const method = req.method;
  let m;

  // GET /api/sessions                      → summaries, newest first
  if (method === 'GET' && p === '/api/sessions') {
    const list = await listSessions();
    list.sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
    return sendJSON(res, 200, { sessions: list, stats: await storeStats() });
  }

  // POST /api/sessions                     → create (client may suggest an id)
  if (method === 'POST' && p === '/api/sessions') {
    const body = await readBody(req);
    const session = normalize({
      id: body.id && ID_RE.test(String(body.id)) ? String(body.id) : newId(),
      title: body.title,
      createdAt: body.createdAt,
      updatedAt: body.createdAt,
      messages: body.messages,
    });
    const file = sessionPath(session.id);
    await writeSession(file, session);
    log('WRITE', `${c.amber(session.id)}.json ${c.dim('· new ·')} ${c.bold(session.title)}`);
    return sendJSON(res, 201, { session, ...await wrap(session, file) });
  }

  // DELETE /api/sessions                   → wipe the whole store
  if (method === 'DELETE' && p === '/api/sessions') {
    const names = await fs.readdir(DATA_DIR);
    await Promise.all(names.filter((n) => n.endsWith('.json')).map((n) => fs.rm(path.join(DATA_DIR, n), { force: true })));
    log('WIPE', `${c.red(names.length + ' file(s) removed')}`);
    return sendJSON(res, 200, { ok: true, stats: await storeStats() });
  }

  // GET /api/export                        → single downloadable archive
  if (method === 'GET' && p === '/api/export') {
    const names = (await fs.readdir(DATA_DIR)).filter((n) => n.endsWith('.json'));
    const all = [];
    for (const n of names) {
      try { all.push(normalize(JSON.parse(await fs.readFile(path.join(DATA_DIR, n), 'utf8')))); } catch {}
    }
    all.sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
    log('DUMP', `${c.green(all.length + ' sessions exported')}`);
    return send(res, 200, JSON.stringify({ exportedAt: nowISO(), count: all.length, sessions: all }, null, 2), {
      'Content-Type': 'application/json; charset=utf-8',
      'Content-Disposition': `attachment; filename="ledger-${new Date().toISOString().slice(0, 10)}.json"`,
    });
  }

  // POST /api/reply                        → model stub (client persists the result)
  if (method === 'POST' && p === '/api/reply') {
    const body = await readBody(req);
    const message = String(body.message || '').slice(0, 20000);
    if (!message.trim()) return sendJSON(res, 400, { error: 'message is required' });
    let session = { id: 'draft', title: 'Draft', messages: [] };
    if (body.id) {
      try { session = (await readSession(body.id)).session; } catch {}
    }
    const reply = await generateReply({ message, session, stats: await storeStats() });
    log('REPLY', `${c.dim(`${message.length}→${reply.length} chars`)}`);
    return sendJSON(res, 200, { reply, ts: nowISO() });
  }

  // /api/sessions/:id
  m = p.match(/^\/api\/sessions\/([^/]+)$/);
  if (m) {
    const id = decodeURIComponent(m[1]);

    if (method === 'GET') {
      const { session, file, bytes } = await readSession(id);
      if (url.searchParams.get('download') === '1') {
        return send(res, 200, JSON.stringify(session, null, 2), {
          'Content-Type': 'application/json; charset=utf-8',
          'Content-Disposition': `attachment; filename="${session.id}.json"`,
        });
      }
      return sendJSON(res, 200, { session, file: path.relative(ROOT, file), bytes });
    }

    if (method === 'PUT' || method === 'PATCH') {
      const body = await readBody(req);
      const merged = normalize({
        id,
        title: body.title,
        createdAt: body.createdAt,
        messages: body.messages,
      });
      const touch = !(url.searchParams.get('touch') === '0' || body.touch === false);
      merged.updatedAt = touch ? nowISO() : (body.updatedAt || merged.updatedAt);
      if (!Array.isArray(body.messages)) {
        // metadata-only patch: keep what's on disk
        const disk = (await readSession(id)).session;
        merged.messages = disk.messages;
        merged.createdAt = disk.createdAt;
        if (!body.title) merged.title = disk.title;
      }
      const file = sessionPath(id);
      await writeSession(file, merged);
      const bytes = (await fs.stat(file)).size;
      log('SAVE', `${c.amber(id)}.json ${c.dim(`· ${humanBytes(bytes)} · ${merged.messages.length} msg`)}`);
      return sendJSON(res, 200, { session: merged, file: path.relative(ROOT, file), bytes });
    }

    if (method === 'DELETE') {
      const file = sessionPath(id);
      const raw = await fs.readFile(file, 'utf8');
      await fs.rm(file, { force: true });
      log('DELETE', `${c.red(id + '.json')} ${c.dim('· restorable via POST for the length of the toast')}`, c.red);
      return sendJSON(res, 200, { ok: true, restored: JSON.parse(raw) });
    }
  }

  const e = new Error(`no route: ${method} ${p}`);
  e.status = 404;
  throw e;
}

const wrap = async (session, file) => ({
  file: path.relative(ROOT, file),
  bytes: (await fs.stat(file)).size,
  stats: await storeStats(),
});

/* ------------------------------ server ---------------------------------- */

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'GET,POST,PUT,PATCH,DELETE,OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

  try {
    if (req.method === 'OPTIONS') return send(res, 204, '');
    if (url.pathname.startsWith('/api/')) return await api(req, res, url);
    return await serveStatic(res, url.pathname);
  } catch (err) {
    const status = err.status || 500;
    if (status !== 404) console.error(`${c.dim(stamp())} ${c.red('ERROR')} ${err.message}`);
    if (!res.headersSent) sendJSON(res, status, { error: err.message });
    else res.end();
  }
});

(async function boot() {
  await fs.mkdir(DATA_DIR, { recursive: true });
  const stats = await storeStats();
  server.listen(PORT, () => {
    console.log(`\n${c.bold('  LEDGER')} ${c.dim('— conversations, filed as JSON')}`);
    console.log(`  ${c.dim('store')}  ${DATA_DIR} ${c.dim(`(${stats.count} session${stats.count === 1 ? '' : 's'}, ${stats.readable})`)}`);
    console.log(`  ${c.dim('serve ')}  ${c.green(`http://localhost:${PORT}`)}\n`);
  });
})();

/* ── markdown-lite ──────────────────────────────────────────────── */
function inline(s) {
  return esc(s)
    .replace(/`([^`\n]+)`/g, '<code>$1</code>')
    .replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>')
    .replace(/(^|[\s(])_([^_\n]+)_(?=[\s).,!?:;]|$)/g, '$1<em>$2</em>')
    .replace(/\n/g, '<br>');
}
function md(text) {
  const src = String(text ?? '');
  const re = /```([\w-]*)\n?([\s\S]*?)```/g;
  let out = '', last = 0, m;
  while ((m = re.exec(src))) {
    out += inline(src.slice(last, m.index));
    out += `<pre data-lang="${esc(m[1] || 'code')}"><code>${esc(m[2].replace(/\n$/, ''))}</code></pre>`;
    last = m.index + m[0].length;
  }
  return out + inline(src.slice(last));
}

async function saveNow() {
  const s = state.current;
  if (!s) return;
  clearTimeout(state.saveTimer);
  setStatus('saving', 'saving…');
  try {
    const payload = { id: s.id, title: s.title, createdAt: s.createdAt, messages: s.messages };
    if (state.pending) {
      const r = await api('/api/sessions', { method: 'POST', body: JSON.stringify(payload) });
      state.pending = false;
      state.current = r.session; state.file = r.file; state.fileBytes = r.bytes;
    } else {
      const r = await api(`/api/sessions/${s.id}`, { method: 'PUT', body: JSON.stringify(payload) });
      state.current = r.session; state.file = r.file; state.fileBytes = r.bytes;

    setStatus('saved', `saved ${clock()}`);

}

async function removeChat(id) {
  const local = state.sessions.find((s) => s.id === id);
  const isCurrent = state.current && state.current.id === id;
  if (state.pending && isCurrent) { state.current = blank(); state.renderedFor = null; render(); return; }
  try {
    const r = await api(`/api/sessions/${id}`, { method: 'DELETE' });

}

function typewrite(node, text) {
  if (!node) return Promise.resolve();
  return new Promise((done) => {
    let i = 0;
    const tick = () => {
      i = Math.min(text.length, i + Math.max(3, Math.round(text.length / 160)));
      node.innerHTML = md(text.slice(0, i));
      nearBottom() && scrollDown();
      if (i < text.length) requestAnimationFrame(tick); else done();
    };
    requestAnimationFrame(tick);
  });
}

Chat whose memory is a folder of JSON files.

      ]
    }

## API
| method | path | effect |
| --- | --- | --- |
| GET | `/api/sessions` | summaries + store stats |
| POST | `/api/sessions` | create (`{id?, title?, messages?}`) |
| GET | `/api/sessions/:id` | full session — `?download=1` for an attachment |
| PUT/PATCH | `/api/sessions/:id` | overwrite; `?touch=0` skips `updatedAt` (renames) |
| DELETE | `/api/sessions/:id` | unlink, returns the deleted object for Undo |
| DELETE | `/api/sessions` | empty the store |
| GET | `/api/export` | all sessions as one downloadable archive |
| POST | `/api/reply` | model stub — swap `generateReply()` in `server.js` for a real call |

Writes go to a temp file then `rename()`, so a crash can't truncate a conversation.
Turn off the stub by replacing `generateReply()`; history, autosave and the UI are unchanged.

**How persistence actually behaves**

- A new chat is *local only* until the first message — no empty files pile up in the store.
- Autosave is debounced at 650 ms and reported honestly in the header pill: `not saved yet` → `writing…` → `saved 14:02:07`.
- `beforeunload` flushes via `sendBeacon`, so closing the tab mid-thought doesn't lose the last exchange.
- Delete is two-step everywhere (hover-row trash, header trash, "Clear store"), and the returned file object feeds a **Undo** toast that POSTs it straight back.
- The right rail renders the literal JSON of the open session — path, byte count, truncated message array — and flashes each time the file changes size.
